124 std::vector<SpatialChunk*> CullChunks(const Math::Frustumf& fr) const noexcept
125 {
126 std::vector<SpatialChunk*> out;
127 const uint32_t w = grid.width(), d = grid.height();
128 const float cell = float(chunkSize);
129 const float exz = 0.5f * cell; // x,z extents (cell half size)
130
131 // 1) Coarse: frustum AABB -> candidate cell range (XZ)
132 const Math::AABB3f frAabb = ComputeFrustumAABB_WS(fr);
133 using Signed = long long;
134 auto floor_div_x = [&](float worldX) noexcept -> Signed {
135 return static_cast<Signed>(std::floor(double(worldX - levelOriginWS.x) / double(cell)));
136 };
137 auto floor_div_z = [&](float worldZ) noexcept -> Signed {
138 return static_cast<Signed>(std::floor(double(worldZ - levelOriginWS.z) / double(cell)));
139 };
140
141 Signed ix0 = floor_div_x(frAabb.lb.x);
142 Signed ix1 = floor_div_x(frAabb.ub.x);
143 Signed iz0 = floor_div_z(frAabb.lb.z);
144 Signed iz1 = floor_div_z(frAabb.ub.z);
145
146 // マージン：フラスタムのAABBはセル境界に対して誤差がある可能性があるため、少し余裕を持たせる
147 const Signed margin = 1; // まずは1セル。問題が残るなら増やす
148
149 ix0 -= margin; ix1 += margin;
150 iz0 -= margin; iz1 += margin;
151
152 const Signed W = static_cast<Signed>(w);
153 const Signed D = static_cast<Signed>(d);
154 if (ix0 > ix1) std::swap(ix0, ix1);
155 if (iz0 > iz1) std::swap(iz0, iz1);
156
157 // If frustum AABB doesn't overlap this grid at all, early out.
158 if (ix1 < 0 || iz1 < 0 || ix0 >= W || iz0 >= D) {
159 return out;
160 }
161
162 ix0 = std::clamp<Signed>(ix0, 0, W - 1);
163 ix1 = std::clamp<Signed>(ix1, 0, W - 1);
164 iz0 = std::clamp<Signed>(iz0, 0, D - 1);
165 iz1 = std::clamp<Signed>(iz1, 0, D - 1);
166
167 // 2) Precise: per-chunk real AABB (if available). Otherwise fallback to cell AABB.
168 for (Signed z = iz0; z <= iz1; ++z) {
169 for (Signed x = ix0; x <= ix1; ++x) {
170 auto& chunk = grid(static_cast<ChunkSizeType>(x), static_cast<ChunkSizeType>(z));
171 if (chunk.GetEntityManager().GetEntityCount() == 0) continue;
172
173 // 最終判定：チャンクの実AABBがあるならそれを使う
174 if (chunk.HasStaticBounds()) {
175 if (fr.IntersectsAABB(chunk.GetStaticBoundsWS())) {
176 out.push_back(const_cast<SpatialChunk*>(&chunk));
177 }
178 continue;
179 }
180
181 // フォールバック（実AABB無し）：従来のセルAABB + YOverlap
182 const float cx = levelOriginWS.x + (float(x) + 0.5f) * cell;
183 const float cz = levelOriginWS.z + (float(z) + 0.5f) * cell;
184
185 float cyEff, eyEff;
186 if (!Math::Frustumf::ComputeYOverlapAtXZ(fr, cx, cz, -chunkSize, chunkSize, cyEff, eyEff)) {
187 continue;
188 }
189
190 const Math::Vec3f center{ cx, cyEff, cz };
191 const Math::Vec3f extent{ exz, eyEff, exz };
192 if (fr.IntersectsAABB(center, extent)) {
193 out.push_back(const_cast<SpatialChunk*>(&chunk));
194 }
195 }
196 }
197 return out;
198 }

277 std::vector<SpatialChunk*> CullChunks(const Math::Vec3f& center, float radius) const noexcept
278 {
279 std::vector<SpatialChunk*> out;
280 if (!std::isfinite(radius) || radius <= 0.0f) {
281 radius = 0.0f;
282 }
283 const float r = (radius < 0.0f) ? 0.0f : radius;
284 const float r2 = r * r;
285
286 using Signed = long long;
287 const float cell = float(chunkSize);
288 const float e = 0.5f * cell;
289
290 auto floor_div = [&](float v) -> Signed {
291 return static_cast<Signed>(std::floor(double(v) / double(cell)));
292 };
293
294 Signed ix0 = floor_div(center.x - r);
295 Signed iz0 = floor_div(center.z - r);
296 Signed ix1 = floor_div(center.x + r);
297 Signed iz1 = floor_div(center.z + r);
298
299 // マージン：フラスタムのAABBはセル境界に対して誤差がある可能性があるため、少し余裕を持たせる
300 const Signed margin = 1; // まずは1セル。問題が残るなら増やす
301
302 ix0 -= margin; ix1 += margin;
303 iz0 -= margin; iz1 += margin;
304
305 const Signed w = static_cast<Signed>(grid.width());
306 const Signed d = static_cast<Signed>(grid.height());
307
308 ix0 = std::clamp<Signed>(ix0, 0, w - 1);
309 iz0 = std::clamp<Signed>(iz0, 0, d - 1);
310 ix1 = std::clamp<Signed>(ix1, 0, w - 1);
311 iz1 = std::clamp<Signed>(iz1, 0, d - 1);
312
313 out.reserve(static_cast<size_t>((ix1 - ix0 + 1) * (iz1 - iz0 + 1)));
314
315 auto dist2_point_aabb_xz = [&](const Math::Vec3f& p, float cx, float cz, float ex, float ez) -> float {
316 auto clamp = [](float v, float lo, float hi) { return v < lo ? lo : (v > hi ? hi : v); };
317 const float qx = clamp(p.x, cx - ex, cx + ex);
318 const float qz = clamp(p.z, cz - ez, cz + ez);
319 const float dx = p.x - qx, dz = p.z - qz;
320 return dx * dx + dz * dz;
321 };
322
323 for (Signed z = iz0; z <= iz1; ++z) {
324 for (Signed x = ix0; x <= ix1; ++x) {
325 const float cx = levelOriginWS.x + (x + 0.5f) * cell;
326 const float cz = levelOriginWS.z + (z + 0.5f) * cell;
327
328 if (dist2_point_aabb_xz(center, cx, cz, e, e) > r2) continue;
329
330 auto& chunk = grid(ChunkSizeType(x), ChunkSizeType(z));
331 if (chunk.GetEntityManager().GetEntityCount() > 0)
332 out.push_back(const_cast<SpatialChunk*>(&chunk));
333 }
334 }
335 return out;
336 }

338 static inline float Dist2PointAABB3D(const Math::Vec3f& p,
339 const Math::Vec3f& c, // AABB center
340 const Math::Vec3f& e) // AABB extent
341 {
342 auto clamp = [](float v, float lo, float hi) { return v < lo ? lo : (v > hi ? hi : v); };
343 const float qx = clamp(p.x, c.x - e.x, c.x + e.x);
344 const float qy = clamp(p.y, c.y - e.y, c.y + e.y);
345 const float qz = clamp(p.z, c.z - e.z, c.z + e.z);
346 const float dx = p.x - qx, dy = p.y - qy, dz = p.z - qz;
347 return dx * dx + dy * dy + dz * dz;
348 }

350 // 可視チャンクを「カメラ位置から近い順」に上位K件返す
351 std::vector<SpatialChunk*> CullChunksNear(const Math::Frustumf& fr,
352 const Math::Vec3f& camPos,
353 size_t maxCount = (std::numeric_limits<size_t>::max)()) const noexcept
354 {
355 std::vector<SpatialChunk*> out;
356
357 struct Item { SpatialChunk* sc; float d2; };
358 std::vector<Item> items; items.reserve(128);
359
360 const uint32_t w = grid.width(), d = grid.height();
361 const float cell = float(chunkSize);
362 const float exz = 0.5f * cell; // x,z extents (cell half size)
363
364 // 1) Coarse: frustum AABB -> candidate cell range (XZ)
365 const Math::AABB3f frAabb = ComputeFrustumAABB_WS(fr);
366 using Signed = long long;
367 auto floor_div_x = [&](float worldX) noexcept -> Signed {
368 return static_cast<Signed>(std::floor(double(worldX - levelOriginWS.x) / double(cell)));
369 };
370 auto floor_div_z = [&](float worldZ) noexcept -> Signed {
371 return static_cast<Signed>(std::floor(double(worldZ - levelOriginWS.z) / double(cell)));
372 };
373
374 Signed ix0 = floor_div_x(frAabb.lb.x);
375 Signed ix1 = floor_div_x(frAabb.ub.x);
376 Signed iz0 = floor_div_z(frAabb.lb.z);
377 Signed iz1 = floor_div_z(frAabb.ub.z);
378
379 // マージン：フラスタムのAABBはセル境界に対して誤差がある可能性があるため、少し余裕を持たせる
380 const Signed margin = 1; // まずは1セル。問題が残るなら増やす
381
382 ix0 -= margin; ix1 += margin;
383 iz0 -= margin; iz1 += margin;
384
385 const Signed W = static_cast<Signed>(w);
386 const Signed D = static_cast<Signed>(d);
387 if (ix0 > ix1) std::swap(ix0, ix1);
388 if (iz0 > iz1) std::swap(iz0, iz1);
389
390 // If frustum AABB doesn't overlap this grid at all, early out.
391 if (ix1 < 0 || iz1 < 0 || ix0 >= W || iz0 >= D) {
392 return out;
393 }
394
395 ix0 = std::clamp<Signed>(ix0, 0, W - 1);
396 ix1 = std::clamp<Signed>(ix1, 0, W - 1);
397 iz0 = std::clamp<Signed>(iz0, 0, D - 1);
398 iz1 = std::clamp<Signed>(iz1, 0, D - 1);
399
400 // 2) Precise: per-chunk real AABB (if available). Otherwise fallback to cell AABB.
401 for (Signed z = iz0; z <= iz1; ++z) {
402 for (Signed x = ix0; x <= ix1; ++x) {
403 auto& chunk = grid(static_cast<ChunkSizeType>(x), static_cast<ChunkSizeType>(z));
404 if (chunk.GetEntityManager().GetEntityCount() == 0) continue;
405
406 // 最終判定：チャンクの実AABBがあるならそれを使う
407 if (chunk.HasStaticBounds()) {
408 const auto& bounds = chunk.GetStaticBoundsWS();
409 if (fr.IntersectsAABB(bounds)) {
410 const float d2 = Dist2PointAABB3D(camPos, bounds.center(), bounds.extent());
411 if (chunk.GetEntityManager().GetEntityCount() > 0)
412 items.push_back({ const_cast<SpatialChunk*>(&chunk), d2 });
413 }
414 continue;
415 }
416
417 // フォールバック（実AABB無し）：従来のセルAABB + YOverlap
418 const float cx = levelOriginWS.x + (float(x) + 0.5f) * cell;
419 const float cz = levelOriginWS.z + (float(z) + 0.5f) * cell;
420
421 float cyEff, eyEff;
422 if (!Math::Frustumf::ComputeYOverlapAtXZ(fr, cx, cz, -chunkSize, chunkSize, cyEff, eyEff)) {
423 continue;
424 }
425
426 const Math::Vec3f center{ cx, cyEff, cz };
427 const Math::Vec3f extent{ exz, eyEff, exz };
428 if (fr.IntersectsAABB(center, extent)) {
429 const float d2 = Dist2PointAABB3D(camPos, center, extent);
430 items.push_back({ const_cast<SpatialChunk*>(&chunk), d2 });
431 }
432 }
433 }
434
435 if (items.empty()) return {};
436 const size_t K = (std::min)(maxCount, items.size());
437
438 auto cmp = [](const Item& a, const Item& b) {
439 return a.d2 < b.d2;
440 };
441
442 if (items.size() > maxCount) {
443 const size_t K = maxCount;
444 std::nth_element(items.begin(), items.begin() + K, items.end(), cmp);
445 items.resize(K);
446 }
447 std::sort(items.begin(), items.end(), cmp);
448
449 out.reserve(K);
450 for (auto& it : items) out.push_back(it.sc);
451 return out;
452 }

464 uint32_t CullChunkLine(const Math::Frustumf& fr,
465 Math::Vec3f cp, float hy, Debug::LineVertex* outLine,
466 uint32_t capacity, uint32_t displayCount) const noexcept
467 {
468 // 残りは従来のセルAABBで表示
469 const uint32_t w = grid.width(), d = grid.height();
470 const float cell = float(chunkSize);
471 const float exz = 0.5f * cell; // x,z は各セルの半径
472
473 uint32_t validCount = 0;
474 float maxLengthD2 = std::pow(displayCount * chunkSize, 2.0f);
475
476 Math::Vec3f localCamPos = cp - levelOriginWS;
477 const uint32_t centerX = static_cast<uint32_t>(std::clamp<int32_t>(static_cast<int32_t>(std::floor(localCamPos.x / cell)), 0, static_cast<int32_t>(w - 1)));
478 const uint32_t centerZ = static_cast<uint32_t>(std::clamp<int32_t>(static_cast<int32_t>(std::floor(localCamPos.z / cell)), 0, static_cast<int32_t>(d - 1)));
479
480 for (uint32_t z = 0; z < d; ++z) {
481 for (uint32_t x = 0; x < w; ++x) {
482 const float cx = levelOriginWS.x + (x + 0.5f) * cell;
483 const float cz = levelOriginWS.z + (z + 0.5f) * cell;
484
485 Math::Vec2 vec = { cx - cp.x, cz - cp.z };
486 float len = vec.lengthSquared();
487
488 if (len > maxLengthD2) continue; // 表示距離外
489
490 if (capacity - validCount < 30) break; // ラインバッファ不足
491
492 float cyEff, eyEff;
493 if (!Math::Frustumf::ComputeYOverlapAtXZ(fr, cx, cz,
494 std::numeric_limits<float>::lowest(), (std::numeric_limits<float>::max)(), cyEff, eyEff)) {
495 continue; // 縦に一切重ならない → 可視になり得ない
496 }
497
498 const Math::Vec3f center{ cx, cp.y, cz };
499 const Math::Vec3f extent{ exz, hy, exz };
500
501 if (fr.IntersectsAABB(center, extent))
502 {
503 uint32_t rgb = Math::LerpColor(0xFFFFFFFF, 0x000000FF, len / maxLengthD2);
504
505 outLine[validCount++] = { Math::Vec3f(center.x - extent.x, center.y - extent.y, center.z - extent.z), rgb };
506 outLine[validCount++] = { Math::Vec3f(center.x - extent.x, center.y + extent.y, center.z - extent.z), rgb };
507 outLine[validCount++] = { Math::Vec3f(center.x + extent.x, center.y - extent.y, center.z - extent.z), rgb };
508 outLine[validCount++] = { Math::Vec3f(center.x + extent.x, center.y + extent.y, center.z - extent.z), rgb };
509 outLine[validCount++] = { Math::Vec3f(center.x - extent.x, center.y - extent.y, center.z + extent.z), rgb };
510 outLine[validCount++] = { Math::Vec3f(center.x - extent.x, center.y + extent.y, center.z + extent.z), rgb };
511
512 auto& chunk = grid(x, z);
513 if (chunk.HasStaticBounds())
514 {
515 Math::AABB3f aabb = chunk.GetStaticBoundsWS();
516
517 const uint32_t baseColor = (x == centerX && z == centerZ) ?
518 0xFF0000FF : 0x00FF00FF;
519
520 const uint32_t rgba = Math::LerpColor(baseColor, 0x000000FF, len / maxLengthD2);
521 auto lineVertices = Debug::MakeAABBLineVertices(aabb, rgba);
522 for (const auto& v : lineVertices) {
523 outLine[validCount++] = v;
524 }
525 }
526 }
527 }
528 }
529
530 return validCount;
531 }

539 void ReloadCell(uint32_t cx, uint32_t cy, SpatialChunkRegistry& reg) {
540 SpatialChunk& cell = grid(cx, cy);

548 }
