94 enum class RefineReason : uint32_t {
95 None = 0,
96 MidBand = 1 << 0, // 画面占有の“曖昧帯域”
97 Elongated = 1 << 1, // AABB が細長い（球の過大評価が大きい）
98 NearClip = 1 << 2, // 近クリップ近傍
99 NearEdge = 1 << 3, // NDC 端（±1 付近）
100 LodBoundary = 1 << 4 // LOD しきい値の近傍（確定取りしたい）
101 };

104 inline RefineReason operator|(RefineReason a, RefineReason b) {
105 return static_cast<RefineReason>(static_cast<uint32_t>(a) | static_cast<uint32_t>(b));
106 }
107 inline RefineReason& operator|=(RefineReason& a, RefineReason b) { return a = (a | b); }
108 inline bool any(RefineReason r) { return static_cast<uint32_t>(r) != 0; }
