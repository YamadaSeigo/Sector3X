271 // ===== 位置/スカラー: lerp(a,b,alpha) を最速更新 =====
272 inline void update_scalar_lerp(
273 float* __restrict dst,
274 const float* __restrict a,
275 const float* __restrict b,
276 size_t N,
277 float alpha,
278 const UpdateSet& upd,
279 float sparseThreshold = 0.30f) // 30% 未満なら疎とみなす
280 {
281#if defined(__AVX2__)
282 const __m256 vA = _mm256_set1_ps(alpha);
283
284 const bool hasIdx = (upd.indices && upd.updateCount > 0);
285 const float density = (N == 0) ? 0.f : float(upd.updateCount) / float(N);
286
287 if (!hasIdx || density >= sparseThreshold) {
288 // ==== 密ルート: 全走査 + ブレンド ====
289 const __m256i vZeroI = _mm256_setzero_si256();
290 size_t i = 0;
291 for (; i + 8 <= N; i += 8) {
292 const __m256 va = _mm256_loadu_ps(a + i);
293 const __m256 vb = _mm256_loadu_ps(b + i);
294 const __m256 vold = _mm256_loadu_ps(dst + i);
295 const __m256 vlerp = _mm256_fmadd_ps(_mm256_sub_ps(vb, va), vA, va);
296
297 __m256 vm = _mm256_castsi256_ps(vZeroI); // デフォルトは 0 マスク
298 if (upd.mask01) {
299 const __m256i mi = _mm256_cmpgt_epi32(
300 _mm256_loadu_si256((const __m256i*)(upd.mask01 + i)),
301 vZeroI);
302 vm = _mm256_castsi256_ps(mi);
303 }
304 else {
305 // マスクが無ければ全更新
306 vm = _mm256_castsi256_ps(_mm256_cmpeq_epi32(vZeroI, vZeroI));
307 }
308 const __m256 vout = _mm256_blendv_ps(vold, vlerp, vm);
309 _mm256_storeu_ps(dst + i, vout);
310 }
311 for (; i < N; ++i) {
312 const float lerp = a[i] + (b[i] - a[i]) * alpha;
313 if (!upd.mask01 || upd.mask01[i]) dst[i] = lerp;
314 }
315 }
316 else {
317 // ==== 疎ルート: indices を使って gather 更新 ====
318 const int scale = 4; // float*
319 size_t i = 0;
320 for (; i + 8 <= upd.updateCount; i += 8) {
321 __m256i vidx = _mm256_loadu_si256((const __m256i*)(upd.indices + i));
322 __m256 va = _mm256_i32gather_ps(a, vidx, scale);
323 __m256 vb = _mm256_i32gather_ps(b, vidx, scale);
324 __m256 vlerp = _mm256_fmadd_ps(_mm256_sub_ps(vb, va), vA, va);
325
326 // scatter（ネイティブ命令が無い世代）→ 1要素ずつストア
327 alignas(32) uint32_t idxbuf[8];
328 _mm256_store_si256((__m256i*)idxbuf, vidx);
329 alignas(32) float tmp[8];
330 _mm256_store_ps(tmp, vlerp);
331 for (int k = 0; k < 8; ++k) dst[idxbuf[k]] = tmp[k];
332 }
333 for (; i < upd.updateCount; ++i) {
334 const uint32_t id = upd.indices[i];
335 dst[id] = a[id] + (b[id] - a[id]) * alpha;
336 }
337 }
338#else
339 // AVX2 なし → 簡易フォールバック
340 const bool hasIdx = (upd.indices && upd.updateCount > 0);
341 const float density = (N == 0) ? 0.f : float(upd.updateCount) / float(N);
342 if (!hasIdx || density >= sparseThreshold) {
343 for (size_t i = 0; i < N; ++i) {
344 const float lerp = a[i] + (b[i] - a[i]) * alpha;
345 if (!upd.mask01 || upd.mask01[i]) dst[i] = lerp;
346 }
347 }
348 else {
349 for (size_t j = 0; j < upd.updateCount; ++j) {
350 const uint32_t id = upd.indices[j];
351 dst[id] = a[id] + (b[id] - a[id]) * alpha;
352 }
353 }
354#endif
355 }

357 // ===== クォータニオン: 最短経路 nlerp + 正規化 =====
358 // prev(ax..aw), curr(bx..bw) → dst(qx..qw)
359 inline void update_quat_nlerp_shortest(
360 float* __restrict qx, float* __restrict qy, float* __restrict qz, float* __restrict qw,
361 const float* __restrict ax, const float* __restrict ay, const float* __restrict az, const float* __restrict aw,
362 const float* __restrict bx, const float* __restrict by, const float* __restrict bz, const float* __restrict bw,
363 size_t N, float alpha,
364 const UpdateSet& upd,
365 float sparseThreshold = 0.30f)
366 {
367#if defined(__AVX2__)
368 const __m256 vA = _mm256_set1_ps(alpha);
369 const bool hasIdx = (upd.indices && upd.updateCount > 0);
370 const float density = (N == 0) ? 0.f : float(upd.updateCount) / float(N);
371
372 auto do_one = [&](uint32_t id) {
373 // 最短経路: dot(a,b) < 0 → b を反転
374 float dx = ax[id] * bx[id] + ay[id] * by[id] + az[id] * bz[id] + aw[id] * bw[id];
375 float sx = bx[id], sy = by[id], sz = bz[id], sw = bw[id];
376 if (dx < 0.f) { sx = -sx; sy = -sy; sz = -sz; sw = -sw; }
377 float x = ax[id] + (sx - ax[id]) * alpha;
378 float y = ay[id] + (sy - ay[id]) * alpha;
379 float z = az[id] + (sz - az[id]) * alpha;
380 float w = aw[id] + (sw - aw[id]) * alpha;
381 float invL = 1.0f / std::sqrt(x * x + y * y + z * z + w * w);
382 qx[id] = x * invL; qy[id] = y * invL; qz[id] = z * invL; qw[id] = w * invL;
383 };
384
385 if (!hasIdx || density >= sparseThreshold) {
386 const __m256 vHalf = _mm256_set1_ps(0.5f); //（必要なら rsqrt 改良に使用）
387 size_t i = 0;
388 for (; i + 8 <= N; i += 8) {
389 // a と b のロード
390 __m256 axv = _mm256_loadu_ps(ax + i), ayv = _mm256_loadu_ps(ay + i), azv = _mm256_loadu_ps(az + i), awv = _mm256_loadu_ps(aw + i);

393 // dot(a,b)
394 __m256 dot = _mm256_fmadd_ps(axv, bxv,
395 _mm256_fmadd_ps(ayv, byv,
396 _mm256_fmadd_ps(azv, bzv, _mm256_mul_ps(awv, bwv))));
397
398 // dot<0 → b を反転
399 __m256 maskNeg = _mm256_castsi256_ps(_mm256_cmpgt_epi32(_mm256_setzero_si256(), _mm256_castps_si256(dot)));
400 // 符号反転は XOR( signMask )
401 const __m256 signMask = _mm256_set1_ps(-0.0f);
402 bxv = _mm256_xor_ps(bxv, _mm256_and_ps(signMask, maskNeg));
403 byv = _mm256_xor_ps(byv, _mm256_and_ps(signMask, maskNeg));
404 bzv = _mm256_xor_ps(bzv, _mm256_and_ps(signMask, maskNeg));
405 bwv = _mm256_xor_ps(bwv, _mm256_and_ps(signMask, maskNeg));
406
407 // lerp
408 __m256 lx = _mm256_fmadd_ps(_mm256_sub_ps(bxv, axv), vA, axv);
409 __m256 ly = _mm256_fmadd_ps(_mm256_sub_ps(byv, ayv), vA, ayv);
410 __m256 lz = _mm256_fmadd_ps(_mm256_sub_ps(bzv, azv), vA, azv);
411 __m256 lw = _mm256_fmadd_ps(_mm256_sub_ps(bwv, awv), vA, awv);

415 _mm256_fmadd_ps(ly, ly,
416 _mm256_fmadd_ps(lz, lz, _mm256_mul_ps(lw, lw))));

419 lx = _mm256_mul_ps(lx, invL);
420 ly = _mm256_mul_ps(ly, invL);
421 lz = _mm256_mul_ps(lz, invL);
422 lw = _mm256_mul_ps(lw, invL);
423
424 if (upd.mask01) {
425 const __m256i mi = _mm256_cmpgt_epi32(
426 _mm256_loadu_si256((const __m256i*)(upd.mask01 + i)),
427 _mm256_setzero_si256());

432 _mm256_storeu_ps(qy + i, _mm256_blendv_ps(oy, ly, vm));
433 _mm256_storeu_ps(qz + i, _mm256_blendv_ps(oz, lz, vm));

436 else {

438 _mm256_storeu_ps(qy + i, ly);
439 _mm256_storeu_ps(qz + i, lz);
440 _mm256_storeu_ps(qw + i, lw);

442 }
443 for (; i < N; ++i) { if (!upd.mask01 || upd.mask01[i]) do_one((uint32_t)i); }

458 __m256 bwv = _mm256_i32gather_ps(bw, vidx, scale);

464 byv = _mm256_xor_ps(byv, _mm256_and_ps(signMask, maskNeg));
465 bzv = _mm256_xor_ps(bzv, _mm256_and_ps(signMask, maskNeg));
466 bwv = _mm256_xor_ps(bwv, _mm256_and_ps(signMask, maskNeg));

471 __m256 lw = _mm256_fmadd_ps(_mm256_sub_ps(bwv, awv), vA, awv);

473 __m256 len2 = _mm256_fmadd_ps(lx, lx, _mm256_fmadd_ps(ly, ly, _mm256_fmadd_ps(lz, lz, _mm256_mul_ps(lw, lw))));

477 lz = _mm256_mul_ps(lz, invL);
