414 Math::Quatf CalcCurrentRotation() const noexcept {
415 std::shared_lock lock(sharedMutex);
416 Math::Quatf outRot;
417
418 // マウス → 角度（1pxあたり何ラジアン回すかを sens* で決める）
419 float yaw = dx * sensX_rad_per_px; // 右に動かすと右旋回にしたい等で符号調整
420 float pitch = dy * sensY_rad_per_px;
421
422 // ピッチ制限（オススメ：累積角で管理）
423 float newPitch = std::clamp(pitchAccum + pitch, Math::Deg2Rad(-89.0f), Math::Deg2Rad(89.0f));
424 pitch = newPitch - pitchAccum;
425
426 // 1) Yaw をワールドUpで適用
427 const Math::Vec3f worldUp{ 0,1,0 };
428 Math::Quatf qYaw = Math::Quatf::FromAxisAngle(worldUp, yaw);
429 outRot = qYaw * rot;
430 outRot.Normalize();
431
432 // 2) Yaw 後の Right を取り直して Pitch
433 Math::Vec3f right = rot.RotateVector(Math::Vec3f{ 1,0,0 });
434 Math::Quatf qPitch = Math::Quatf::FromAxisAngle(right, pitch);
435 outRot = qPitch * rot;
436 outRot.Normalize();
437
438 return outRot;
