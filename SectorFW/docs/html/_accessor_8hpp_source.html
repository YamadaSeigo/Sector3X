145 template<typename PtrType, size_t Index>
146 void GetMemberStartPtrImpl(BufferType* base, size_t capacity, size_t& offset, PtrType& value) noexcept
147 {
148 auto memPtr = std::get<Index>(PtrType::ptr_tuple);
149 static_assert(std::is_member_object_pointer_v<decltype(memPtr)>); // メンバーオブジェクトポインタであることを確認
150 auto& member = value.*memPtr;
151 using MemberRawType = std::remove_reference_t<decltype(member)>;
152 using MemberType = std::remove_pointer_t<MemberRawType>;
153
154 if constexpr (IsToPtr<MemberType>::value) {
155 GetMemberStartPtr<MemberType>(
156 base, capacity, offset, member,
157 std::make_index_sequence<std::tuple_size_v<decltype(MemberType::ptr_tuple)>>{}
158 );
159 }
160 else {
161 offset = AlignTo(offset, alignof(MemberType));
162 if constexpr (std::is_pointer_v<MemberRawType>) {
163 // メンバがポインタ → reinterpret_castで代入可能
164 member = reinterpret_cast<MemberRawType>(base + offset);
165 }
166 else {
167 // メンバが値型（floatなど）→ 代入不能 → コンパイルエラーを防ぐためstatic_assert
168 static_assert(std::is_pointer_v<MemberRawType>, "Member must be a pointer type");
169 }
170 offset += sizeof(MemberType) * capacity;
171 }
