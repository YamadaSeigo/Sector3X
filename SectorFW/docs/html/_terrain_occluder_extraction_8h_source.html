34 inline float Dist2PointAABB(const Math::Vec3f& p, const Math::AABB3f& b)
35 {
36 float dx = 0.f; if (p.x < b.lb.x) dx = b.lb.x - p.x; else if (p.x > b.ub.x) dx = p.x - b.ub.x;
37 float dy = 0.f; if (p.y < b.lb.y) dy = b.lb.y - p.y; else if (p.y > b.ub.y) dy = p.y - b.ub.y;
38 float dz = 0.f; if (p.z < b.lb.z) dz = b.lb.z - p.z; else if (p.z > b.ub.z) dz = p.z - b.ub.z;
39 return dx * dx + dy * dy + dz * dz;
40 }

50 // Project 8 corners of an AABB to NDC and then to screen, return covered pixel area (conservative rectangle)
51 // Returns 0 if completely behind the near plane (w <= 0 for all corners) or if the rectangle is degenerate.
52 inline float AABBScreenAreaPx(const Math::AABB3f& b, const float viewProj[16], uint32_t vpW, uint32_t vpH)
53 {
54 // 8 corners
55 Math::Vec3f c[8] = {
56 {b.lb.x, b.lb.y, b.lb.z}, {b.ub.x, b.lb.y, b.lb.z}, {b.lb.x, b.ub.y, b.lb.z}, {b.ub.x, b.ub.y, b.lb.z},
57 {b.lb.x, b.lb.y, b.ub.z}, {b.ub.x, b.lb.y, b.ub.z}, {b.lb.x, b.ub.y, b.ub.z}, {b.ub.x, b.ub.y, b.ub.z}
58 };
59 float minx = +std::numeric_limits<float>::infinity();
60 float miny = +std::numeric_limits<float>::infinity();
61 float maxx = -std::numeric_limits<float>::infinity();
62 float maxy = -std::numeric_limits<float>::infinity();
63 bool anyInFront = false;
64
65 Math::Vec4f clipPts[8];
66
67 Math::Matrix4x4f viewProjMat;
68 memcpy(viewProjMat.data(), viewProj, sizeof(float) * 16);
69 Math::TransformPoints(viewProjMat, c, clipPts, 8);
70
71 for (int i = 0; i < 8; ++i) {
72 auto v = clipPts[i];
73
74 if (v.w > 0.0f) anyInFront = true; // at least one corner in front of near plane
75 // If h.w <= 0, push it slightly in front (very conservative clamp) to keep bounds usable
76 const float w = (v.w == 0.0f ? 1e-6f : v.w);
77 const float ndcX = v.x / w;
78 const float ndcY = v.y / w;
79 // clamp ndc to [-2,2] to avoid numeric explosion when far behind; still conservative
80 const float nx = std::fmax(-2.f, std::fmin(2.f, ndcX));
81 const float ny = std::fmax(-2.f, std::fmin(2.f, ndcY));
82 const float sx = (nx * 0.5f + 0.5f) * float(vpW);
83 const float sy = (1.0f - (ny * 0.5f + 0.5f)) * float(vpH); // y-down screen
84 minx = std::fmin(minx, sx); maxx = std::fmax(maxx, sx);
85 miny = std::fmin(miny, sy); maxy = std::fmax(maxy, sy);
86 }
87 if (!anyInFront) return 0.0f;
88 const float wpx = std::fmax(0.0f, maxx - minx);
89 const float hpx = std::fmax(0.0f, maxy - miny);
90 const float area = wpx * hpx;
91 return (area > 1.0f ? area : 0.0f);
92 }

168 // Internal: backface check
169 inline bool IsFrontFacing(const Math::Vec3f& a, const Math::Vec3f& b, const Math::Vec3f& c,
170 const Math::Vec3f& cameraPos, float cosThresh)
171 {
172 Math::Vec3f ab{ b.x - a.x, b.y - a.y, b.z - a.z };
173 Math::Vec3f ac{ c.x - a.x, c.y - a.y, c.z - a.z };
174 Math::Vec3f n{ ab.y * ac.z - ab.z * ac.y, ab.z * ac.x - ab.x * ac.z, ab.x * ac.y - ab.y * ac.x };
175 Math::Vec3f v{ cameraPos.x - a.x, cameraPos.y - a.y, cameraPos.z - a.z };
176 const float nl = std::sqrt(n.x * n.x + n.y * n.y + n.z * n.z);
177 const float vl = std::sqrt(v.x * v.x + v.y * v.y + v.z * v.z);
178 if (nl <= 1e-18f || vl <= 1e-18f) return false;
179 const float dot = (n.x * v.x + n.y * v.y + n.z * v.z) / (nl * vl);
180 return (dot >= cosThresh);
181 }

183 // Core API: rank clusters by *screen coverage*, LOD-select, then expand triangles.
184 // outClusterIds: kept clusters in descending priority (coverage)
185 void ExtractOccluderTriangles_ScreenCoverageLOD(
186 const TerrainClustered& t,
187 const OccluderExtractOptions& opt,
188 std::vector<uint32_t>& outClusterIds,
189 std::vector<SoftTriWorld>& outTrisWorld,
190 std::vector<SoftTriClip>* outTrisClip // optional; may be nullptr
191 )
192 {
193 outClusterIds.clear();
194 outTrisWorld.clear();
195 if (outTrisClip) outTrisClip->clear();
196
197 // Preconditions
198 if (!opt.viewProj) return; // need VP for coverage ranking
199
200 // 1) Score clusters by screen-space AABB area
201 struct Scored { uint32_t id; float area; float d2; };
202 std::vector<Scored> sc; sc.reserve(t.clusters.size());
203
204 const float maxD2 = (opt.maxDistance > 0.0f) ? (opt.maxDistance * opt.maxDistance) : std::numeric_limits<float>::infinity();
205
206 for (uint32_t id = 0; id < (uint32_t)t.clusters.size(); ++id) {
207 const auto& cr = t.clusters[id];
208 const float d2 = Dist2PointAABB(opt.cameraPos, cr.bounds);
209 if (d2 > maxD2) continue; // hard-cut by distance
210 const float areaPx = AABBScreenAreaPx(cr.bounds, opt.viewProj, opt.viewportW, opt.viewportH);
211 if (areaPx < opt.minAreaPx) continue; // too tiny
212 sc.push_back({ id, areaPx, d2 });
213 }
214 if (sc.empty()) return; // nothing significant
215
216 // Highest coverage first
217 const uint32_t keep = std::min<uint32_t>((uint32_t)sc.size(), opt.maxClusters);
218 std::nth_element(sc.begin(), sc.begin() + keep - 1, sc.end(), [](const Scored& a, const Scored& b) { return a.area > b.area; });
219 std::sort(sc.begin(), sc.begin() + keep, [](const Scored& a, const Scored& b) { return a.area > b.area; });
220
221 outClusterIds.reserve(keep);
222
223 // 2) LOD selector
224 DefaultLodSelector defaultSel;
225 const ILodSelector& selector = opt.lodSelector ? *opt.lodSelector : (const ILodSelector&)defaultSel;
226
227 // 3) Expand triangles from the chosen LOD
228 auto addTri = [&](const Math::Vec3f& a, const Math::Vec3f& b, const Math::Vec3f& c) {
229 if (opt.backfaceCull && !IsFrontFacing(a, b, c, opt.cameraPos, opt.faceCosThreshold)) return;
230 outTrisWorld.push_back({ a,b,c });
231 if (outTrisClip && opt.makeClipSpace && opt.viewProj) {
232 SoftTriClip tc{};
233 MulRowMajor4x4_Pos(opt.viewProj, a, tc.v0);
234 MulRowMajor4x4_Pos(opt.viewProj, b, tc.v1);
235 MulRowMajor4x4_Pos(opt.viewProj, c, tc.v2);
236 outTrisClip->push_back(tc);
237 }
238 };
239
240 const auto* verts = t.vertices.data();
241 const auto* idx = t.indexPool.data();
242
243 for (uint32_t i = 0; i < keep; ++i) {
244 const uint32_t cid = sc[i].id; outClusterIds.push_back(cid);
245 const auto& cr = t.clusters[cid];
246
247 // distance for LOD
248 const float dist = std::sqrt(sc[i].d2);
249 const uint32_t lod = selector.selectLOD(cid, dist, sc[i].area);
250
251 // Index range for chosen LOD
252 IndexRange range;
253 bool haveRange = false;
254 if (opt.getLodRange) {
255 haveRange = opt.getLodRange(opt.terrainForGetter ? opt.terrainForGetter : (const void*)&t, cid, lod, range);
256 }
257 if (!haveRange) {
258 // fallback to cluster's base range and optionally decimate
259 range.offset = cr.indexOffset;
260 range.count = cr.indexCount;
261
262 const uint32_t stride = (lod < 3 ? std::max<uint32_t>(1, opt.lodDecimate[lod]) : opt.lodDecimate[2]);
263 if (stride > 1) {
264 // Emit every 'stride' triangle
265 for (uint32_t k = 0, tri = 0; k + 2 < range.count; k += 3, ++tri) {
266 if ((tri % stride) != 0) continue;
267 const auto& a = verts[idx[range.offset + k + 0]].pos;
268 const auto& b = verts[idx[range.offset + k + 1]].pos;
269 const auto& c = verts[idx[range.offset + k + 2]].pos;
270 addTri(a, b, c);
271 }
272 continue;
273 }
274 }
275
276 // Emit full triangles from the chosen LOD range
277 for (uint32_t k = 0; k + 2 < range.count; k += 3) {
278 const auto& a = verts[idx[range.offset + k + 0]].pos;
279 const auto& b = verts[idx[range.offset + k + 1]].pos;
280 const auto& c = verts[idx[range.offset + k + 2]].pos;
281 addTri(a, b, c);
282 }
283 }
284 }

459 const auto pushIf = [&](const Math::Vec3f& p, const Math::Vec3f& n) {
460 // +X / -X
461 if (n.x >= co.normalDotThresh) xsPos.push_back(p.x);
462 if (-n.x >= co.normalDotThresh) xsNeg.push_back(p.x);
463 // +Y / -Y
464 if (n.y >= co.normalDotThresh) ysPos.push_back(p.y);
465 if (-n.y >= co.normalDotThresh) ysNeg.push_back(p.y);
466 // +Z / -Z
467 if (n.z >= co.normalDotThresh) zsPos.push_back(p.z);
468 if (-n.z >= co.normalDotThresh) zsNeg.push_back(p.z);

471 for (uint32_t k = 0; k < cr.indexCount; k += 3) {
472 const auto& a = vtx[idx[k + 0]]; const auto& b = vtx[idx[k + 1]]; const auto& c = vtx[idx[k + 2]];
473 // 三角形法線（幾何）
474 Math::Vec3f ab{ b.pos.x - a.pos.x, b.pos.y - a.pos.y, b.pos.z - a.pos.z };
475 Math::Vec3f ac{ c.pos.x - a.pos.x, c.pos.y - a.pos.y, c.pos.z - a.pos.z };
476 Math::Vec3f n{ ab.y * ac.z - ab.z * ac.y, ab.z * ac.x - ab.x * ac.z, ab.x * ac.y - ab.y * ac.x };
477 const float nl = std::sqrt(n.x * n.x + n.y * n.y + n.z * n.z);

479 // 頂点も登録（厳密過ぎないが“外へは出ない”方向の支持点になる）
480 pushIf(a.pos, n); pushIf(b.pos, n); pushIf(c.pos, n);

484 if (arr.empty()) return defv;

486 std::nth_element(arr.begin(), arr.begin() + nth, arr.end());

509 const float maxShrinkY = (orig.ub.y - orig.lb.y) * co.maxShrinkFrac;
510 const float maxShrinkZ = (orig.ub.z - orig.lb.z) * co.maxShrinkFrac;

520 inner.lb.z = (std::min)(inner.lb.z, inner.ub.z - 1e-5f);

534 inline Math::Vec3f FaceNormal(int f) {
535 switch (f) {
536 case 0: return { 0, 0,-1 }; case 1: return { 0, 0, 1 };
537 case 2: return { -1, 0, 0 }; case 3: return { 1, 0, 0 };
538 case 4: return { 0,-1, 0 }; case 5: return { 0, 1, 0 };

542 inline void ProjectFaceQuad(const Math::Vec3f world[4], const float VP[16], uint32_t vpW, uint32_t vpH,
543 FaceQuad& out, bool dilate1px)
544 {
545 out.valid = true;
546 float minx = +1e9, miny = +1e9, maxx = -1e9, maxy = -1e9;
547 float minNdcZ = +1e9;
548 for (int i = 0; i < 4; ++i) {
549 MulRowMajor4x4_Pos(VP, world[i], out.clip[i]);
550 const float w = (out.clip[i][3] == 0.f ? 1e-6f : out.clip[i][3]);
551 const float nx = std::fmax(-2.f, std::fmin(2.f, out.clip[i][0] / w));
552 const float ny = std::fmax(-2.f, std::fmin(2.f, out.clip[i][1] / w));
553 const float nz = out.clip[i][2] / w; // NDC z
554 out.sx[i] = (nx * 0.5f + 0.5f) * vpW;
555 out.sy[i] = (1.f - (ny * 0.5f + 0.5f)) * vpH;
556 minx = (std::min)(minx, out.sx[i]); maxx = (std::max)(maxx, out.sx[i]);
557 miny = (std::min)(miny, out.sy[i]); maxy = (std::max)(maxy, out.sy[i]);
558 minNdcZ = (std::min)(minNdcZ, nz);
559 }
560 if (dilate1px) { minx -= 1; miny -= 1; maxx += 1; maxy += 1; }
561 const float wpx = std::fmax(0.f, maxx - minx);
562 const float hpx = std::fmax(0.f, maxy - miny);
563 out.areaPx = wpx * hpx;
564 out.minDepthNdc = minNdcZ;
565 if (out.areaPx <= 0.f) out.valid = false;
566 }

568 // クラスターのAABBから、可視な面のみを貢献度ベースで厳選し、最大N面だけ採用
569 inline void ReduceAabbFacesForCluster(const Math::AABB3f& bounds,
570 const Math::Vec3f& camPos,
571 const float VP[16], uint32_t vpW, uint32_t vpH,
572 const AabbFacesReduceOptions& ropt,
573 CoverageMask2D& mask,
574 std::vector<FaceQuad>& outQuads)
575 {
576 outQuads.clear();
577 Math::Vec3f c[8]; BuildAabbWorldCorners(bounds, c);
578
579 struct Cand { FaceQuad q; float score = 0.0f; };
580 std::array<Cand, 6> cand; uint32_t candN = 0;
581 const Math::Vec3f center{ (bounds.lb.x + bounds.ub.x) * 0.5f, (bounds.lb.y + bounds.ub.y) * 0.5f, (bounds.lb.z + bounds.ub.z) * 0.5f };
582 Math::Vec3f viewDir{ camPos.x - center.x, camPos.y - center.y, camPos.z - center.z };
583 const float vlen = std::sqrt(viewDir.x * viewDir.x + viewDir.y * viewDir.y + viewDir.z * viewDir.z);
584 if (vlen > 1e-9f) { viewDir.x /= vlen; viewDir.y /= vlen; viewDir.z /= vlen; }
585
586 for (int f = 0; f < 6; ++f) {
587 const Math::Vec3f n = FaceNormal(f);
588 const float cosv = (n.x * viewDir.x + n.y * viewDir.y + n.z * viewDir.z);
589 if (cosv <= ropt.visCos) continue; // 不可視（ほぼ背面）は除外
590
591 FaceQuad fq; Math::Vec3f w[4];
592 for (int i = 0; i < 4; ++i) w[i] = c[kFaceCorner[f][i]];
593 ProjectFaceQuad(w, VP, vpW, vpH, fq, ropt.dilate1px);
594 if (!fq.valid) continue;
595 fq.faceIndex = f;
596 const float score = (std::max)(0.f, cosv) * fq.areaPx;
597 cand[candN++] = { fq, score };
598 }
599 if (!candN) return;
600
601 std::sort(cand.begin(), cand.begin() + candN, [](const Cand& a, const Cand& b) { return a.score > b.score; });
602
603 for (uint32_t i = 0; i < candN && outQuads.size() < ropt.maxQuadsPerCluster; ++i) {
604 float addedRatio = 0.f;
605 const uint32_t added = mask.countUncoveredTilesAndUpdate(cand[i].q, ropt.depthBias, &addedRatio, /*commit=*/false);
606 if (added == 0 || addedRatio < ropt.minAddedTileRatio) {
607 continue;
608 }
609 (void)mask.countUncoveredTilesAndUpdate(cand[i].q, ropt.depthBias, nullptr, /*commit=*/true);
610 outQuads.push_back(cand[i].q);
611 }
612 }

699 inline Math::Vec3f CalcGridNormal(float hl, float hr, float hd, float hu, float dx, float dz)

711 // --- 高さテクスチャの座標系（ワールド→テクスチャUV）を保持する記述子 ---
712 struct HeightTexMapping {
713 // height テクスチャ（row-major: v行×u列）。16bit/8bit等の場合は外で正規化して float 配列に。
714 const float* tex = nullptr;
715 int texW = 0, texH = 0; // テクスチャサイズ
716
717 // ワールド→UV 変換: u = (x - originX) * worldToTexU + uOffset
718 // v = (z - originZ) * worldToTexV + vOffset
719 float originX = 0.f, originZ = 0.f; // ワールド原点（地形タイルの左下など）
720 float worldToTexU = 1.f, worldToTexV = 1.f; // 1/worldUnitsPerTexel
721 float uOffset = 0.f, vOffset = 0.f; // タイルオフセット（タイルインデックス由来）
722
723 // 高さスケール・オフセット（テクスチャ→ワールドY）
724 float heightScale = 1.f; // 例: テクスチャが[0,1]ならワールドメートルへのスケール
725 float heightOffset = 0.f;
726
727 // サンプリングモード
728 bool clampUV = true; // true: クランプ, false: リピート
729 };

731 // 生成済み H01（0..1 高さ配列）を使って HeightTexMapping を組み立てるユーティリティ
732 inline HeightTexMapping MakeHeightTexMappingFromTerrainParams(
733 const TerrainBuildParams& p,
734 const std::vector<float>& H01 // TerrainClustered::Build(..., &H01) で受け取った配列
735 ) {
736 using namespace SFW::Graphics;
737 HeightTexMapping m{};
738
739 // 高さ“テクスチャ”の実体は H01（row-major: y*W + x）
740 m.tex = H01.data();
741 m.texW = static_cast<int>(p.cellsX + 1); // 頂点数 = セル数+1
742 m.texH = static_cast<int>(p.cellsZ + 1);
743
744 // ワールド→テクセル変換： 1texel = cellSize [m]
745 m.originX = 0.0f; // TerrainClustered は (x*cellSize, z*cellSize) で配置
746 m.originZ = 0.0f; // → 原点は (0,0) でよい
747 m.worldToTexU = 1.0f / p.cellSize; // u = (x - originX) * (1/cellSize)
748 m.worldToTexV = 1.0f / p.cellSize; // v = (z - originZ) * (1/cellSize)
749
750 // 単一大域テクスチャなのでタイルオフセットは 0（タイル運用ならここに加算）
751 m.uOffset = 0.0f;
752 m.vOffset = 0.0f;
753
754 // 高さ 0..1 → ワールドY： y = h * heightScale + heightOffset
755 m.heightScale = p.heightScale;
756 m.heightOffset = p.offset.y;
757
758 // タイル境界での外挿は避けたいので基本はクランプ
759 m.clampUV = true;
760
761 return m;
762 }

764 inline int clampi(int x, int lo, int hi) { return (x < lo ? lo : (x > hi ? hi : x)); }
765 inline float lerp(float a, float b, float t) { return a + (b - a) * t; }

767 // バイリニアサンプル（HeightTexMapping→float 高さ）
768 inline float SampleHeight_Bilinear(const HeightTexMapping& m, float x, float z)
769 {
770 float u = (x - m.originX) * m.worldToTexU + m.uOffset;
771 float v = (z - m.originZ) * m.worldToTexV + m.vOffset;
772 if (m.clampUV) {
773 u = std::fmax(0.f, std::fmin(u, float(m.texW - 1)));
774 v = std::fmax(0.f, std::fmin(v, float(m.texH - 1)));
775 }
776 else {
777 // リピート
778 auto wrap = [](float t, float n) { t = std::fmod(t, n); if (t < 0) t += n; return t; };
779 u = wrap(u, float(m.texW - 1));
780 v = wrap(v, float(m.texH - 1));
781 }
782 const int x0 = (int)std::floor(u); const int x1 = clampi(x0 + 1, 0, m.texW - 1);
783 const int y0 = (int)std::floor(v); const int y1 = clampi(y0 + 1, 0, m.texH - 1);
784 const float tu = u - float(x0);
785 const float tv = v - float(y0);
786 const float h00 = m.tex[y0 * m.texW + x0];
787 const float h10 = m.tex[y0 * m.texW + x1];
788 const float h01 = m.tex[y1 * m.texW + x0];
789 const float h11 = m.tex[y1 * m.texW + x1];
790 const float h0 = lerp(h00, h10, tu);
791 const float h1 = lerp(h01, h11, tu);
792 const float h = lerp(h0, h1, tv);
793 return h * m.heightScale + m.heightOffset;
794 }

798 {
799 const HeightTexMapping& map = *reinterpret_cast<const HeightTexMapping*>(user);

857 // グリッド頂点の高さをサンプル
858 std::vector<float> H((gx + 1) * (gz + 1));
859 for (uint32_t iz = 0; iz <= gz; ++iz) {
860 const float z = z0 + dz * float(iz);
861 for (uint32_t ix = 0; ix <= gx; ++ix) {
862 const float x = x0 + dx * float(ix);
863 float h = SampleHeight_Bilinear(map, x, z);
864 h = std::fmax(hopt.heightClampMin, std::fmin(h, hopt.heightClampMax));
865 H[iz * (gx + 1) + ix] = h;
866 }
867 }
868
869 auto P = [&](uint32_t ix, uint32_t iz) {
870 const float x = x0 + dx * float(ix);
871 const float z = z0 + dz * float(iz);
872 const float y = H[iz * (gx + 1) + ix];
873 return Math::Vec3f{ x,y,z };
874 };
875
876 const float cellDiag = std::sqrt(dx * dx + dz * dz);
877
878 auto emitTri = [&](const Math::Vec3f& a, const Math::Vec3f& c, const Math::Vec3f& b) {
879 if (opt.backfaceCull && !IsFrontFacing(a, b, c, opt.cameraPos, opt.faceCosThreshold)) return;
880 outTrisWorld.push_back({ a,b,c });
881 if (outTrisClip && opt.makeClipSpace && opt.viewProj) {
882 SoftTriClip tc{};
883 MulRowMajor4x4_Pos(opt.viewProj, a, tc.v0);
884 MulRowMajor4x4_Pos(opt.viewProj, b, tc.v1);
885 MulRowMajor4x4_Pos(opt.viewProj, c, tc.v2);
886 outTrisClip->push_back(tc);
887 }
888 };
889
890 for (uint32_t iz = 0; iz < gz; ++iz) {
891 for (uint32_t ix = 0; ix < gx; ++ix) {
892 // 4隅の高さ
893 const float h00 = H[(iz + 0) * (gx + 1) + (ix + 0)];
894 const float h10 = H[(iz + 0) * (gx + 1) + (ix + 1)];
895 const float h01 = H[(iz + 1) * (gx + 1) + (ix + 0)];
896 const float h11 = H[(iz + 1) * (gx + 1) + (ix + 1)];
897
898 // 中心差分で法線
899 const float hl = h00, hr = h10, hd = h00, hu = h01; // 近傍代理（厳密でなくてOK）
900 Math::Vec3f n = CalcGridNormalFromSamples(hl, hr, hd, hu, dx, dz);
901 if (n.y < hopt.upDotMin) continue; // 上向きでなければスキップ
902
903 // スロープタンジェント（近似）
904 const float tanSlope = std::sqrt((std::max)(0.f, (1.f - n.y * n.y))) / (std::max)(1e-6f, n.y);
905 if (hopt.maxSlopeTan > 0.f && tanSlope > hopt.maxSlopeTan) continue;
906
907 // 高さバイアス（下方向へ）
908 auto bias = [&](float y) { return y - (hopt.bias.baseDown + hopt.bias.slopeK * tanSlope * cellDiag); };
909
910 Math::Vec3f p00 = P(ix + 0, iz + 0); p00.y = bias(p00.y);
911 Math::Vec3f p10 = P(ix + 1, iz + 0); p10.y = bias(p10.y);
912 Math::Vec3f p01 = P(ix + 0, iz + 1); p01.y = bias(p01.y);
913 Math::Vec3f p11 = P(ix + 1, iz + 1); p11.y = bias(p11.y);
914
915 // 2三角（対角は適宜選択）。ここでは (00,10,01) と (01,10,11)
916 emitTri(p00, p10, p01);
917 emitTri(p01, p10, p11);
918 }
919 }

923 inline void ExtractOccluderTriangles_HeightmapCoarse_Hybrid(
924 const TerrainClustered& t,
925 const HeightTexMapping& map,
926 HeightCoarseOptions2 hopt,
927 const OccluderExtractOptions& opt,
928 std::vector<uint32_t>& outClusterIds,
929 std::vector<SoftTriWorld>& outTrisWorld,
930 std::vector<SoftTriClip>* outTrisClip)
931 {
932 outClusterIds.clear(); outTrisWorld.clear(); if (outTrisClip) outTrisClip->clear();
933 if (!opt.viewProj) return;
934
935 // 画面占有率でクラスタ選別
936 struct Scored { uint32_t id; float area; float d2; };
937 std::vector<Scored> sc; sc.reserve(t.clusters.size());
938 const float maxD2 = (opt.maxDistance > 0.0f) ? (opt.maxDistance * opt.maxDistance) : std::numeric_limits<float>::infinity();
939 for (uint32_t id = 0; id < (uint32_t)t.clusters.size(); ++id) {
940 const auto& cr = t.clusters[id];
941 const float d2 = Dist2PointAABB(opt.cameraPos, cr.bounds);
942 if (d2 > maxD2) continue;
943 const float areaPx = AABBScreenAreaPx(cr.bounds, opt.viewProj, opt.viewportW, opt.viewportH);
944 if (areaPx < opt.minAreaPx) continue;
945 sc.push_back({ id, areaPx, d2 });
946 }
947 if (sc.empty()) return;
948
949 // 前提: sc[] は {id, area, d2} が詰まっている
950 constexpr struct CenterBiasOptions {
951 bool enable = true;
952 float sigmaPx = 320.0f; // 中心からの距離に対するガウス幅（px）
953 float gain = 1.0f; // バイアス強度（0で無効）
954 uint32_t reserveCenterN = 8; // 中心窓から必ず確保する数（0で無効）
955 float centerWindowFrac = 2.0f; // 画面中央の長方形（幅/高さの比）
956 } copt; // 必要に応じて外から渡してもOK
957
958 auto ProjectPointToScreen = [&](const Math::Vec3f& p, float& sx, float& sy)->bool {
959 float h[4];
960 MulRowMajor4x4_Pos(opt.viewProj, p, h);
961 if (h[3] <= 0.0f) return false;
962 const float nx = std::fmax(-2.f, std::fmin(2.f, h[0] / h[3]));
963 const float ny = std::fmax(-2.f, std::fmin(2.f, h[1] / h[3]));
964 sx = (nx * 0.5f + 0.5f) * float(opt.viewportW);
965 sy = (1.f - (ny * 0.5f + 0.5f)) * float(opt.viewportH);
966 return true;
967 };
968
969 struct Scored2 {
970 uint32_t id; float area; float score; bool inCenter;
971 };
972 std::vector<Scored2> ranked; ranked.reserve(sc.size());
973
974 const float cx = 0.5f * float(opt.viewportW);
975 const float cy = 0.5f * float(opt.viewportH);
976 const float winW = float(opt.viewportW) * copt.centerWindowFrac;
977 const float winH = float(opt.viewportH) * copt.centerWindowFrac;
978 const float left = cx - 0.5f * winW, right = cx + 0.5f * winW;
979 const float top = cy - 0.5f * winH, bottom = cy + 0.5f * winH;
980
981 for (const auto& s : sc) {
982 const auto& cr = t.clusters[s.id];
983 // AABB中心を画面座標へ（近似で十分）
984 const Math::Vec3f ctr{ (cr.bounds.lb.x + cr.bounds.ub.x) * 0.5f,
985 (cr.bounds.lb.y + cr.bounds.ub.y) * 0.5f,
986 (cr.bounds.lb.z + cr.bounds.ub.z) * 0.5f };
987 float sx = 0, sy = 0; bool vis = ProjectPointToScreen(ctr, sx, sy);
988
989 float score = s.area;
990 bool inside = false;
991 if (vis && copt.enable && copt.gain > 0.f && copt.sigmaPx > 0.f) {
992 const float dx = sx - cx, dy = sy - cy;
993 const float r2 = dx * dx + dy * dy;
994 const float sig2 = copt.sigmaPx * copt.sigmaPx;
995 const float boost = copt.gain * std::exp(-r2 / (2.0f * sig2));
996 score = s.area * (1.0f + boost); // 中心に近いほどスコア加点
997 inside = (sx >= left && sx <= right && sy >= top && sy <= bottom);
998 }
999 ranked.push_back({ s.id, s.area, score, inside });
1000 }
1001
1002 // スコア降順で並べる
1003 std::sort(ranked.begin(), ranked.end(),
1004 [](const Scored2& a, const Scored2& b) { return a.score > b.score; });
1005
1006 // 収容上限
1007 const uint32_t cap = (opt.maxClusters > 0)
1008 ? std::min<uint32_t>((uint32_t)ranked.size(), opt.maxClusters)
1009 : (uint32_t)ranked.size();
1010
1011 outClusterIds.clear();
1012 outClusterIds.reserve(cap);
1013
1014 std::vector<float> areas;
1015 areas.reserve(cap);
1016
1017 // 中心窓から N 個“先取り”
1018 if (copt.enable && copt.reserveCenterN > 0) {
1019 uint32_t taken = 0;
1020 for (const auto& r : ranked) {
1021 if (!r.inCenter) continue;
1022 outClusterIds.push_back(r.id);
1023 areas.push_back(r.area);
1024 if (++taken >= copt.reserveCenterN || outClusterIds.size() >= cap) break;
1025 }
1026 }
1027
1028 // 残りはスコア順で埋める（重複回避）
1029 auto already = [&](uint32_t id) {
1030 return std::find(outClusterIds.begin(), outClusterIds.end(), id) != outClusterIds.end();
1031 };
1032 for (const auto& r : ranked) {
1033 if (outClusterIds.size() >= cap) break;
1034 if (already(r.id)) continue;
1035 outClusterIds.push_back(r.id);
1036 areas.push_back(r.area);
1037 }
1038
1039 // “keep で ranked を再プッシュ”は不要！
1040 // ここからは outClusterIds を使って三角形を生成するだけ。
1041 for (uint32_t i = 0; i < outClusterIds.size(); ++i) {
1042 const uint32_t cid = outClusterIds[i];
1043 const auto& cr = t.clusters[cid];
1044
1045 // 高さの粗サーフェスのみを出力（必要ならここに側面1枚の処理を足す）
1046 BuildHeightCoarseSurfaceForCluster_Mapped(cr.bounds, areas[i], map, hopt, opt, outTrisWorld, outTrisClip);
1047 }
1048 }

1102 };
