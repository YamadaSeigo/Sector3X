65def generate_enum():
66 lines = []

74def generate_sdl_map():
75 lines = []
76 lines.append("constexpr std::array<Key, 512> SDLKeyToCommonKey = [] {")
77 lines.append(" std::array<Key, 512> map{};")
78 for i, (name, sdl_vk) in enumerate(key_definitions.items()):
79 if sdl_vk is not None:
80 sdl_key, _ = sdl_vk
81 lines.append(f" map[{sdl_key}] = Key::{name};")
82 lines.append(" return map;")
83 lines.append("}();")
84 return "\n".join(lines)
