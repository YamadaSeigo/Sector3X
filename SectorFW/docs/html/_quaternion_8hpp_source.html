32 void Normalize() noexcept {
33 const T len2 = LengthSq();
34 const T eps = std::numeric_limits<T>::epsilon();
35 if (len2 <= eps) { x = 0; y = 0; z = 0; w = 1; return; }
36 const T inv = T(1) / std::sqrt(len2);

43 // オイラー（ピッチ= X, ヨー= Y, ロール= Z : 適用順は Yaw→Pitch→Roll を想定）
44 static Quat FromEuler(T pitch, T yaw, T roll) noexcept {
45 const T h = T(0.5);
46 const T cy = std::cos(yaw * h), sy = std::sin(yaw * h);
47 const T cp = std::cos(pitch * h), sp = std::sin(pitch * h);
48 const T cr = std::cos(roll * h), sr = std::sin(roll * h);
49
50 // （注）この式は Y->X->Z の順（右掛け適用）を想定
51 return Quat(
52 sr * cp * cy - cr * sp * sy,
53 cr * sp * cy + sr * cp * sy,
54 cr * cp * sy - sr * sp * cy,
55 cr * cp * cy + sr * sp * sy
56 );
57 }

59 static Quat FromAxisAngle(const Vec3<T>& axis, T angleRad) noexcept {
60 const T len = std::sqrt(axis.x * axis.x + axis.y * axis.y + axis.z * axis.z);
61 if (len <= std::numeric_limits<T>::epsilon()) return Quat(); // identity
62 const T inv = T(1) / len;
63 const T half = angleRad * T(0.5);
64 const T s = std::sin(half), c = std::cos(half);
65 return Quat(axis.x * inv * s, axis.y * inv * s, axis.z * inv * s, c);
66 }

77 static Quat Slerp(const Quat& a, const Quat& b, T t) noexcept {
78 T dot = a.x * b.x + a.y * b.y + a.z * b.z + a.w * b.w;
79 Quat b2 = b;
80 if (dot < T(0)) { dot = -dot; b2 = Quat(-b.x, -b.y, -b.z, -b.w); }
81
82 // 丸め誤差対策
83 dot = std::clamp(dot, T(-1), T(1));
84
85 if (dot > T(0.9995)) {
86 Quat r{ a.x + t * (b2.x - a.x),
87 a.y + t * (b2.y - a.y),
88 a.z + t * (b2.z - a.z),
89 a.w + t * (b2.w - a.w) };
90 r.Normalize();
91 return r;
92 }
93
94 const T theta0 = std::acos(dot);
95 const T theta = theta0 * t;
96 const T s0 = std::cos(theta) - dot * std::sin(theta) / std::sin(theta0);
97 const T s1 = std::sin(theta) / std::sin(theta0);
98 return Quat(s0 * a.x + s1 * b2.x,
99 s0 * a.y + s1 * b2.y,
100 s0 * a.z + s1 * b2.z,
101 s0 * a.w + s1 * b2.w);
102 }

104 // （右掛け合成）「this の後に q を適用」
105 Quat operator*(const Quat& q) const noexcept {
106 return Quat(
107 w * q.x + x * q.w + y * q.z - z * q.y,
108 w * q.y - x * q.z + y * q.w + z * q.x,
109 w * q.z + x * q.y - y * q.x + z * q.w,
110 w * q.w - x * q.x - y * q.y - z * q.z
111 );
112 }

114 // q = swing * twist となるように、
115 // 指定軸 axis 周りの回転(twist)と、それ以外の回転(swing)に分解する。
116 // axis はワールド/ローカルどちらの軸かは呼び出し側の解釈次第。
117 // 単位クォータニオン前提（不安なら Normalize() してから呼ぶ）
118 void DecomposeSwingTwist(const Vec3<T>& axis,
119 Quat& outSwing,
120 Quat& outTwist) const noexcept
121 {
122 // axis を正規化
123 Vec3<T> n = axis;
124 {
125 const T len = std::sqrt(n.x * n.x + n.y * n.y + n.z * n.z);
126 const T eps = std::numeric_limits<T>::epsilon();
127 if (len <= eps) {
128 // 軸がゼロベクトルなら分解不能なので、
129 // twist = identity, swing = this として返す
130 outTwist = Quat(T(0), T(0), T(0), T(1));
131 outSwing = *this;
132 return;
133 }
134 const T invLen = T(1) / len;
135 n.x *= invLen; n.y *= invLen; n.z *= invLen;
136 }
137
138 // q のベクトル部
139 const Vec3<T> v{ x, y, z };
140
141 // v を軸 n に射影
142 const T projLen = n.x * v.x + n.y * v.y + n.z * v.z;
143 const Vec3<T> proj{ n.x * projLen, n.y * projLen, n.z * projLen };
144
145 // twist: スカラー部はそのまま、ベクトル部を axis 方向だけにしたクォータニオン
146 Quat twist{ proj.x, proj.y, proj.z, w };
147
148 // proj がほぼゼロなら「軸周りの回転はほぼ無い」とみなして identity
149 const T projLenSq = proj.x * proj.x + proj.y * proj.y + proj.z * proj.z;
150 if (projLenSq <= std::numeric_limits<T>::epsilon()) {
151 twist = Quat(T(0), T(0), T(0), T(1));
152 }
153 else {
154 twist.Normalize();
155 }
156
157 // swing = q * conj(twist)
158 const Quat conjTwist(-twist.x, -twist.y, -twist.z, twist.w);
159 Quat swing = (*this) * conjTwist;
160 swing.Normalize();
161
162 outSwing = swing;
163 outTwist = twist;
164 }
165
166 // 指定軸まわりの回転成分（twist）だけを残す
167 Quat KeepTwist(const Vec3<T>& axis) const noexcept
168 {
169 Quat swing, twist;
170 DecomposeSwingTwist(axis, swing, twist);
171 return twist;
172 }

174 // 指定軸 axis 周りの回転成分(twist)だけを除去したクォータニオンを返す。
175 // -> swing だけを返す
176 Quat RemoveTwist(const Vec3<T>& axis) const noexcept
177 {
178 Quat swing, twist;
179 DecomposeSwingTwist(axis, swing, twist);
180 return swing;
181 }

187 struct Basis {
188 Vec3<T> right;

193 // q は単位クォータニオン前提（不安なら Normalize() してから呼ぶ）
194 template<typename T>
195 inline Basis<T> FastBasisFromQuat(const Quat<T>& q) noexcept {
196 const T x = q.x, y = q.y, z = q.z, w = q.w;
197
198 const T xx = x + x, yy = y + y, zz = z + z;
199 const T wx = w * xx, wy = w * yy, wz = w * zz;
200 const T xx2 = x * xx, xy2 = x * yy, xz2 = x * zz;
201 const T yy2 = y * yy, yz2 = y * zz, zz2 = z * zz;
202
203 Basis<T> b;
204 // 列ベクトル（Right, Up, Forward(+Z)）
205 b.right = { T(1) - (yy2 + zz2), xy2 + wz, xz2 - wy };
206 b.up = { xy2 - wz, T(1) - (xx2 + zz2), yz2 + wx };
207 b.forward = { xz2 + wy, yz2 - wx, T(1) - (xx2 + yy2) };
208 return b;
209 }

239 template<typename T>
240 inline Vec3<T> NormalizeSafe(const Vec3<T>& v) noexcept {
241 const T l2 = v.x * v.x + v.y * v.y + v.z * v.z;
242 if (l2 <= std::numeric_limits<T>::epsilon()) return { T(0),T(0),T(0) };
243 const T inv = T(1) / std::sqrt(l2);
244 return { v.x * inv, v.y * inv, v.z * inv };
245 }

247 template<typename T>
248 inline void OrthonormalizeBasis(Basis<T>& b) noexcept {
249 // r を正規化
250 Vec3<T> r = NormalizeSafe(b.right);
251
252 // u から r 成分を引いて正規化（Gram–Schmidt）
253 Vec3<T> u = {
254 b.up.x - (r.x * (b.up.x * r.x + b.up.y * r.y + b.up.z * r.z)),
255 b.up.y - (r.y * (b.up.x * r.x + b.up.y * r.y + b.up.z * r.z)),
256 b.up.z - (r.z * (b.up.x * r.x + b.up.y * r.y + b.up.z * r.z)),
257 };
258 u = NormalizeSafe(u);
259
260 // f は r×u から作る（右手系）。入力 f と反転していたら符号合わせ
261 Vec3<T> f = {
262 r.y * u.z - r.z * u.y,
263 r.z * u.x - r.x * u.z,
264 r.x * u.y - r.y * u.x
265 };
266 // 入力 forward と向きが逆なら反転
267 T s = (f.x * b.forward.x + f.y * b.forward.y + f.z * b.forward.z) < T(0) ? T(-1) : T(1);
268 f = { f.x * s, f.y * s, f.z * s };
269
270 b.right = r; b.up = u; b.forward = f;
271 }

273 // --- Basis(+Z列) → Quaternion（数値安定版） ----------------------
274 // ※ Basis.forward は「+Z列」（FastBasisFromQuat と同じ定義）を期待
275 template<typename T>
276 inline Quat<T> QuatFromBasisPlusZ(Basis<T> b) noexcept {
277 // 入力の微妙な非直交・非単位を補正
278 OrthonormalizeBasis(b);
279
280 // 列ベクトルを 3x3 行列に配置（FastBasisFromQuat と往復一致させる）
281 // R = [ r u f ]（列）
282 const T m00 = b.right.x, m01 = b.up.x, m02 = b.forward.x;
283 const T m10 = b.right.y, m11 = b.up.y, m12 = b.forward.y;
284 const T m20 = b.right.z, m21 = b.up.z, m22 = b.forward.z;
285
286 const T trace = m00 + m11 + m22;
287 Quat<T> q;
288
289 if (trace > T(0)) {
290 T s = std::sqrt(trace + T(1)) * T(2);
291 q.w = T(0.25) * s;
292 q.x = (m21 - m12) / s;
293 q.y = (m02 - m20) / s;
294 q.z = (m10 - m01) / s;
295 }
296 else if (m00 > m11 && m00 > m22) {
297 T s = std::sqrt(T(1) + m00 - m11 - m22) * T(2);
298 q.w = (m21 - m12) / s;
299 q.x = T(0.25) * s;
300 q.y = (m01 + m10) / s;
301 q.z = (m02 + m20) / s;
302 }
303 else if (m11 > m22) {
304 T s = std::sqrt(T(1) + m11 - m00 - m22) * T(2);
305 q.w = (m02 - m20) / s;
306 q.x = (m01 + m10) / s;
307 q.y = T(0.25) * s;
308 q.z = (m12 + m21) / s;
309 }
310 else {
311 T s = std::sqrt(T(1) + m22 - m00 - m11) * T(2);
312 q.w = (m10 - m01) / s;
313 q.x = (m02 + m20) / s;
314 q.y = (m12 + m21) / s;
315 q.z = T(0.25) * s;
316 }
317
318 // 丸め誤差対策
319 q.Normalize();
320 return q;
321 }

324 inline Quat<T> QuatFromBasis(const Vec3<T>& right,
325 const Vec3<T>& up,
