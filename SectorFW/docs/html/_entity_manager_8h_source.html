249 template<typename T>
250 void AddComponent(EntityID id, const T& value) {
251 if constexpr (ComponentTypeRegistry::IsSparse<T>()) {
252 GetSparseStore<T>().Add(id, value);
253 return;
254 }
255
256 ComponentTypeID typeID = ComponentTypeRegistry::GetID<T>();
257 ComponentMask currentMask = GetMask(id);
258 currentMask.set(typeID);
259
260 Archetype* newArch = archetypeManager.GetOrCreate(currentMask);
261 ArchetypeChunk* newChunk = newArch->GetOrCreateChunk();
262
263 // 旧ロケーションは共有ロックで取得（重い移動処理はロック外で実行）
264 EntityLocation oldLoc;
265 {
266 std::shared_lock<std::shared_mutex> rlock(locationsMutex);
267 oldLoc = locations.at(id);
268 }
269 ArchetypeChunk* oldChunk = oldLoc.chunk;
270 size_t oldIndex = oldLoc.index;
271
272 size_t newIndex = newChunk->AddEntity(id);
273
274 // 必要なデータのみ移動
275 auto& oldChunkLayout = ArchetypeChunk::Accessor::GetLayoutInfo(oldChunk);
276 auto& oldChunkLayoutIdx = ArchetypeChunk::Accessor::GetLayoutInfoIdx(oldChunk);
277 auto& newChunkLayout = ArchetypeChunk::Accessor::GetLayoutInfo(newChunk);
278 auto& newChunkLayoutIdx = ArchetypeChunk::Accessor::GetLayoutInfoIdx(newChunk);
279 for (const auto& [comp, idx] : oldChunkLayoutIdx) {
280 auto& info = oldChunkLayout[idx];
281 auto& newInfo = newChunkLayout[newChunkLayoutIdx.at(comp)];
282 size_t index = 0;
283 for (const auto& i : info)
284 {
285 auto src = ArchetypeChunk::Accessor::GetBuffer(oldChunk) + i.offset;
286 auto dst = ArchetypeChunk::Accessor::GetBuffer(newChunk) + newInfo.get(index).value().get().offset;
287 auto* dstElem = static_cast<uint8_t*>(dst) + newIndex * i.stride;
288 auto* srcElem = static_cast<const uint8_t*>(src) + oldIndex * i.stride;
289 // i.stride 単位=1要素のサイズという前提
290 // メタに「列の element_size / trivial フラグ」が持てるならそれを使う。
291 if (/* element is trivially copyable */ true) {
292 std::memcpy(dstElem, srcElem, i.stride);
293 }
294 else {
295 // 要素型が分からない場合は、列側に「ムーブ/コピー関数ポインタ」を持たせる設計に寄せると安全
296 // new (dstElem) ElemType(*reinterpret_cast<const ElemType*>(srcElem));
297 // reinterpret_cast<const ElemType*>(srcElem)->~ElemType(); // swap-pop側が破棄するなら不要
298 }
299
300 index++;
301 }
302 }
303
304 newChunk->GetColumn<T>()[newIndex] = value;
305
306 // スワップ相手のロケーション更新と自分の更新は「排他ロック」下で一気に
307 {
308 const size_t lastIndexBefore = oldChunk->GetEntityCount() - 1;
309 std::unique_lock<std::shared_mutex> wlock(locationsMutex);
310 if (oldIndex < lastIndexBefore) {
311 EntityID swappedId = ArchetypeChunk::Accessor::GetEntityIDs(oldChunk)[lastIndexBefore];
312 auto it = locations.find(swappedId);
313 if (it != locations.end()) [[unlikely]] it->second = { oldChunk, oldIndex };
314 }
315 oldChunk->RemoveEntitySwapPop(oldIndex);
316 locations[id] = { newChunk, newIndex };
317 }
318 }

325 template<typename T>
326 void RemoveComponent(EntityID id) {
327 ComponentTypeID typeID = ComponentTypeRegistry::GetID<T>();
328 ComponentMask oldMask = GetMask(id);
329 if (!oldMask.test(typeID)) return; // 持ってないなら何もしない
330
331 ComponentMask newMask = oldMask;
332 newMask.reset(typeID);
333
334 if (ComponentTypeRegistry::IsSparseMask(ComponentMask().set(typeID))) {
335 GetSparseStore<T>().Remove(id);
336 return;
337 }
338
339 // 旧ロケーションは共有ロックで取得
340 EntityLocation oldLoc;
341 {
342 std::shared_lock<std::shared_mutex> rlock(locationsMutex);
343 oldLoc = locations.at(id);
344 }
345 ArchetypeChunk* oldChunk = oldLoc.chunk;
346 size_t oldIndex = oldLoc.index;
347
348 Archetype* newArch = archetypeManager.GetOrCreate(newMask);
349 ArchetypeChunk* newChunk = newArch->GetOrCreateChunk();
350 size_t newIndex = newChunk->AddEntity(id);
351
352 // 必要なデータのみ移動
353 auto& oldChunkLayout = ArchetypeChunk::Accessor::GetLayoutInfo(oldChunk);
354 auto& oldChunkLayoutIdx = ArchetypeChunk::Accessor::GetLayoutInfoIdx(oldChunk);
355 auto& newChunkLayout = ArchetypeChunk::Accessor::GetLayoutInfo(newChunk);
356 auto& newChunkLayoutIdx = ArchetypeChunk::Accessor::GetLayoutInfoIdx(newChunk);
357 for (const auto& [comp, idx] : oldChunkLayout) {
358 if (comp == typeID) continue;
359
360 auto& info = oldChunkLayout[idx];
361 auto& newInfo = newChunkLayout[newChunkLayoutIdx.at(comp)];
362 size_t index = 0;
363 for (const auto& i : info)
364 {
365 auto src = ArchetypeChunk::Accessor::GetBuffer(oldChunk) + i.offset;
366 auto dst = ArchetypeChunk::Accessor::GetBuffer(newChunk) + newInfo.get(index).value().get().offset;
367 std::memcpy(static_cast<uint8_t*>(dst) + newIndex * i.stride,
368 static_cast<const uint8_t*>(src) + oldIndex * i.stride,
369 i.stride);
370
371 index++;
372 }
373 }
374
375 {
376 const size_t lastIndexBefore = oldChunk->GetEntityCount() - 1;
377 std::unique_lock<std::shared_mutex> wlock(locationsMutex);
378 if (oldIndex < lastIndexBefore) {
379 EntityID swappedId = ArchetypeChunk::Accessor::GetEntityIDs(oldChunk)[lastIndexBefore];
380 auto it = locations.find(swappedId);
381 if (it != locations.end()) it->second = { oldChunk, oldIndex };
382 }
383 oldChunk->RemoveEntitySwapPop(oldIndex);
384 locations[id] = { newChunk, newIndex };
385 }

502 "Non-trivial T must be move/copy constructible for in-place placement.");
503 }

530 template<typename Variant, typename U, size_t Index>
531 void ExpansionMemberImpl(std::vector<Variant>& vec, U& value) {
532 auto ptr = std::get<Index>(U::member_ptr_tuple);
533 static_assert(std::is_member_object_pointer_v<decltype(ptr)>); // メンバーオブジェクトポインタであることを確認
534 auto& member = value.*ptr;
535
536 using MemberT = std::remove_cvref_t<decltype(member)>; //constと参照を除去した型
537
538 if constexpr (IsSoAComponent<MemberT>) {
539 ExpansionMember<Variant>(vec, member, std::make_index_sequence<std::tuple_size_v<typename MemberT::tuple_type>>{});
540 }
541 else if constexpr (IsPrimitive<MemberT>) {
542 vec.push_back(member);
543 }
544 else {
545 static_assert(std::is_same_v<MemberT, void>, "Unsupported member type: must be primitive or reflectable (HasMemberPtr)");
546 }
547 }

578 void MemorySetChunkSoA(T* base, size_t index, const T& value, size_t capacity, std::index_sequence<Is...>) {
579 auto charBase = reinterpret_cast<BufferType*>(base);
580 size_t offset = 0;
581 auto members = GetMember(value);
582 for (const auto& v : members) {
583 std::visit([&](auto&& val) {
584 using ValType = std::remove_cvref_t<decltype(val)>;
585 offset = AlignTo(offset, alignof(ValType));
586 *(reinterpret_cast<ValType*>(charBase + offset) + index) = val;
587 offset += sizeof(ValType) * capacity;
588 }, v);
589 }

647 template<typename T>
648 struct SparseWrapper : ISparseWrapper {
649 SparseComponentStore<T> store;
654 void Remove(EntityID id) override { store.Remove(id); }
659 void MoveAllTo(EntityManager& dst) override {
660 auto& srcMap = store.GetComponents();
661 auto& dstMap = dst.GetSparseStore<T>().GetComponents();
662 dstMap.reserve(dstMap.size() + srcMap.size());
663 for (auto it = srcMap.begin(); it != srcMap.end(); ++it) {
664 dstMap.insert_or_assign(it->first, std::move(it->second));
665 }
666 srcMap.clear();
667 }
674 void MoveManyTo(EntityManager& dst, const EntityID* ids, size_t n) override {
675 auto& srcMap = store.GetComponents();
676 auto& dstMap = dst.GetSparseStore<T>().GetComponents();
677 dstMap.reserve(dstMap.size() + n);
678 for (size_t i = 0; i < n; ++i) {
679 auto it = srcMap.find(ids[i]);
680 if (it != srcMap.end()) {
681 dstMap.insert_or_assign(it->first, std::move(it->second));
682 srcMap.erase(it);
683 }
684 }
685 }
