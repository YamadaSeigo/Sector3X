157 std::vector<SpatialChunk*> CullChunks(const Math::Vec3f& center, float radius) const noexcept
158 {
159 std::vector<SpatialChunk*> out;
160 if (!m_root) return out;
161
162 if (!std::isfinite(radius) || radius <= 0.0f) radius = 0.0f;
163 const float r = (radius < 0.0f) ? 0.0f : radius;
164 const float r2 = r * r;
165
166 const auto localCenter = center - levelOriginWS;
167
168 auto dist2_point_aabb_xz = [&](const Node& n) -> float {
169 auto clamp = [](float v, float lo, float hi) { return v < lo ? lo : (v > hi ? hi : v); };
170 const float qx = clamp(localCenter.x, n.bounds.lb.x, n.bounds.ub.x);
171 const float qz = clamp(localCenter.z, n.bounds.lb.y, n.bounds.ub.y);
172 const float dx = localCenter.x - qx, dz = localCenter.z - qz;
173 return dx * dx + dz * dz;
174 };
175
176 std::function<void(const Node&)> rec = [&](const Node& n) {
177 if (dist2_point_aabb_xz(n) > r2) return;
178 if (n.isLeaf()) {
179 if (n.chunk.GetEntityManager().GetEntityCount() > 0)
180 out.push_back(const_cast<SpatialChunk*>(&n.chunk));
181 return;
182 }
183 for (int i = 0; i < 4; ++i) if (n.child[i]) rec(*n.child[i]);
184 };
185 rec(*m_root);
186
187 return out;
188 }

190 static inline float Dist2PointAABB3D(const Math::Vec3f& p,
191 const Math::Vec2f& c,
192 const Math::Vec2f& e)
193 {
194 auto clamp = [](float v, float lo, float hi) { return v < lo ? lo : (v > hi ? hi : v); };
195 const float qx = clamp(p.x, c.x - e.x, c.x + e.x);
196 const float qz = clamp(p.z, c.y - e.y, c.y + e.y);
197 const float dx = p.x - qx, dz = p.z - qz;
198 return dx * dx + dz * dz;
199 }

201 std::vector<SpatialChunk*> CullChunksNear(const Math::Frustumf& fr,
202 const Math::Vec3f& camPos,
203 size_t maxCount = (std::numeric_limits<size_t>::max)()) const noexcept
204 {
205 struct Item { SpatialChunk* sc; float d2; };
206
207 std::vector<Item> items; items.reserve(128);
208 if (!m_root) return {};
209
210 // 可視葉 (chunk, bounds) を同時に収集する内部再帰
211 std::function<void(const Node&)> rec = [&](const Node& n) {
212 if (!nodeIntersectsFrustum(n, fr, -m_minLeaf, m_minLeaf)) return;
213 if (n.isLeaf()) {
214 Math::Vec2f c = (n.bounds.lb + n.bounds.ub) * 0.5f;
215 c.x += levelOriginWS.x;
216 c.y += levelOriginWS.z;
217 const Math::Vec2f e = (n.bounds.ub - n.bounds.lb) * 0.5f;
218 const float d2 = this->Dist2PointAABB3D(camPos, c, e);
219 if (n.chunk.GetEntityManager().GetEntityCount() > 0)
220 items.push_back({ const_cast<SpatialChunk*>(&n.chunk), d2 });
221 return;
222 }
223 for (int i = 0; i < 4; ++i) if (n.child[i]) rec(*n.child[i]);
224 };
225 rec(*m_root);
226
227 if (items.empty()) return {};
228 const size_t K = (std::min)(maxCount, items.size());
229 std::nth_element(items.begin(), items.begin() + K, items.end(),
230 [](const Item& a, const Item& b) { return a.d2 < b.d2; });
231 items.resize(K);
232 std::sort(items.begin(), items.end(),
233 [](const Item& a, const Item& b) { return a.d2 < b.d2; });
234
235 std::vector<SpatialChunk*> out; out.reserve(K);
236 for (auto& it : items) out.push_back(it.sc);
237 return out;
238 }

263 uint32_t CullChunkLine(const Math::Frustumf& fr,
264 Math::Vec3f cp, float hy, Debug::LineVertex* outLine,
265 uint32_t capacity, uint32_t displayCount) const noexcept
266 {
267 if (!m_root || capacity < 6 || displayCount == 0) return 0;
268
269 std::vector<AABB> boxes;
270 boxes.reserve(64);
271
272 cullRecursive(*m_root, fr, cp.y - hy, cp.y + hy, boxes);
273
274 struct Item { AABB box; float dist; };
275 std::vector<Item> items; items.reserve(boxes.size());
276 auto clampToBox = [&](const AABB& b, const Math::Vec3f& p) {
277 return Math::Vec2f{
278 std::clamp(p.x, b.lb.x + levelOriginWS.x, b.ub.x + levelOriginWS.x),
279 std::clamp(p.z, b.lb.y + levelOriginWS.z, b.ub.y + levelOriginWS.z)
280 };
281 };
282 for (const auto& b : boxes) {
283 const Math::Vec2f q = clampToBox(b, cp);
284 Math::Vec2f vec{ q.x - cp.x, q.y - cp.z };
285 const float d = vec.length();
286 items.push_back({ b, d });
287 }
288 std::nth_element(items.begin(), items.begin() + std::min<size_t>(displayCount, items.size()), items.end(),
289 [](const Item& a, const Item& b) { return a.dist < b.dist; });
290 const size_t useN = std::min<size_t>(displayCount, items.size());
291
292 // 3) 色は距離グラデ（近=白→遠=黒）
293 float maxD = 0.f; for (size_t i = 0; i < useN; ++i) maxD = (std::max)(maxD, items[i].dist);
294 if (maxD <= 1e-6f) maxD = 1.f;
295
296 uint32_t written = 0;
297 for (const auto& box : boxes) {
298 const Math::Vec2f center = Math::Vec2f{ levelOriginWS.x, levelOriginWS.z } + box.center();
299 const Math::Vec2f extent = box.size() * 0.5f;
300
301 const Math::Vec2f vec{ center.x - cp.x, center.y - cp.z };
302 const float len = vec.length();
303
304 if (len > maxD) continue; // 表示距離外
305
306 if (capacity - written < 8) break; // 必須：容量チェック
307
308 const uint32_t rgba = Math::LerpColor(0xFFFFFFFFu, 0x00000000u, len / maxD);
309
310 outLine[written + 0] = { Math::Vec3f(center.x - extent.x, cp.y - hy, center.y - extent.y), rgba };
311 outLine[written + 1] = { Math::Vec3f(center.x - extent.x, cp.y + hy, center.y - extent.y), rgba };
312 outLine[written + 2] = { Math::Vec3f(center.x + extent.x, cp.y - hy, center.y - extent.y), rgba };
313 outLine[written + 3] = { Math::Vec3f(center.x + extent.x, cp.y + hy, center.y - extent.y), rgba };
314 outLine[written + 4] = { Math::Vec3f(center.x - extent.x, cp.y - hy, center.y + extent.y), rgba };
315 outLine[written + 5] = { Math::Vec3f(center.x - extent.x, cp.y + hy, center.y + extent.y), rgba };
316 outLine[written + 6] = { Math::Vec3f(center.x + extent.x, cp.y - hy, center.y + extent.y), rgba };
317 outLine[written + 7] = { Math::Vec3f(center.x + extent.x, cp.y + hy, center.y + extent.y), rgba };
318
319 written += 8;
320 }
321 return written;
322 }

336 SpatialChunk* EnsureLeafForPoint(Math::Vec3f p)
337 {
338 if (!inBounds(p.x, p.z)) {
339 p.x = std::clamp(p.x, 0.f, float(m_worldW) - 1e-6f);
340 p.z = std::clamp(p.z, 0.f, float(m_worldH) - 1e-6f);
341 }
342 Node* n = m_root.get();
343 while (canSplit(*n)) {
344 if (n->isLeaf()) ensureChildren(*n); // 葉のとき子を作る
345 const int qi = quadrant(*n, p.x, p.z); //xz 統一
346 n = n->child[qi].get();
347 }
348 return &n->chunk;
349 }

500 static bool intersects(const AABB& a, const AABB& b) noexcept {
501 return !(a.ub.x <= b.lb.x || a.lb.x >= b.ub.x || a.ub.y <= b.lb.y || a.lb.y >= b.ub.y);
502 }
503 static bool intersects(const AABB& b, const Circle& c) noexcept {
504 const float cx = std::clamp(c.cx, b.lb.x, b.ub.x);
505 const float cy = std::clamp(c.cy, b.lb.y, b.ub.y);
506 const float dx = cx - c.cx, dy = cy - c.cy;
507 return (dx * dx + dy * dy) <= (c.r * c.r);
508 }
509 bool inBounds(float x, float z) const noexcept {
510 return (0.f <= x && x < float(m_worldW)) && (0.f <= z && z < float(m_worldH));
511 }
512 bool canSplit(const Node& n) const noexcept {
513 const float w = n.bounds.ub.x - n.bounds.lb.x;
514 const float h = n.bounds.ub.y - n.bounds.lb.y;
515 return (w > m_minLeaf) && (h > m_minLeaf);
516 }
517 int quadrant(const Node& n, float x, float z) const noexcept {
518 const float mx = 0.5f * (n.bounds.lb.x + n.bounds.ub.x);
519 const float mz = 0.5f * (n.bounds.lb.y + n.bounds.ub.y);
520
521 const bool east = (x >= mx);
522 bool north = (z >= mz); // デフォは North
523 if constexpr (kZPositive == ZPositive::South) north = !north;
524 // 0: NW, 1: NE, 2: SW, 3: SE（X?Zでの方位）
525 return (north ? 0 : 2) + (east ? 1 : 0);
526 }

528 void ensureChildren(Node& n)
529 {
530 if (!n.isLeaf()) return;
531 const float mx = 0.5f * (n.bounds.lb.x + n.bounds.ub.x);
532 const float mz = 0.5f * (n.bounds.lb.y + n.bounds.ub.y);
533 const AABB quads[4] = {
534 { { n.bounds.lb.x, mz }, { mx, n.bounds.ub.y } }, // NW
535 { { mx, mz }, { n.bounds.ub.x, n.bounds.ub.y } }, // NE
536 { { n.bounds.lb.x, n.bounds.lb.y }, { mx, mz } }, // SW
537 { { mx, n.bounds.lb.y }, { n.bounds.ub.x, mz } } // SE
538 };
539 for (int i = 0; i < 4; ++i) {
540 n.child[i] = std::make_unique<Node>();
541 n.child[i]->bounds = quads[i];
542 n.child[i]->depth = uint8_t(n.depth + 1);
543 n.child[i]->generation = 0;
544 }
545 m_leafCount += 3; // 旧葉1 -> 子4 で +3
546 }

548 Node* descendToLeaf(Node& cur, float x, float z, bool createIfMissing) {
549 auto posFn = [](ECS::EntityID id, ECS::EntityManager& mgr) -> std::optional<Math::Vec3f> {
550 auto tf = mgr.ReadComponent<CTransform>(id);
551 if (!tf.has_value()) return std::nullopt;
552
553 return tf->location;
554 };
555
556 Node* n = &cur;
557 while (canSplit(*n)) {
558 if (n->isLeaf()) {
559 if (!createIfMissing) break;
560
561 if (n->chunk.GetEntityManager().GetEntityCount() > 0) {
562 // 子化に伴う“再配置 Split”を先に実行（ensureChildrenは内部でやる派）
563 subdivideAndReassign(*n, posFn); // Entity を子葉へ全て移す
564 }
565 else {
566 ensureChildren(*n); // 中身が空なら生成だけ
567 }
568 }
569 const int qi = quadrant(*n, x, z);
570 n = n->child[qi].get();
571 }
572 return n;
573 }

594 void queryCircle(const Node& n, const Circle& c, std::vector<const SpatialChunk*>& out) const
595 {

720 {
721 if (!leaf.isLeaf() || !canSplit(leaf)) return;

724 ensureChildren(leaf);
725
726 ECS::EntityManager& src = leaf.chunk.GetEntityManager();

730 const std::optional<Math::Vec3f> pos = posFn(id, src); // 位置は "今ここ" の EM から見える
731 if (!pos) return nullptr;

733 return &leaf.child[qi]->chunk.GetEntityManager(); // 宛先 EM
734 };

743
744 // 4) 親ノード世代を更新（RegisterAllChunks 時にキーへ反映）
745 ++leaf.generation;
746 }

752 size_t CoalesceUnderutilized()
753 {
754 if (!m_root) return 0;
755 std::vector<Node*> stack; stack.push_back(m_root.get());
756 std::vector<Node*> post;
757 while (!stack.empty()) {
758 Node* n = stack.back(); stack.pop_back();
759 post.push_back(n);
760 for (auto& c : n->child) if (c) stack.push_back(c.get());
761 }
762 size_t mergedParents = 0;
763
764 // 子→親の順で評価（後置）
765 for (auto it = post.rbegin(); it != post.rend(); ++it) {
766 Node* n = *it;
767 // 子が4つ揃っていて全て葉？
768 bool allLeaf = true;
769 for (int i = 0; i < 4; ++i) { if (!n->child[i] || !n->child[i]->isLeaf()) { allLeaf = false; break; } }
770 if (!allLeaf) continue;
771
772 // 合計件数
773 size_t sum = 0;
774 for (int i = 0; i < 4; ++i) sum += n->child[i]->chunk.GetEntityManager().GetEntityCount();
775
776 if (sum <= m_minPerLeafCount) {
777 // 1) 子→親へ統合
778 auto& dst = n->chunk.GetEntityManager();
779 for (int i = 0; i < 4; ++i) {
780 auto& src = n->child[i]->chunk.GetEntityManager();
781 (void)dst.MergeFromAll(src); // 全移送（非スパース＋スパース）:contentReference[oaicite:5]{index=5} :contentReference[oaicite:6]{index=6}
782 }
783 // 2) 子を破棄して葉に戻す
784 n->child = {};
785 ++n->generation;
786 m_leafCount -= 3; // 4->1
787 ++mergedParents;
788 }
789 }
790 return mergedParents;
791 }
