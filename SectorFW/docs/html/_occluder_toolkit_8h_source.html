156 // ----- Reuse render LOD thresholds for Occluder LOD -----
157 // Generic LOD selector with hysteresis & bias.
158 // TTh must have: std::array<float,N> T; float hysteresisUp; float hysteresisDown;
159 template<class TTh>
160 int SelectLodGeneric(float s, const TTh& th, int lodCount, int prevLod, float globalBias)
161 {
162 if (lodCount <= 1) return 0;
163 float biasScale = std::pow(2.0f, globalBias);
164 auto t = [&](int i, bool up) {
165 float h = up ? (1.0f + th.hysteresisUp) : (1.0f - th.hysteresisDown);
166 return th.Tpx[i] * biasScale * (1.0f - 0.1f * i) * h;
167 };
168 bool goingUp = (prevLod > 0 && s > th.Tpx[prevLod - 1]);
169 if (s > t(0, goingUp)) return 0;
170 if (lodCount == 2) return 1;
171 if (s > t(1, goingUp)) return 1;
172 if (lodCount == 3) return 2;
173 if (s > t(2, goingUp)) return 2;
174 return (std::min)(lodCount - 1, 3);
175 }

177 // Make occluder thresholds stricter & widen hysteresis to stabilize.
178 template<class TTh>
179 TTh MakeOccluderThresholds(const TTh& visTh, float scale = 1.25f,
180 float up = 0.25f, float down = 0.03f)
181 {
182 TTh th = visTh;
183 for (size_t i = 0; i < th.Tpx.size(); ++i) th.Tpx[i] *= scale;
184 th.hysteresisUp = (std::max)(th.hysteresisUp, up);
185 th.hysteresisDown = (std::max)(th.hysteresisDown, down);
186 return th;
187 }

198 // Decide occluder LOD using thresholds (recommended).
199 template<class TTh>
200 OccluderLOD DecideOccluderLOD_FromThresholds(float s_occ,
201 const TTh& visTh,
202 int prevOccLod /*0..2*/,
203 int renderLod /*0..*/,
204 float extraGlobalBias = 0.0f)
205 {
206 TTh occTh = MakeOccluderThresholds(visTh);
207 const int occLodCount = 3;
208 float globalBias = OccluderBiasFromRenderLod(renderLod) + extraGlobalBias;
209 int prevIdx = std::clamp(prevOccLod, 0, occLodCount - 1);
210 int idx = SelectLodGeneric(s_occ, occTh, occLodCount, prevIdx, globalBias);
211 if (idx <= 0) return OccluderLOD::Near;
212 if (idx == 1) return OccluderLOD::Mid;
213 return OccluderLOD::Far;
214 }
