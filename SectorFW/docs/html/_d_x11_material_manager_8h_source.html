40 struct MaterialBindingCache {
41 bool valid = false; // キャッシュが有効かどうか
42 bool contiguous = false;
43 UINT minSlot = 0;
44 UINT count = 0;
45 std::vector<CacheType> contiguousViews;
46 std::vector<std::pair<UINT, CacheType>> individualViews;

52 void PushOrOverwrite(std::pair<UINT, CacheType> bind) {
53 if (count == 0 || valid == false)
54 {
55 contiguousViews.clear();
56 individualViews.clear();
57
58 valid = true;
59 contiguous = true;
60 minSlot = bind.first;
61 count = 1;
62 contiguousViews.push_back(bind.second);
63 return;
64 }
65
66 if (contiguous)
67 {
68 auto slot = bind.first;
69 auto maxSlot = minSlot + count;
70 // 範囲内なら上書き
71 if (minSlot <= slot && slot < maxSlot)
72 {
73 contiguousViews[slot - minSlot] = bind.second;
74 }
75 //範囲外だけど連続的(先頭)
76 else if (slot == (std::max)(minSlot, (UINT)1) - 1)
77 {
78 contiguousViews.resize(contiguousViews.size() + 1);
79 for (int i = (int)contiguousViews.size() - 1; i > 0; --i)
80 {
81 contiguousViews[i] = contiguousViews[i - 1];
82 }
83 contiguousViews[0] = bind.second;
84 minSlot = slot;
85 }
86 //範囲外だけど連続的(末尾)
87 else if (slot == maxSlot)
88 {
89 contiguousViews.push_back(bind.second);
90 count++;
91 }
92 //範囲外で連続的でない
93 else
94 {
95 contiguous = false;
96 for (UINT i = minSlot; i < maxSlot; ++i)
97 {
98 individualViews.emplace_back(i, contiguousViews[i - minSlot]);
99 }
100 contiguousViews.clear();
101 individualViews.emplace_back(slot, bind.second);
102 minSlot = (std::min)(minSlot, slot);
103 count++;
104 }
105 }
106 else
107 {
108 std::bitset<128> usedSlots;
109 for (auto i = 0; i < individualViews.size(); ++i)
110 {
111 // 既存のスロットなら上書き
112 if (individualViews[i].first == bind.first)
113 {
114 individualViews[i].second = bind.second;
115 return;
116 }
117 usedSlots.set(individualViews[i].first);
118 }
119
120 individualViews.emplace_back(bind.first, bind.second);
121 count++;
122 usedSlots.set(bind.first);
123 minSlot = (std::min)(minSlot, bind.first);
124 auto maxSlot = minSlot + count;
125 for (UINT i = minSlot; i <= maxSlot; ++i) {
126 if (!usedSlots.test(i)) {
127 return; // 連続でない
128 }
129 }
130
131 // 連続になった
132 contiguous = true;
133 contiguousViews.resize(count);
134 for (const auto& [slot, view] : individualViews) {
135 contiguousViews[slot - minSlot] = view;
136 }
137 individualViews.clear();
138 }
139 }
140 };
