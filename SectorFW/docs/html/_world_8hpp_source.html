94 void LoadLevel(const std::string levelName, IThreadExecutor* executor = nullptr, bool active = true, ExecutedCustomFunc loadedCustom = nullptr)
95 {
96 LevelCustumFunc(levelName, [&](auto& holder) {
97 auto& level = holder.level;
98
99 // 1) メイン側で多重ロード防止（必須）
100 if (!level->TryBeginLoading()) {
101 LOG_WARNING("Level {%s} is already loaded or loading.", levelName.c_str());
102
103 if (loadedCustom) loadedCustom(this);
104
105 return;
106 }
107
108 level->ChangeState(ELevelState::Loaded, false);// “ロード完了フラグを落とす
109
110 // 同期ロードなら従来通り
111 if (!executor)
112 {
113 if (holder.loadingFunc) holder.loadingFunc(&world.GetServiceLocator(), level.get());
114
115 ELevelState setBits = ELevelState::Loaded;
116 ELevelState clearBits = ELevelState::Loading;
117 if (active) {
118 setBits = setBits | ELevelState::Active;
119 }
120 else {
121 clearBits = clearBits | ELevelState::Active;
122 }
123
124 // ロード完了後、状態を更新
125 level->UpdateState(setBits, clearBits);
126
127 if (loadedCustom) loadedCustom(this);
128 return;
129 }
130
131 // 2) 非同期ロード：ワーカーに重い処理だけ投げる
132 auto* levelPtr = level.get(); // レベルが破棄されない限り有効
133 auto loadingFunc = holder.loadingFunc; // コピーして安全に（holder参照を掴まない）
134 auto* worldPtr = &world; // WorldはGameEngineが潰すまで生きてる想定
135
136 executor->Submit([worldPtr, levelPtr, loadingFunc, levelName, loadedCustom, active]()
137 {
138 // ここは “並列OKな処理だけ” に限定（GPU/レンダスレッド専用/World状態変更はNG）
139 if (loadingFunc) loadingFunc(&worldPtr->GetServiceLocator(), levelPtr);
140
141 // 3) 完了通知をメインに返す（Requestキューはmutexで守られてる）
142 auto& rs = worldPtr->GetRequestServiceNoLock();
143 rs.PushCommand(rs.CreateLambdaCommand(
144 [levelPtr, loadedCustom, active](Session* s, IThreadExecutor* ex)
145 {
146 ELevelState setBits = ELevelState::Loaded;
147 ELevelState clearBits = ELevelState::Loading;
148 if (active) {
149 setBits = setBits | ELevelState::Active;
150 }
151 else {
152 clearBits = clearBits | ELevelState::Active;
153 }
154
155 // メイン(=FlashAllCommand経由)で確定させる
156 levelPtr->UpdateState(setBits, clearBits);
157
158 if (loadedCustom) loadedCustom(s);
159 }
160 ));
161 });
162 });
163 }

165 void CleanLevel(const std::string levelName)
166 {
167 LevelCustumFunc(levelName, [&](auto& holder) {
168 auto& pLevel = holder.level;
169
170 auto state = pLevel->GetState();
171 // trueの場合Loadingビットを立てる
172 if (!pLevel->TryBeginClean()) {
173 // 未ロードまたはロード中の場合はスキップ
174 LOG_WARNING("Level {%s} is unloaded or loading.", levelName.c_str());
175 return;
176 };
177
178 // レベルを未ロード | 非アクティブにする
179 ELevelState clearBits = ELevelState::Loaded | ELevelState::Active;
180 pLevel->ChangeState(clearBits, false);
181
182 pLevel->Clean(world.serviceLocator);
183 if (holder.cleanFunc) {
184 holder.cleanFunc(&world.GetServiceLocator(), pLevel.get());
185 }
186
187 // クリーン完了後、Loadingビットを落とす
188 pLevel->ChangeState(ELevelState::Loading, false);
189 });

593 void UpdateAllLevels(double deltaTime, IThreadExecutor* executor) {
594#ifdef _ENABLE_IMGUI
595 {
596 auto g = Debug::BeginTreeWrite(); // lock & back buffer
597 auto& frame = g.data();
598 frame.items.clear();
599
600 // 例えばプリオーダ＋depth 指定で平坦化したツリーを詰める
601 frame.items.push_back({ /*id=*/frame.items.size(), /*depth=*/Debug::WorldTreeDepth::TREEDEPTH_WORLD, /*leaf=*/false, "World" });
602 } // guard のデストラクトで unlock。swap は UI スレッドで。
603#endif
604
605 std::vector<std::function<void(ECS::ServiceLocator&, double, IThreadExecutor*)>> mainLevelFunc;
606 std::vector<std::function<void(ECS::ServiceLocator&, double, IThreadExecutor*)>> subLevelFunc;
607
608 std::apply([&](auto&... levelVecs)
609 {
610 (..., [](auto& vecs, auto& mainFunc, auto& subFunc) {
611 for (auto& holder : vecs) {
612 auto& level = holder.level;
613 auto levelState = level->GetState();
614
615 // 非アクティブ | 未ロード | ローディング中 レベルはスキップ
616 if (!HasAll(levelState, ELevelState::Active | ELevelState::Loaded)
617 || HasAny(levelState, ELevelState::Loading)) {
618#ifdef _ENABLE_IMGUI
619 level->ShowDebugInactiveLevelInfoUI();
620#endif
621 continue;
622 }
623
624 if (HasAny(levelState, ELevelState::Main)) {
625 mainFunc.push_back([&level](auto& locator, double delta, auto* te) {
626 //levelの更新処理。inc/core/Level.hppのLevelクラスのUpdate関数を呼び出す
627 level->Update(locator, delta, te);
628 });
629 }
630 else if (HasAny(levelState, ELevelState::Sub)) {
631 subFunc.push_back([&level](auto& locator, double delta, auto* te) {
632 level->UpdateLimited(locator, delta, te);
633 });
634 }
635 }
636 }(levelVecs, mainLevelFunc, subLevelFunc));
637 }, levelSets);
638
639#ifdef _ENABLE_IMGUI
640 {
641 auto g = Debug::BeginTreeWrite(); // lock & back buffer
642 auto& frame = g.data();
643 frame.items.push_back({ /*id=*/frame.items.size(), /*depth=*/Debug::WorldTreeDepth::TREEDEPTH_LEVEL, /*leaf=*/false, "GlobalSystem" });
644 }
645
646 globalSystem.ShowDebugSystemTree((uint32_t)Debug::WorldTreeDepth::TREEDEPTH_LEVELNODE);
647#endif
648
649 ThreadCountDownLatch latch((int)mainLevelFunc.size());
650 //メインのレベルの更新処理を並行で実行
651 for (auto& f : mainLevelFunc)
652 {
653 executor->Submit(
654 [this,
655 task = std::move(f), // 「ラムダへの move キャプチャ」
656 deltaTime,
657 executor,
658 &latch]()
659 {
660 task(serviceLocator, deltaTime, executor);
661 latch.CountDown();
662 }
663 );
664 }
665
666 //グローバルシステムの更新処理実行
667 globalSystem.UpdateGlobal(serviceLocator, executor);
668
669 //サブレベルの更新処理実行
670 for (auto& f : subLevelFunc) f(serviceLocator, deltaTime, executor);
671
672 latch.Wait();
673 }
