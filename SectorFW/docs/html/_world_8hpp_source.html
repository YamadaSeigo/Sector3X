94 void LoadLevel(const std::string levelName, IThreadExecutor* executor = nullptr, bool active = true, ExecutedCustomFunc loadedCustom = nullptr)
95 {
96 LevelCustumFunc(levelName, [&](auto& holder) {
97 auto& level = holder.level;
98
99 // 1) メイン側で多重ロード防止（必須）
100 if (!level->TryBeginLoading()) {
101 LOG_WARNING("Level {%s} is already loaded or loading.", levelName.c_str());
102
103 if (loadedCustom) loadedCustom(this);
104
105 return;
106 }
107
108 level->ChangeState(ELevelState::Loaded, false);// “ロード完了フラグを落とす
109
110 // 同期ロードなら従来通り
111 if (!executor)
112 {
113 if (holder.loadingFunc) holder.loadingFunc(&world.GetServiceLocator(), level.get());
114
115 ELevelState setBits = ELevelState::Loaded;
116 ELevelState clearBits = ELevelState::Loading;
117 if (active) {
118 setBits = setBits | ELevelState::Active;
119 }
120 else {
121 clearBits = clearBits | ELevelState::Active;
122 }
123
124 // ロード完了後、状態を更新
125 level->UpdateState(setBits, clearBits);
126
127 if (loadedCustom) loadedCustom(this);
128 return;
129 }
130
131 // 2) 非同期ロード：ワーカーに重い処理だけ投げる
132 auto* levelPtr = level.get(); // レベルが破棄されない限り有効
133 auto loadingFunc = holder.loadingFunc; // コピーして安全に（holder参照を掴まない）
134 auto* worldPtr = &world; // WorldはGameEngineが潰すまで生きてる想定
135
136 executor->Submit([worldPtr, levelPtr, loadingFunc, levelName, loadedCustom, active]()
137 {
138 // ここは “並列OKな処理だけ” に限定（GPU/レンダスレッド専用/World状態変更はNG）
139 if (loadingFunc) loadingFunc(&worldPtr->GetServiceLocator(), levelPtr);
140
141 // 3) 完了通知をメインに返す（Requestキューはmutexで守られてる）
142 auto& rs = worldPtr->GetRequestServiceNoLock();
143 rs.PushCommand(rs.CreateLambdaCommand(
144 [levelPtr, loadedCustom, active](Session* s, IThreadExecutor* ex)
145 {
146 ELevelState setBits = ELevelState::Loaded;
147 ELevelState clearBits = ELevelState::Loading;
148 if (active) {
149 setBits = setBits | ELevelState::Active;
150 }
151 else {
152 clearBits = clearBits | ELevelState::Active;
153 }
154
155 // メイン(=FlashAllCommand経由)で確定させる
156 levelPtr->UpdateState(setBits, clearBits);
157
158 if (loadedCustom) loadedCustom(s);
159 }
160 ));
161 });
162 });
163 }

165 void CleanLevel(const std::string levelName)
166 {
167 LevelCustumFunc(levelName, [&](auto& holder) {
168 auto& pLevel = holder.level;
169
170 auto state = pLevel->GetState();
171 // trueの場合Loadingビットを立てる
172 if (!pLevel->TryBeginClean()) {
173 // 未ロードまたはロード中の場合はスキップ
174 LOG_WARNING("Level {%s} is unloaded or loading.", levelName.c_str());
175 return;
176 };
177
178 // レベルを未ロード | 非アクティブにする
179 ELevelState clearBits = ELevelState::Loaded | ELevelState::Active;
180 pLevel->ChangeState(clearBits, false);
181
182 pLevel->Clean(world.serviceLocator);
183 if (holder.cleanFunc) {
184 holder.cleanFunc(&world.GetServiceLocator(), pLevel.get());
185 }
186
187 // クリーン完了後、Loadingビットを落とす
188 pLevel->ChangeState(ELevelState::Loading, false);
189 });

568 void UpdateAllLevels(double deltaTime, IThreadExecutor* executor) {
569#ifdef _ENABLE_IMGUI
570 {
571 auto g = Debug::BeginTreeWrite(); // lock & back buffer
572 auto& frame = g.data();
573 frame.items.clear();
574
575 // 例えばプリオーダ＋depth 指定で平坦化したツリーを詰める
576 frame.items.push_back({ /*id=*/frame.items.size(), /*depth=*/Debug::WorldTreeDepth::TREEDEPTH_WORLD, /*leaf=*/false, "World" });
577 } // guard のデストラクトで unlock。swap は UI スレッドで。
578#endif
579 std::vector<std::function<void(ECS::ServiceLocator&, double, IThreadExecutor*)>> mainLevelFunc;
580 std::vector<std::function<void(ECS::ServiceLocator&, double, IThreadExecutor*)>> subLevelFunc;
581
582 std::apply([&](auto&... levelVecs)
583 {
584 (..., [](auto& vecs, auto& mainFunc, auto& subFunc) {
585 for (auto& holder : vecs) {
586 auto& level = holder.level;
587 auto levelState = level->GetState();
588
589 // 非アクティブ | 未ロード | ローディング中 レベルはスキップ
590 if (!HasAll(levelState, ELevelState::Active | ELevelState::Loaded)
591 || HasAny(levelState, ELevelState::Loading)) {
592#ifdef _ENABLE_IMGUI
593 level->ShowDebugInactiveLevelInfoUI();
594#endif
595 continue;
596 }
597
598 if (HasAny(levelState, ELevelState::Main)) {
599 mainFunc.push_back([&level](auto& locator, double delta, auto* te) {
600 level->Update(locator, delta, te);
601 });
602 }
603 else if (HasAny(levelState, ELevelState::Sub)) {
604 subFunc.push_back([&level](auto& locator, double delta, auto* te) {
605 level->UpdateLimited(locator, delta, te);
606 });
607 }
608 }
609 }(levelVecs, mainLevelFunc, subLevelFunc));
610 }, levelSets);
611
612#ifdef _ENABLE_IMGUI
613 {
614 auto g = Debug::BeginTreeWrite(); // lock & back buffer
615 auto& frame = g.data();
616 frame.items.push_back({ /*id=*/frame.items.size(), /*depth=*/Debug::WorldTreeDepth::TREEDEPTH_LEVEL, /*leaf=*/false, "GlobalSystem" });
617 }
618
619 globalSystem.ShowDebugSystemTree((uint32_t)Debug::WorldTreeDepth::TREEDEPTH_LEVELNODE);
620#endif
621
622 ThreadCountDownLatch latch((int)mainLevelFunc.size());
623 //メインのレベルの更新処理を並行で実行
624 for (auto& f : mainLevelFunc)
625 {
626 // コンテナの要素 f をタスク用に move
627 auto task = std::move(f);
628
629 executor->Submit(
630 [task = std::move(task), // 「ラムダへの move キャプチャ」
631 this,
632 deltaTime,
633 executor,
634 &latch]()
635 {
636 task(serviceLocator, deltaTime, executor);
637 latch.CountDown();
638 }
639 );
640 }
641
642 //グローバルシステムの更新処理実行
643 globalSystem.UpdateGlobal(serviceLocator, executor);
644
645 //サブレベルの更新処理実行
646 for (auto& f : subLevelFunc) f(serviceLocator, deltaTime, executor);
647
648 latch.Wait();
649 }
