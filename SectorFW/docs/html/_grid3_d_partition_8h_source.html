189 std::vector<SpatialChunk*> CullChunks(const Math::Vec3f& center, float radius) const noexcept
190 {
191 std::vector<SpatialChunk*> out;
192 if (!std::isfinite(radius) || radius <= 0.0f) {
193 radius = 0.0f;
194 }
195
196 using Signed = long long;
197 const float r = (radius < 0.0f) ? 0.0f : radius;
198 const float r2 = r * r;
199
200 const float cell = float(chunkSize);
201 const float e = 0.5f * cell;
202
203 auto floor_div = [&](float v) -> Signed {
204 return static_cast<Signed>(std::floor(double(v) / double(cell)));
205 };
206
207 Signed ix0 = floor_div(center.x - r);
208 Signed iy0 = floor_div(center.y - r);
209 Signed iz0 = floor_div(center.z - r);
210 Signed ix1 = floor_div(center.x + r);
211 Signed iy1 = floor_div(center.y + r);
212 Signed iz1 = floor_div(center.z + r);
213
214 const Signed w = static_cast<Signed>(grid.width());
215 const Signed h = static_cast<Signed>(grid.height());
216 const Signed d = static_cast<Signed>(grid.depth());
217
218 ix0 = std::clamp<Signed>(ix0, 0, w - 1);
219 iy0 = std::clamp<Signed>(iy0, 0, h - 1);
220 iz0 = std::clamp<Signed>(iz0, 0, d - 1);
221 ix1 = std::clamp<Signed>(ix1, 0, w - 1);
222 iy1 = std::clamp<Signed>(iy1, 0, h - 1);
223 iz1 = std::clamp<Signed>(iz1, 0, d - 1);
224
225 out.reserve(static_cast<size_t>((ix1 - ix0 + 1) * (iy1 - iy0 + 1) * (iz1 - iz0 + 1)));
226
227 for (Signed z = iz0; z <= iz1; ++z) {
228 for (Signed y = iy0; y <= iy1; ++y) {
229 for (Signed x = ix0; x <= ix1; ++x) {
230 const Math::Vec3f c = levelOriginWS + Math::Vec3f{ (float(x) + 0.5f) * cell, (float(y) + 0.5f) * cell, (float(z) + 0.5f) * cell };
231 const Math::Vec3f ex{ e, e, e };
232 if (Dist2PointAABB3D(center, c, ex) > r2) continue;
233
234 auto& chunk = grid(ChunkSizeType(x), ChunkSizeType(y), ChunkSizeType(z));
235 if (chunk.GetEntityManager().GetEntityCount() > 0)
236 out.push_back(const_cast<SpatialChunk*>(&chunk));
237 }
238 }
239 }
240 return out;
241 }

243 static inline float Dist2PointAABB3D(const Math::Vec3f& p,
244 const Math::Vec3f& c,
245 const Math::Vec3f& e)
246 {
247 auto clamp = [](float v, float lo, float hi) { return v < lo ? lo : (v > hi ? hi : v); };
248 const float qx = clamp(p.x, c.x - e.x, c.x + e.x);
249 const float qy = clamp(p.y, c.y - e.y, c.y + e.y);
250 const float qz = clamp(p.z, c.z - e.z, c.z + e.z);
251 const float dx = p.x - qx, dy = p.y - qy, dz = p.z - qz;
252 return dx * dx + dy * dy + dz * dz;
253 }

255 std::vector<SpatialChunk*> CullChunksNear(const Math::Frustumf& fr,
256 const Math::Vec3f& camPos,
257 size_t maxCount = (std::numeric_limits<size_t>::max)()) const noexcept
258 {
259 struct Item { SpatialChunk* sc; float d2; };
260 std::vector<Item> items; items.reserve(128);
261
262 const uint32_t w = grid.width(), h = grid.height(), d = grid.depth();
263 const float cell = float(chunkSize);
264 const float e = 0.5f * cell;
265
266 for (uint32_t z = 0; z < d; ++z) {
267 for (uint32_t y = 0; y < h; ++y) {
268 for (uint32_t x = 0; x < w; ++x) {
269 const Math::Vec3f center = levelOriginWS + Math::Vec3f{ (x + 0.5f) * cell, (y + 0.5f) * cell, (z + 0.5f) * cell };
270 const Math::Vec3f extent{ e, e, e };
271 if (!fr.IntersectsAABB(center, extent)) continue;
272
273 const float d2 = Dist2PointAABB3D(camPos, center, extent);
274 auto& chunk = grid(x, y, z);
275 if (chunk.GetEntityManager().GetEntityCount() > 0)
276 items.push_back({ const_cast<SpatialChunk*>(&chunk), d2 });
277 }
278 }
279 }
280
281 if (items.empty()) return {};
282 const size_t K = (std::min)(maxCount, items.size());
283 std::nth_element(items.begin(), items.begin() + K, items.end(),
284 [](const Item& a, const Item& b) { return a.d2 < b.d2; });
285 items.resize(K);
286 std::sort(items.begin(), items.end(),
287 [](const Item& a, const Item& b) { return a.d2 < b.d2; });
288
289 std::vector<SpatialChunk*> out; out.reserve(K);
290 for (auto& it : items) out.push_back(it.sc);
291 return out;
292 }

302 uint32_t CullChunkLine(const Math::Frustumf& fr,
303 Math::Vec3f cp,
304 float, // dummy
305 Debug::LineVertex* outLine,
306 uint32_t capacity,
307 uint32_t displayCount) const noexcept
308 {
309 const uint32_t w = grid.width(), h = grid.height(), d = grid.depth();
310 const float cell = float(chunkSize);
311 const float e = 0.5f * cell;
312
313 const uint32_t kVertsPerCell = 24; // 12本のエッジ × 2頂点
314 uint32_t written = 0;
315 const float maxLen = displayCount * chunkSize;
316
317 auto push_edge = [&](const Math::Vec3f& a, const Math::Vec3f& b, uint32_t rgb) {
318 if (capacity - written < 2) return false;
319 outLine[written + 0] = { a, rgb };
320 outLine[written + 1] = { b, rgb };
321 written += 2;
322 return true;
323 };
324
325 for (uint32_t z = 0; z < d; ++z) {
326 for (uint32_t y = 0; y < h; ++y) {
327 for (uint32_t x = 0; x < w; ++x) {
328 if (capacity - written < kVertsPerCell) return written; // これ以上書けない
329
330 const Math::Vec3f c = levelOriginWS + Math::Vec3f{ (x + 0.5f) * cell, (y + 0.5f) * cell, (z + 0.5f) * cell };
331 const Math::Vec3f ex{ e, e, e };
332
333 // 距離でフェード（XZ距離ではなく3D距離）
334 const float len = (c - cp).length();
335 if (len > maxLen) continue;
336
337 if (!fr.IntersectsAABB(c, ex)) continue;
338
339 const uint32_t rgb = Math::LerpColor(0xFFFFFFFFu, 0x000000FFu, len / maxLen);
340
341 // 8頂点
342 Math::Vec3f v000{ c.x - e, c.y - e, c.z - e };
343 Math::Vec3f v001{ c.x - e, c.y - e, c.z + e };
344 Math::Vec3f v010{ c.x - e, c.y + e, c.z - e };
345 Math::Vec3f v011{ c.x - e, c.y + e, c.z + e };
346 Math::Vec3f v100{ c.x + e, c.y - e, c.z - e };
347 Math::Vec3f v101{ c.x + e, c.y - e, c.z + e };
348 Math::Vec3f v110{ c.x + e, c.y + e, c.z - e };
349 Math::Vec3f v111{ c.x + e, c.y + e, c.z + e };
350
351 // 12エッジ
352 push_edge(v000, v001, rgb); // -x,-y, z
353 push_edge(v000, v010, rgb); // -x, y,-z
354 push_edge(v000, v100, rgb); // x,-y,-z
355 push_edge(v111, v101, rgb);
356 push_edge(v111, v110, rgb);
357 push_edge(v111, v011, rgb);
358 push_edge(v010, v011, rgb);
359 push_edge(v010, v110, rgb);
360 push_edge(v100, v101, rgb);
361 push_edge(v100, v110, rgb);
362 push_edge(v001, v011, rgb);
363 push_edge(v001, v101, rgb);
364 }
365 }
366 }
367 return written;
