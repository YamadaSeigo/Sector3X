29 struct DesignerHeightMap
30 {
31 uint32_t width = 0;
32 uint32_t height = 0;
33 std::vector<float> data; // 0..1
34
35 bool IsValid() const { return width > 0 && height > 0 && !data.empty(); }
36
37 // u,v: 0..1 （タイル全体に対する UV）
38 float Sample(float u, float v) const
39 {
40 if (!IsValid()) return 0.0f;
41
42 u = std::clamp(u, 0.0f, 1.0f);
43 v = std::clamp(v, 0.0f, 1.0f);
44
45 float x = u * (width - 1);
46 float y = v * (height - 1);
47
48 int x0 = (int)std::floor(x);
49 int y0 = (int)std::floor(y);
50 int x1 = (std::min)(x0 + 1, (int)width - 1);
51 int y1 = (std::min)(y0 + 1, (int)height - 1);
52
53 float tx = x - (float)x0;
54 float ty = y - (float)y0;
55
56 float h00 = data[y0 * width + x0];
57 float h10 = data[y0 * width + x1];
58 float h01 = data[y1 * width + x0];
59 float h11 = data[y1 * width + x1];
60
61 float hx0 = std::lerp(h00, h10, tx);
62 float hx1 = std::lerp(h01, h11, tx);
63 return std::lerp(hx0, hx1, ty); // 0..1
64 }
65 };
