21 // パーマネーションテーブル 256*2
22 std::vector<int> base(256);
23 for (int i = 0; i < 256; i++) base[i] = i;
24 std::mt19937 rng(seed);
25 std::shuffle(base.begin(), base.end(), rng);
26 for (int i = 0; i < 256; i++) { perm[i] = base[i]; perm[256 + i] = base[i]; }
27 }

29 // 単オクターブの Perlin ノイズ（-1..1）
30 float noise(float x, float y) const {
31 int xi = fastfloor(x) & 255;
32 int yi = fastfloor(y) & 255;
33
34 float xf = x - std::floor(x);
35 float yf = y - std::floor(y);
36
37 float u = fade(xf);
38 float v = fade(yf);
39
40 int aa = perm[perm[xi] + yi];
41 int ab = perm[perm[xi] + yi + 1];
42 int ba = perm[perm[xi + 1] + yi];
43 int bb = perm[perm[xi + 1] + yi + 1];
44
45 float x1 = lerp(grad(aa, xf, yf),
46 grad(ba, xf - 1.0f, yf), u);
47 float x2 = lerp(grad(ab, xf, yf - 1.0f),
48 grad(bb, xf - 1.0f, yf - 1.0f), u);
49 return lerp(x1, x2, v); // -1..1
50 }

52 // fBm（オクターブ合成）
53 float fbm(float x, float y, int oct, float lac, float gain) const {
54 float amp = 1.0f;
55 float freq = 1.0f;
56 float sum = 0.0f, norm = 0.0f;
57 for (int i = 0; i < oct; i++) {
58 sum += amp * noise(x * freq, y * freq);
59 norm += amp;
60 freq *= lac;
61 amp *= gain;
62 }
63 return (norm > 0.0f) ? (sum / norm) : 0.0f; // -1..1 を平均化
64 }

69 static inline int fastfloor(float x) { return (x >= 0) ? int(x) : int(x) - 1; }
70 static inline float fade(float t) { return t * t * t * (t * (t * 6 - 15) + 10); }
71 static inline float lerp(float a, float b, float t) { return a + t * (b - a); }
72 static inline float grad(int h, float x, float y) {
73 // 8 方向グラディエント
74 int g = h & 7; // 0..7
75 float u = (g < 4) ? x : y;
76 float v = (g < 4) ? y : x;
77 return ((g & 1) ? -u : u) + ((g & 2) ? -2.0f * v : 2.0f * v);
78 }
