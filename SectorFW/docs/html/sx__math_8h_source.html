59 // 近似等価: |a-b| <= max(abs_tol, rel_tol * max(|a|,|b|))
60 template<Arithmetic T, class R = std::conditional_t<std::is_integral_v<T>, double, T>>
61 [[nodiscard]] constexpr bool approx_equal(T a, T b,
62 R rel_tol = R(1e-6),
63 R abs_tol = R(0)) noexcept {
64 const R ar = static_cast<R>(a), br = static_cast<R>(b);
65 const R diff = std::abs(ar - br);
66 const R limit = (std::max)(abs_tol, rel_tol * (std::max)(std::abs(ar), std::abs(br)));
67 return diff <= limit;
68 }

86 // 線形補間（非クランプ）。浮動小数はFMAを使用
87 template<Arithmetic A, Arithmetic B, Arithmetic U>
88 [[nodiscard]] constexpr auto lerp(A a, B b, U t) noexcept {
89 using R = std::common_type_t<A, B, U>;
90 const R ar = static_cast<R>(a), br = static_cast<R>(b), tr = static_cast<R>(t);
91 if constexpr (std::is_floating_point_v<R>) {
92 return std::fma(tr, (br - ar), ar); // ar + (br - ar) * tr
93 }
94 else {
95 return ar + (br - ar) * tr;
96 }
97 }

106 // (a..b) → t 逆補間
107 template<Arithmetic A, Arithmetic B, Arithmetic X>
108 [[nodiscard]] constexpr auto inverse_lerp(A a, B b, X x) noexcept {
109 using R = std::common_type_t<A, B, X>;
110 const R ar = static_cast<R>(a), br = static_cast<R>(b), xr = static_cast<R>(x);
111 const R denom = br - ar;
112 if (denom == R(0)) return R(0);
113 return (xr - ar) / denom;

149 // ラップ・モジュロ
150 //-------------------------------------
151 template<Arithmetic X, Arithmetic A, Arithmetic B>
152 [[nodiscard]] constexpr auto wrap(X x, A lo, B hi) noexcept {
153 using R = std::common_type_t<X, A, B>;
154 const R l = static_cast<R>(lo), h = static_cast<R>(hi);
155 const R w = h - l;
156 if constexpr (std::is_floating_point_v<R>) {
157 R m = std::fmod(static_cast<R>(x) - l, w);
158 if (m < R(0)) m += w;
159 return m + l;
160 }
161 else {
162 R v = static_cast<R>(x) - l;
163 R m = v % w;
164 if (m < R(0)) m += w;
165 return m + l;
166 }
167 }

265 uint8_t ng = extract(nearColor, 16);

272 uint8_t fa = extract(farColor, 0);

276 };

279 uint8_t g = lerp(ng, fg);
280 uint8_t b = lerp(nb, fb);
281 uint8_t a = lerp(na, fa);
