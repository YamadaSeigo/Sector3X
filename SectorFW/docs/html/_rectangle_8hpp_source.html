77 // AABB 8頂点を NDC に投影（列ベクトル規約 / row-major）
78 // ・VP = P * V を渡すこと（列ベクトルの標準）
79 // ・clip.w <= 0（視点背面）は NDC 化できないので除外
80 // ・GL の −1..1 にも対応（分岐）
81 //======================================================================
82 inline NdcRectWithW ProjectAABBToNdc(const Matrix4x4f& VP,
83 const float lbx, const float lby, const float lbz,
84 const float ubx, const float uby, const float ubz,
85 ClipZRange zrange = ClipZRange::ZeroToOne) noexcept
86 {
87 const float xs[8] = { lbx, ubx, lbx, ubx, lbx, ubx, lbx, ubx };
88 const float ys[8] = { lby, lby, uby, uby, lby, lby, uby, uby };
89 const float zs[8] = { lbz, lbz, lbz, lbz, ubz, ubz, ubz, ubz };
90
91 float xmin = FLT_MAX, ymin = FLT_MAX, xmax = -FLT_MAX, ymax = -FLT_MAX, wmin = FLT_MAX;
92 bool any = false;
93
94 for (int i = 0; i < 8; ++i) {
95 float cx, cy, cz, cw;
96 MulPoint_RowMajor_ColVec(VP, xs[i], ys[i], zs[i], cx, cy, cz, cw);
97
98 // 背面（cw<=0）は NDC にできない。必要ならここで近似クリップ。
99 if (cw <= 0.0f) continue;
100
101 float ndcX = cx / cw;
102 float ndcY = cy / cw;
103 float ndcZ = cz / cw;
104
105 // Z 範囲の正規化（必要なら −1..1 → 0..1 へ寄せる or 逆）
106 if (zrange == ClipZRange::NegOneToOne) {
107 // GL: z∈[-1,1] をそのまま持つ。ここでは矩形算出なので z は未使用。
108 }
109 else {
110 // DX: z∈[0,1] 前提。ここも矩形算出では未使用。
111 }
112
113 xmin = (std::min)(xmin, ndcX); xmax = (std::max)(xmax, ndcX);
114 ymin = (std::min)(ymin, ndcY); ymax = (std::max)(ymax, ndcY);
115 wmin = (std::min)(wmin, cw);
116 any = true;
117 }
118
119 // 1点も NDC 化できなければ invalid
120 if (!any) return { 0,0,0,0, wmin, false };
121
122 // NDC は通常 [-1,1]。ここではそのまま返す（後でピクセルへ変換）
123 return { xmin, ymin, xmax, ymax, wmin, true };
124 }

127 // NDC 矩形 → ピクセル矩形 変換
128 // ・NDC x,y ∈ [-1,1] を viewport へ
129 // ・originTopLeft=true: y ピクセル上向き→下向き（D3D標準）
130 //======================================================================
131 inline Rectangle NdcToPixels(const NdcRectWithW& ndc, const ViewportRect& vp) noexcept
132 {
133 if (!ndc.valid || vp.width <= 0 || vp.height <= 0) return {};
134
135 // NDC [-1,1] → [0,1]
136 const float nx0 = (ndc.xmin * 0.5f) + 0.5f;
137 const float ny0 = (ndc.ymin * 0.5f) + 0.5f;
138 const float nx1 = (ndc.xmax * 0.5f) + 0.5f;
139 const float ny1 = (ndc.ymax * 0.5f) + 0.5f;
140
141 Rectangle r{};
142 r.valid = true;
143
144 // x: 0..1 → 0..W
145 r.x0 = nx0 * vp.width;
146 r.x1 = nx1 * vp.width;
147
148 // y: 上端の扱い
149 if (vp.originTopLeft) {
150 // D3D/UI: (0,0) が上左。NDC +1 が上、-1 が下 なので反転してから高さを掛ける
151 const float py0 = (1.0f - ny0) * vp.height;
152 const float py1 = (1.0f - ny1) * vp.height;
153 r.y0 = (std::min)(py0, py1);
154 r.y1 = (std::max)(py0, py1);
155 }
156 else {
157 // GL 的下原点
158 r.y0 = ny0 * vp.height;
159 r.y1 = ny1 * vp.height;
160 }
161
162 r.Normalize();
163 return r;
164 }

196 inline bool TrivialRejectClip(const std::array<Vec4f, 8>& clip) {
197 // クリップ空間での6面に対する同側外側判定 (D3D: 近面 z>=0, 遠面 z<=w)
198 // x < -w / x > w / y < -w / y > w / z < 0 / z > w
199 auto all_out = [&](auto pred)->bool {
200 for (const auto& c : clip) if (!pred(c)) return false;
201 return true;
202 };
203 if (all_out([](const Vec4f& c) { return c.x < -c.w; })) return true; // left
204 if (all_out([](const Vec4f& c) { return c.x > c.w; })) return true; // right
205 if (all_out([](const Vec4f& c) { return c.y < -c.w; })) return true; // bottom
206 if (all_out([](const Vec4f& c) { return c.y > c.w; })) return true; // top
207 if (all_out([](const Vec4f& c) { return c.z < 0.0f; })) return true; // near
208 if (all_out([](const Vec4f& c) { return c.z > c.w; })) return true; // far
209 return false;
210 }

214 {0,1},{1,3},{3,2},{2,0}, // bottom face (y=0 側と仮定したいが順序は角の定義次第、ここでは頂点配列順に依存しない)
215 {4,5},{5,7},{7,6},{6,4}, // top face

219 // 近面 z=0 に対するエッジクリップ (clip空間上での線分交点)
220 inline bool IntersectEdgeWithNearZ0(const Vec4f& a, const Vec4f& b, Vec4f& out) {
221 const float za = a.z, zb = b.z;
222 const bool ina = (za >= 0.0f), inb = (zb >= 0.0f);
223 if (ina == inb) return false; // 片方のみが内側のときだけ交点がある
224 const float t = za / (za - zb); // z=0 での比例
225 // 同次座標は線形補間でOK（クリップ空間）
226 out.x = a.x + (b.x - a.x) * t;
227 out.y = a.y + (b.y - a.y) * t;
228 out.z = 0.0f;
229 out.w = a.w + (b.w - a.w) * t;
230 return out.w > 0.0f; // 後段の除算保護
231 }

233 // 互換API（既存呼び出し箇所はそのまま）
234 // 1) Rectangle を返す版
235 inline Rectangle ProjectAABBToScreenRect(
236 const AABB<float, Vec3f>& box,
237 const Matrix<4, 4, float>& worldViewProj,
238 float viewportWidth, float viewportHeight,
239 float viewportX = 0.0f, float viewportY = 0.0f,
240 float clampMargin = 0.0f)
241 {
242 Rectangle out{};
243 // ---- 8コーナー ----
244 const Vec3f& L = box.lb;
245 const Vec3f& U = box.ub;
246 std::array<Vec3f, 8> corners = {
247 Vec3f{L.x,L.y,L.z}, Vec3f{U.x,L.y,L.z}, Vec3f{L.x,U.y,L.z}, Vec3f{U.x,U.y,L.z},
248 Vec3f{L.x,L.y,U.z}, Vec3f{U.x,L.y,U.z}, Vec3f{L.x,U.y,U.z}, Vec3f{U.x,U.y,U.z}
249 };
250
251 // ---- クリップ空間へ ----
252 std::array<Vec4f, 8> clip{};
253 for (int i = 0; i < 8; ++i) clip[i] = ToClip(worldViewProj, corners[i]);
254
255 // ---- 6面の同側外側なら不可視（早期リジェクト）----
256 if (TrivialRejectClip(clip)) {
257 // 両方とも不可視のまま返す（screen.visible=false, ndc.valid=false）
258 return out;
259 }
260
261 // ---- 候補点（z>=0,w>0 の頂点 + 近面(z=0)交点）を収集 ----
262 // これを元に NDC/Screen の min/max と wmin を同時計算
263 std::array<Vec4f, 20> cand{}; int n = 0;
264
265 for (int i = 0; i < 8; ++i) {
266 if (clip[i].z >= 0.0f && clip[i].w > 0.0f) cand[n++] = clip[i];
267 }
268 for (const auto& e : kEdges) {
269 Vec4f inter;
270 if (IntersectEdgeWithNearZ0(clip[e[0]], clip[e[1]], inter)) {
271 if (n < 20) cand[n++] = inter;
272 }
273 }
274 if (n == 0) {
275 // 近面の背後のみ → 画面/ndcともに無効
276 return out;
277 }
278
279 // ---- Screen の min/max と wmin を 1ループで構築 ----
280
281 float scrMinX = +std::numeric_limits<float>::infinity();
282 float scrMinY = +std::numeric_limits<float>::infinity();
283 float scrMaxX = -std::numeric_limits<float>::infinity();
284 float scrMaxY = -std::numeric_limits<float>::infinity();
285
286 float wmin = +std::numeric_limits<float>::infinity();
287
288 for (int i = 0; i < n; ++i) {
289 const Vec4f& c = cand[i];
290 if (c.w <= 0.0f) continue; // 念のため
291
292 // wmin（クリップ空間のW。最前の“保守的W” = 最小w）
293 wmin = (std::min)(wmin, c.w);
294
295 // NDC
296 const float invw = 1.0f / c.w;
297 const float x = c.x * invw; // [-1,1]
298 const float y = c.y * invw; // [-1,1]
299
300 // Screen（D3D規約：そのまま 0..W/H へ）
301 const float sx = viewportX + (x * 0.5f + 0.5f) * viewportWidth;
302 const float sy = viewportY + (y * 0.5f + 0.5f) * viewportHeight;
303
304 scrMinX = (std::min)(scrMinX, sx);
305 scrMinY = (std::min)(scrMinY, sy);
306 scrMaxX = (std::max)(scrMaxX, sx);
307 scrMaxY = (std::max)(scrMaxY, sy);
308 }
309
310 // ---- Rectangle（スクリーン）用。ビューポートにクランプ＋margin ----
311 float rminx = (std::max)(viewportX - clampMargin, scrMinX);
312 float rminy = (std::max)(viewportY - clampMargin, scrMinY);
313 float rmaxx = (std::min)(viewportX + viewportWidth + clampMargin, scrMaxX);
314 float rmaxy = (std::min)(viewportY + viewportHeight + clampMargin, scrMaxY);
315
316 if (rminx < rmaxx && rminy < rmaxy) {
317 out.x0 = rminx; out.y0 = rminy;
318 out.x1 = rmaxx; out.y1 = rmaxy;
319 out.valid = true;
320 }
321 else {
322 out = Rectangle{}; // visible=false
323 }
324
325 return out;
326 }

354 // 8点を同次クリップ空間へ: c = VPW * [x y z 1]^T（列ベクトル前提）
355 static inline void Transform8_Clip_AVX2(const Matrix4x4f& M,
356 const float xs[8], const float ys[8], const float zs[8],
357 __m256& cx, __m256& cy, __m256& cz, __m256& cw) noexcept
358 {
359 const __m256 X = _mm256_loadu_ps(xs);
360 const __m256 Y = _mm256_loadu_ps(ys);
361 const __m256 Z = _mm256_loadu_ps(zs);
362 const __m256 ONE = _mm256_set1_ps(1.0f);
363
364 // row-major 行列 × 列ベクトル = 「各行と (x,y,z,1) のドット積」
365 // cx = m00*X + m01*Y + m02*Z + m03
366 cx = _mm256_set1_ps(M[0][3]);
367 cx = _mm256_fmadd_ps(_mm256_set1_ps(M[0][0]), X, cx);
368 cx = _mm256_fmadd_ps(_mm256_set1_ps(M[0][1]), Y, cx);
369 cx = _mm256_fmadd_ps(_mm256_set1_ps(M[0][2]), Z, cx);
370
371 cy = _mm256_set1_ps(M[1][3]);
372 cy = _mm256_fmadd_ps(_mm256_set1_ps(M[1][0]), X, cy);
373 cy = _mm256_fmadd_ps(_mm256_set1_ps(M[1][1]), Y, cy);
374 cy = _mm256_fmadd_ps(_mm256_set1_ps(M[1][2]), Z, cy);
375
376 cz = _mm256_set1_ps(M[2][3]);
377 cz = _mm256_fmadd_ps(_mm256_set1_ps(M[2][0]), X, cz);
378 cz = _mm256_fmadd_ps(_mm256_set1_ps(M[2][1]), Y, cz);
379 cz = _mm256_fmadd_ps(_mm256_set1_ps(M[2][2]), Z, cz);
380
381 cw = _mm256_set1_ps(M[3][3]);
382 cw = _mm256_fmadd_ps(_mm256_set1_ps(M[3][0]), X, cw);
383 cw = _mm256_fmadd_ps(_mm256_set1_ps(M[3][1]), Y, cw);
384 cw = _mm256_fmadd_ps(_mm256_set1_ps(M[3][2]), Z, cw);
385 // + 1.0f * m[3][3] は上の初期値ですでに加算済み
386 (void)ONE;
387 }

389 // 8点の outcode を一気に作る（LH×ZeroToOne の6平面＋ w<=0）
390 // 各平面の 8bit マスクを返す（bit i は頂点 i の outside）
391 static inline uint8_t OutsideMaskForPlane(__m256 x, __m256 y, __m256 z, __m256 w, int plane) noexcept {
392 __m256 cmp;
393 switch (plane) {
394 case 0: cmp = _mm256_cmp_ps(x, _mm256_sub_ps(_mm256_setzero_ps(), w), _CMP_LT_OS); break; // x < -w
395 case 1: cmp = _mm256_cmp_ps(x, w, _CMP_GT_OS); break; // x > w
396 case 2: cmp = _mm256_cmp_ps(y, _mm256_sub_ps(_mm256_setzero_ps(), w), _CMP_LT_OS); break; // y < -w
397 case 3: cmp = _mm256_cmp_ps(y, w, _CMP_GT_OS); break; // y > w
398 case 4: cmp = _mm256_cmp_ps(z, _mm256_setzero_ps(), _CMP_LT_OS); break; // z < 0
399 case 5: cmp = _mm256_cmp_ps(z, w, _CMP_GT_OS); break; // z > w
400 default: cmp = _mm256_cmp_ps(w, _mm256_setzero_ps(), _CMP_LE_OQ); break; // w <= 0
401 }
402 return static_cast<uint8_t>(_mm256_movemask_ps(cmp)); // 8bit
403 }

427 // 列ベクトル規約 / row-major 格納。VPW = (proj * view * world)
428 inline NdcRectWithW ProjectAABBToNdc_Robust(const Matrix4x4f& VPW, const AABB3f& box) {
429 // 1) AABB 8頂点を clip 空間へ
430 const Vec3f& L = box.lb;
431 const Vec3f& U = box.ub;
432 std::array<Vec3f, 8> corners = {
433 Vec3f{L.x,L.y,L.z}, Vec3f{U.x,L.y,L.z}, Vec3f{L.x,U.y,L.z}, Vec3f{U.x,U.y,L.z},
434 Vec3f{L.x,L.y,U.z}, Vec3f{U.x,L.y,U.z}, Vec3f{L.x,U.y,U.z}, Vec3f{U.x,U.y,U.z}
435 };
436
437 Vec4f clip[8];
438 for (int i = 0; i < 8; ++i) {
439 Vec3f p = corners[i]; // (lb/ub の組合せ)
440 Vec4f v = { p.x, p.y, p.z, 1.0f };
441 clip[i] = VPW * v; // 列ベクトルで右掛け
442 }

444 // 2) 12エッジを列挙
445 static const int E[12][2] = {
446 {0,1},{2,3},{4,5},{6,7}, {0,2},{1,3},{4,6},{5,7}, {0,4},{1,5},{2,6},{3,7}
447 };
448
449 // 3) クリップ多角形の頂点バッファ（最大: 8 + 12*6 の交点だが通常は少量）
450 std::vector<Vec4f> poly; poly.reserve(32);
451
452 // 3a) 最初は「各頂点がクリップ内なら採用」
453 auto inside = [](const Vec4f& c) -> bool {
454 const float x = c.x, y = c.y, z = c.z, w = c.w;
455 if (w <= 0.0f) return false;
456 if (x < -w || x > w) return false;
457 if (y < -w || y > w) return false;
458 if (z < 0.0f || z > w) return false; // LH ZeroToOne
459 return true;
460 };
461 for (int i = 0; i < 8; ++i) if (inside(clip[i])) poly.push_back(clip[i]);
462
463 // 3b) 各エッジについて、各クリップ平面で「外→内」の交点を追加
464 auto addIntersect = [&](const Vec4f& a, const Vec4f& b, int plane) {
465 // plane: 0:x=-w,1:x=+w,2:y=-w,3:y=+w,4:z=0,5:z=w
466 // f(v) >= 0 が内側になる符号関数を作る
467 auto f = [&](const Vec4f& v)->float {
468 switch (plane) {
469 case 0: return v.x + v.w; // x >= -w
470 case 1: return -v.x + v.w; // x <= w
471 case 2: return v.y + v.w; // y >= -w
472 case 3: return -v.y + v.w; // y <= w
473 case 4: return v.z; // z >= 0
474 default:return v.w - v.z; // z <= w
475 }
476 };
477 float fa = f(a), fb = f(b);
478 if ((fa < 0 && fb>0) || (fa > 0 && fb < 0)) {
479 float t = fa / (fa - fb); // 線形補間で交点
480 Vec4f p = a + (b - a) * t; // 同次で補間
481 // w<=0 の数値不安定を避けるガード
482 if (p.w > 1e-6f) poly.push_back(p);
483 }
484 };
485
486 for (auto& e : E) {
487 const Vec4f& a = clip[e[0]];
488 const Vec4f& b = clip[e[1]];
489 // 6平面すべてに対して交差チェック
490 for (int plane = 0; plane < 6; ++plane) addIntersect(a, b, plane);
491 }
492
493 // 万一すべて外：無効
494 if (poly.empty()) return { 0,0,0,0, +INFINITY, false };
495
496 // 4) NDCへ割り算 → min/max
497 float xmin = +1e9f, ymin = +1e9f, xmax = -1e9f, ymax = -1e9f;
498 float wmin = +1e9f;
499 for (const Vec4f& c : poly) {
500 float invw = 1.0f / c.w;
501 float x = c.x * invw;
502 float y = c.y * invw;
503 xmin = (std::min)(xmin, x);
504 ymin = (std::min)(ymin, y);
505 xmax = (std::max)(xmax, x);
506 ymax = (std::max)(ymax, y);
507 wmin = (std::min)(wmin, c.w); // MOC 用
508 }
509
510 // 5) 画面端での安定化（[-1,1]にクランプし、反転に対処）
511 xmin = (std::max)(-1.0f, (std::min)(1.0f, xmin));
512 xmax = (std::max)(-1.0f, (std::min)(1.0f, xmax));
513 ymin = (std::max)(-1.0f, (std::min)(1.0f, ymin));
514 ymax = (std::max)(-1.0f, (std::min)(1.0f, ymax));
515
516 if (xmax < xmin) std::swap(xmin, xmax);
517 if (ymax < ymin) std::swap(ymin, ymax);
518
519 return { xmin, ymin, xmax, ymax, wmin, true };
520 }

522 // 早期破棄つき・高速 NDC 投影。部分交差は robustClip() に委譲。
523 inline NdcRectWithW ProjectAABBToNdc_Fast(const Matrix4x4f& VPW, const AABB3f& box,
524 float min_coverage_for_precise = 0.0f) // 例: 0.001f
525 {
526 // (1) 8頂点→clip
527 float xs[8], ys[8], zs[8];
528 MakeAabbCorners8(box, xs, ys, zs);
529 __m256 cx, cy, cz, cw;
530 Transform8_Clip_AVX2(VPW, xs, ys, zs, cx, cy, cz, cw);
531
532 // (2) Outcode 集計
533 uint8_t all_out_any_plane = 0;
534 uint8_t any_out_any_plane = 0;
535 for (int p = 0; p < 6; ++p) {
536 uint8_t m = OutsideMaskForPlane(cx, cy, cz, cw, p);
537 all_out_any_plane |= (uint8_t)(m == 0xFF); // 全頂点が同じ平面の外
538 any_out_any_plane |= (uint8_t)(m != 0x00); // 1点でも外
539 }
540 // w<=0
541 uint8_t wneg = OutsideMaskForPlane(cx, cy, cz, cw, 6);
542 if (wneg == 0xFF) return { 0,0,0,0, +INFINITY, false }; // 全点 w<=0
543
544 if (all_out_any_plane) {
545 // 完全に外（早期破棄）
546 return { 0,0,0,0, +INFINITY, false };
547 }
548
549 if (!any_out_any_plane && wneg == 0x00) {
550 // (3) 完全内包: SIMD で透視除算 & min/max
551 // invw ≈ rcp + 1回ニュートン (精度安定)
552 __m256 invw = _mm256_rcp_ps(cw);
553 invw = _mm256_mul_ps(_mm256_fnmadd_ps(cw, invw, _mm256_set1_ps(2.0f)), invw); // invw *= (2 - w*invw)
554
555 __m256 ndcX = _mm256_mul_ps(cx, invw);
556 __m256 ndcY = _mm256_mul_ps(cy, invw);
557
558 float xmin = hmin8(ndcX), xmax = hmax8(ndcX);
559 float ymin = hmin8(ndcY), ymax = hmax8(ndcY);
560 float wmin = hmin8(cw);
561
562 // クランプ＆整形
563 xmin = (std::max)(-1.0f, (std::min)(1.0f, xmin));
564 xmax = (std::max)(-1.0f, (std::min)(1.0f, xmax));
565 ymin = (std::max)(-1.0f, (std::min)(1.0f, ymin));
566 ymax = (std::max)(-1.0f, (std::min)(1.0f, ymax));
567 if (xmax < xmin) std::swap(xmin, xmax);
568 if (ymax < ymin) std::swap(ymin, ymax);
569
570 // 早期破棄（小物カットをしたい場合）
571 if (min_coverage_for_precise > 0.0f) {
572 float covX = (std::max)(0.f, (std::min)(1.f, (xmax - xmin) * 0.5f));
573 float covY = (std::max)(0.f, (std::min)(1.f, (ymax - ymin) * 0.5f));
574 if (covX * covY < min_coverage_for_precise) {
575 return { xmin, ymin, xmax, ymax, wmin, false }; // 可視性は低いのでスキップ目安
576 }
577 }
578 return { xmin, ymin, xmax, ymax, wmin, true };
579 }
580
581 // (4) 部分交差: 重いクリップへ（near/x=±w/y=±w/z=w）
582 // 小物しきい値でスキップしたい場合は事前に球近似で cheap 判定を入れてもOK
583 return ProjectAABBToNdc_Robust(VPW, box);
584 }

586 static inline __m256 rcp_nr1(__m256 x) noexcept {
587 __m256 r = _mm256_rcp_ps(x);
588 // r = r * (2 - x*r)
589 return _mm256_mul_ps(_mm256_fnmadd_ps(x, r, _mm256_set1_ps(2.0f)), r);
590 }

592 // 安全除算（配列化用）
593 static inline void safe_div8(const float* num, const float* den, float* out) {
594 constexpr float eps = 1e-6f;
595 for (int i = 0; i < 8; ++i) {
596 float d = std::fabs(den[i]) < eps ? (den[i] < 0 ? -eps : eps) : den[i];
597 out[i] = num[i] / d;
598 }
599 }

601 // WVP (= proj*view*world) で 8球を一括投影して可視判定/矩形を返す
602 // - BoundingSphere は { Vec3 center; float radius; } を想定
603 // - 列ベクトル規約：clip = WVP * [x,y,z,1]^T
604 template<class Mat4, class BoundingSphereT>
605 inline void ProjectSpheresToNdc_Fast_AVX2(
606 const Mat4& WVP,
607 const BoundingSphereT* spheres, // 長さ>=count
608 size_t count,
609 NdcRectWithW* outRects // 長さ>=count
610 ) {
611 const size_t step = 8;
612 for (size_t base = 0; base < count; base += step) {
613 const size_t n = (std::min)(step, count - base);
614
615 // SoA ロード
616 float cx8[8]{}, cy8[8]{}, cz8[8]{}, cw8[8]{};
617 float px8[8]{}, py8[8]{}, pz8[8]{};
618 float pxw8[8]{}, pyw8[8]{}, pzw8[8]{};
619 float ndc_cx8[8]{}, ndc_cy8[8]{}, ndc_cz8[8]{};
620 float ndc_pxx8[8]{}, ndc_pyy8[8]{}, ndc_pzz8[8]{};
621
622 float X[8]{}, Y[8]{}, Z[8]{}, R[8]{};
623 for (size_t i = 0; i < n; ++i) {
624 const auto& s = spheres[base + i];
625 X[i] = s.center.x;
626 Y[i] = s.center.y;
627 Z[i] = s.center.z;
628 R[i] = s.radius;
629 }
630
631 // AVX2 ブロック
632 __m256 VX = _mm256_loadu_ps(X);
633 __m256 VY = _mm256_loadu_ps(Y);
634 __m256 VZ = _mm256_loadu_ps(Z);
635 __m256 VR = _mm256_loadu_ps(R);
636 __m256 ONE = _mm256_set1_ps(1.0f);
637
638 // row-major × 列ベクトル：各行と (x,y,z,1) のドット積
639 auto dot_row = [&](int row, __m256 x, __m256 y, __m256 z, __m256 one) {
640 __m256 sum = _mm256_set1_ps(WVP[row][3]); // m[row][3] * 1
641 sum = _mm256_fmadd_ps(_mm256_set1_ps(WVP[row][0]), x, sum);
642 sum = _mm256_fmadd_ps(_mm256_set1_ps(WVP[row][1]), y, sum);
643 sum = _mm256_fmadd_ps(_mm256_set1_ps(WVP[row][2]), z, sum);
644 (void)one;
645 return sum;
646 };
647
648 // 中心の clip 座標
649 __m256 Cx = dot_row(0, VX, VY, VZ, ONE);
650 __m256 Cy = dot_row(1, VX, VY, VZ, ONE);
651 __m256 Cz = dot_row(2, VX, VY, VZ, ONE);
652 __m256 Cw = dot_row(3, VX, VY, VZ, ONE);
653
654 // +X（局所）: (x+R,y,z,1) = 中心に m[*,0]*R を足すだけ
655 __m256 Px = _mm256_fmadd_ps(_mm256_set1_ps(WVP[0][0]), VR, Cx);
656 __m256 Py = _mm256_fmadd_ps(_mm256_set1_ps(WVP[1][0]), VR, Cy);
657 __m256 Pz = _mm256_fmadd_ps(_mm256_set1_ps(WVP[2][0]), VR, Cz);
658 __m256 Pw = _mm256_fmadd_ps(_mm256_set1_ps(WVP[3][0]), VR, Cw);
659
660 // +Y（局所）
661 __m256 Qx = _mm256_fmadd_ps(_mm256_set1_ps(WVP[0][1]), VR, Cx);
662 __m256 Qy = _mm256_fmadd_ps(_mm256_set1_ps(WVP[1][1]), VR, Cy);
663 __m256 Qz = _mm256_fmadd_ps(_mm256_set1_ps(WVP[2][1]), VR, Cz);
664 __m256 Qw = _mm256_fmadd_ps(_mm256_set1_ps(WVP[3][1]), VR, Cw);
665
666 // +Z（局所）… z 可視範囲推定用
667 __m256 Rx = _mm256_fmadd_ps(_mm256_set1_ps(WVP[0][2]), VR, Cx);
668 __m256 Ry = _mm256_fmadd_ps(_mm256_set1_ps(WVP[1][2]), VR, Cy);
669 __m256 Rz = _mm256_fmadd_ps(_mm256_set1_ps(WVP[2][2]), VR, Cz);
670 __m256 Rw = _mm256_fmadd_ps(_mm256_set1_ps(WVP[3][2]), VR, Cw);
671
672 // 透視除算（1回ニュートンで安定化）
673 __m256 invCw = rcp_nr1(Cw);
674 __m256 invPw = rcp_nr1(Pw);
675 __m256 invQw = rcp_nr1(Qw);
676 __m256 invRw = rcp_nr1(Rw);
677
678 __m256 NdC_x = _mm256_mul_ps(Cx, invCw);
679 __m256 NdC_y = _mm256_mul_ps(Cy, invCw);
680 __m256 NdC_z = _mm256_mul_ps(Cz, invCw);
681
682 __m256 NdP_x = _mm256_mul_ps(Px, invPw); // +X の x
683 __m256 NdQ_y = _mm256_mul_ps(Qy, invQw); // +Y の y
684 __m256 NdR_z = _mm256_mul_ps(Rz, invRw); // +Z の z
685
686 // 画面半径 r_ndc = max(|px.x - c.x|, |py.y - c.y|)
687 __m256 dx = _mm256_andnot_ps(_mm256_set1_ps(-0.f), _mm256_sub_ps(NdP_x, NdC_x)); // fabs
688 __m256 dy = _mm256_andnot_ps(_mm256_set1_ps(-0.f), _mm256_sub_ps(NdQ_y, NdC_y));
689 __m256 r = _mm256_max_ps(dx, dy);
690
691 // NDC 矩形
692 __m256 xmin = _mm256_sub_ps(NdC_x, r);
693 __m256 xmax = _mm256_add_ps(NdC_x, r);
694 __m256 ymin = _mm256_sub_ps(NdC_y, r);
695 __m256 ymax = _mm256_add_ps(NdC_y, r);
696
697 // [-1,1] にクランプ
698 __m256 N1 = _mm256_set1_ps(-1.0f), P1 = _mm256_set1_ps(1.0f);
699 xmin = _mm256_min_ps(P1, _mm256_max_ps(N1, xmin));
700 xmax = _mm256_min_ps(P1, _mm256_max_ps(N1, xmax));
701 ymin = _mm256_min_ps(P1, _mm256_max_ps(N1, ymin));
702 ymax = _mm256_min_ps(P1, _mm256_max_ps(N1, ymax));
703
704 // 配列へストア
705 _mm256_storeu_ps(cx8, Cx); _mm256_storeu_ps(cy8, Cy);
706 _mm256_storeu_ps(cz8, Cz); _mm256_storeu_ps(cw8, Cw);
707 _mm256_storeu_ps(px8, xmin); _mm256_storeu_ps(py8, ymin);
708 _mm256_storeu_ps(pz8, xmax);
709 _mm256_storeu_ps(pxw8, ymax);
710
711 _mm256_storeu_ps(ndc_cx8, NdC_x);
712 _mm256_storeu_ps(ndc_cy8, NdC_y);
713 _mm256_storeu_ps(ndc_cz8, NdC_z);
714 _mm256_storeu_ps(ndc_pxx8, NdP_x);
715 _mm256_storeu_ps(ndc_pyy8, NdQ_y);
716 _mm256_storeu_ps(ndc_pzz8, NdR_z);
717
718 _mm256_storeu_ps(pyw8, Pw);
719 _mm256_storeu_ps(pzw8, Qw);
720 _mm256_storeu_ps(pxw8, Pw); // 既に Pw を格納済だが wmin 計算で再利用
721 _mm256_storeu_ps(pz8, Rw);
722
723 // 各レーンをまとめて可視判定・出力
724 for (size_t i = 0; i < n; ++i) {
725 const float ndc_cx = ndc_cx8[i];
726 const float ndc_cy = ndc_cy8[i];
727 const float ndc_cz = ndc_cz8[i];
728
729 const float ndc_pz = ndc_pzz8[i];
730
731 // 直前で xmin/xmax/ymin/ymax は [-1,1] にクランプ済み
732 float xmin_s = px8[i];
733 float ymin_s = py8[i];
734 float xmax_s = pz8[i];
735 float ymax_s = pxw8[i];
736
737 if (xmax_s < xmin_s) std::swap(xmin_s, xmax_s);
738 if (ymax_s < ymin_s) std::swap(ymin_s, ymax_s);
739
740 // z の簡易交差（中心と +Z で保守的）
741 float zmin_est = (std::min)(ndc_cz, ndc_pz);
742 float zmax_est = (std::max)(ndc_cz, ndc_pz);
743
744 const bool x_overlap = !(xmax_s < -1.0f || xmin_s > 1.0f);
745 const bool y_overlap = !(ymax_s < -1.0f || ymin_s > 1.0f);
746 const bool z_overlap = !(zmax_est < 0.0f || zmin_est > 1.0f);
747
748 // wmin（MOC 等の使用を想定）
749 float wmin = (std::min)({ cw8[i], /* Pw */ pxw8[i], /* Qw */ pzw8[i], /* Rw */ pz8[i] });
750
751 outRects[base + i] = {
752 xmin_s, ymin_s, xmax_s, ymax_s,
753 wmin,
754 (x_overlap && y_overlap && z_overlap)
755 };
756 }
757 }
758 }
