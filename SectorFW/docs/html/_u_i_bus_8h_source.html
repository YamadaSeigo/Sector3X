44 class UiQueue {
45 public:
46 void push(T x) { std::lock_guard<std::mutex> lk(m_); q_.push_back(std::move(x)); }
47 std::vector<T> drain() {
48 std::lock_guard<std::mutex> lk(m_);
49 std::vector<T> out; out.reserve(q_.size());
50 while (!q_.empty()) { out.push_back(std::move(q_.front())); q_.pop_front(); }
51 return out;
52 }
53 private:
54 std::mutex m_;
