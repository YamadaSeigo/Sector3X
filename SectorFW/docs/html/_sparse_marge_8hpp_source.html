26 struct DenseRunner {
27 template<class... Dense, class Fn>
28 requires (sizeof...(Dense) > 0)
29 static size_t Run(ArchetypeManager& am, Fn&& fn) {
30 Query q;
31 q.With<Dense...>(); // Dense 限定クエリ（AllDense 制約） :contentReference[oaicite:4]{index=4}
32 auto chunks = q.MatchingChunks(am); // マスク一致チャンク列挙 :contentReference[oaicite:5]{index=5}
33
34 size_t total = 0;
35 for (auto* chunk : chunks) {
36 // 列ポインタを先に束ねる（GetColumn<T>() は列先頭の生ポインタを返す）
37 // 以後は &col[i] で連続メモリアクセス。 :contentReference[oaicite:6]{index=6}
38 auto cols = std::tuple{ chunk->GetColumn<Dense>().value()... };
39
40 const auto& ids = chunk->GetEntityIDs(); // 昇順 ID 配列を想定 :contentReference[oaicite:7]{index=7}
41 const size_t n = chunk->GetEntityCount();
42
43 for (size_t i = 0; i < n; ++i) {
44 std::apply([&](auto*... col) {
45 fn(ids[i], col[i]...);
46 }, cols);
47 }
48 total += n;
49 }
50 return total;
51 }
52 };

55 // 2) Sparse マージ補助
56 // - unordered_map<EntityID, S> など “疎” を Dense 走査にマージ
57 // - フレーム頭などでキーを一度だけ抽出→昇順ソート
58 // - 各チャンクの entities(昇順) と二本指マージで命中だけ適用

60 struct SparseMerge {
61 // ---- (A) キー抽出 & ソート（フレームに1回推奨） ----
62 template<class SparseMap>
63 static void BuildSortedKeys(const SparseMap& m, std::vector<EntityID>& outSortedKeys) {
64 outSortedKeys.clear();
65 outSortedKeys.reserve(m.size());
66 for (auto& kv : m) outSortedKeys.push_back(kv.first);
67 std::sort(outSortedKeys.begin(), outSortedKeys.end());
68 }
69
70 // ---- (B) 二本指マージで命中だけ fn を呼ぶ ----
71 // fn の形: fn(rowIndexInChunk, const S& sparseValue)
72 template<class SparseMap, class Fn>
73 static size_t MergeJoinApply(const std::vector<EntityID>& chunkIDs,
74 const SparseMap& sparseMap,
75 const std::vector<EntityID>& sortedKeys,
76 Fn&& fn)
77 {
78 size_t i = 0, j = 0, hits = 0;
79 const size_t n = chunkIDs.size(), m = sortedKeys.size();
80
81 while (i < n && j < m) {
82 EntityID a = chunkIDs[i];
83 EntityID b = sortedKeys[j];
84 if (a < b) { ++i; }
85 else if (a > b) { ++j; }
86 else {
87 // 命中: find は命中時のみ実行（回数最小化）
88 if (auto it = sparseMap.find(a); it != sparseMap.end()) {
89 fn(i, it->second);
90 ++hits;
91 }
92 ++i; ++j;
93 }
94 }
95 return hits;
96 }
97
98 // ---- (C) Dense と併用するユーティリティ ----
99 // DenseRunner と同じく Dense 列を握り、命中行にだけ applyDense(row, sparseVal, denseRefs...)
100 template<class S, class... Dense, class Apply>
101 static size_t RunDenseWithSparse(ArchetypeManager& am,
102 const std::unordered_map<EntityID, S>& sparseMap,
103 const std::vector<EntityID>& sortedKeys,
104 Apply&& applyDense)
105 {
106 Query q; q.With<Dense...>(); // Dense 限定でチャンク列挙 :contentReference[oaicite:8]{index=8}
107 auto chunks = q.MatchingChunks(am); // :contentReference[oaicite:9]{index=9}
108 size_t totalHits = 0;
109
110 for (auto* chunk : chunks) {
111 auto cols = std::tuple{ chunk->GetColumn<Dense>().value()... }; // 列ポインタ取得 :contentReference[oaicite:10]{index=10}
112 const auto& ids = chunk->GetEntityIDs(); // :contentReference[oaicite:11]{index=11}
113
114 totalHits += MergeJoinApply(ids, sparseMap, sortedKeys,
115 [&](size_t row, const S& sval) {
116 std::apply([&](auto*... col) {
117 applyDense(row, sval, col[row]...);
118 }, cols);
119 });
120 }
121 return totalHits;
122 }
123 };
