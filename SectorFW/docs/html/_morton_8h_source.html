73 * Bit interleave helpers for 3D Morton 64-bit
74 * -------------------------------------------------------------
75 * 3D Morton は x,y,z の各ビットを [x0,y0,z0,x1,y1,z1,...] の順に並べる。
76 * 各軸 21bit まで安全 (21 * 3 = 63)。

78 [[nodiscard]] static inline constexpr std::uint64_t Part1By2_64(std::uint64_t x) noexcept {
79 x &= 0x1fffffULL; // 21 bits
80 x = (x | (x << 32)) & 0x1f00000000ffffULL;
81 x = (x | (x << 16)) & 0x1f0000ff0000ffULL;
82 x = (x | (x << 8)) & 0x100f00f00f00f00fULL;
83 x = (x | (x << 4)) & 0x10c30c30c30c30c3ULL;
84 x = (x | (x << 2)) & 0x1249249249249249ULL; // 0b001001.. パターン
85 return x;
