155 using namespace ECS;
156
157 ComponentMask mask;
158 (SetMask<Components>(mask), ...);
159
160 // Transformコンポーネントが存在するかどうかをチェック
161 ComponentTypeID typeID = ComponentTypeRegistry::GetID<TransformType>();
162 bool hasTransform = mask.test(typeID);
163
164 EntityID id = EntityID::Invalid();

166 //Transformコンポーネントが存在する場合、パーティションからチャンクを取得

174 {
175 id = (*chunk)->GetEntityManager().AddEntity<Components...>(mask, components...);
176 }
177 }
178 }
179 // Transformコンポーネントが存在しない場合、グローバルエンティティマネージャーを使用
180 else
181 {
182 id = level.partition.GetGlobalEntityManager().AddEntity<Components...>(mask, components...);
183 }
184

474 void UpdateState(ELevelState setBits, ELevelState clearBits,
475 std::memory_order success = std::memory_order_acq_rel,
476 std::memory_order fail = std::memory_order_acquire)
477 {
478 uint32_t expected = state.load(std::memory_order_relaxed);
479 while (true) {
480 uint32_t desired = (expected | static_cast<uint32_t>(setBits)) & ~static_cast<uint32_t>(clearBits);
481 if (state.compare_exchange_weak(expected, desired, success, fail)) {
482 return;
483 }
484 // 失敗時 expected が最新値に更新されるのでループ継続
485 }
486 }

531 bool TryBeginLoading(
532 std::memory_order success = std::memory_order_acq_rel,
533 std::memory_order fail = std::memory_order_acquire
534 ) {
535 uint32_t expected = state.load(std::memory_order_relaxed);
536
537 for (;;) {
538 // すでに Loaded / Loading なら開始できない
539 if (expected & static_cast<uint32_t>(ELevelState::Loaded)) return false;
540 if (expected & static_cast<uint32_t>(ELevelState::Loading)) return false;
541
542 const uint32_t desired = expected | static_cast<uint32_t>(ELevelState::Loading);
543
544 if (state.compare_exchange_weak(expected, desired, success, fail)) {
545 return true; // Loading を立てることに成功
546 }
547 // compare_exchange_weak が失敗すると expected が更新されるので、そのままループ継続
548 }
549 }

555 bool TryBeginClean(

557 std::memory_order fail = std::memory_order_acquire
558 ) {

560
561 for (;;) {
