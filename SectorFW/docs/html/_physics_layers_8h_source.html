77 public:
78 virtual bool ShouldCollide(JPH::ObjectLayer layer1, JPH::BroadPhaseLayer layer2) const override {
79 if (layer1 == Layers::SENSOR) return layer2 == Layers::BPLayers::MOVING; // sensorは動く物体のみ検出
80 // 非移動は移動と、移動は非移動＆移動と
