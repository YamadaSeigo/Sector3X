88 // ------------- 2球の最小包含球（正確） -------------
89 static BoundingSphere Merge(const BoundingSphere& a, const BoundingSphere& b) noexcept {
90 // 片方がもう片方を包含する？
91 Vec3 d = v3_sub(b.center, a.center);
92 T dist = v3_len(d);
93 if (a.radius >= b.radius + dist) return a;
94 if (b.radius >= a.radius + dist) return b;
95 if (dist <= std::numeric_limits<T>::epsilon()) {
96 // 同心円に近い
97 BoundingSphere s;
98 s.center = a.center;
99 s.radius = (std::max)(a.radius, b.radius);
100 return s;
101 }
102 T newR = (dist + a.radius + b.radius) * T(0.5);
103 Vec3 dir = v3_muls(d, T(1) / dist);
104 Vec3 newC = v3_add(a.center, v3_muls(dir, newR - a.radius));
105 return { newC, newR };
106 }

108 // ------------- 逐次拡張（オンライン更新向け、最小ではない） -------------
109 constexpr void expandToFit(const Vec3& p) noexcept {
110 Vec3 diff = v3_sub(p, center);
111 T d2 = v3_len2(diff);
112 if (d2 <= radius * radius) return; // 既に包含
113 T d = std::sqrt(d2);
114 T newR = (radius + d) * T(0.5);
115 // センターを外側へスライド
116 if (d > T(0)) {
117 center = v3_add(center, v3_muls(diff, (newR - radius) / d));
118 }
119 radius = newR;
120 }

122 constexpr void expandToFit(const BoundingSphere& s) noexcept {
123 *this = Merge(*this, s);

126 // ------------- Ritter 法（高速・近似、ほぼ十分に小さい） -------------
127 // points: AoS 配列。Nが小さくてもOK。O(N)。
128 static BoundingSphere FromPointsRitter(const Vec3* points, size_t count) {
129 BoundingSphere s{};
130 if (count == 0) return s;
131 if (count == 1) { s.center = points[0]; s.radius = T(0); return s; }
132
133 // 1) まず適当な点 p0 を選び、そこから最も遠い点 p1、さらに p1 から最も遠い点 p2 を探す
134 size_t i0 = 0;
135 auto farthest_from = [&](size_t idx)->size_t {
136 size_t best = 0;
137 auto c = points[idx];
138 auto bestD2 = v3_len2(v3_sub(points[0], c));
139 for (size_t i = 1; i < count; ++i) {
140 auto d2 = v3_len2(v3_sub(points[i], c));
141 if (d2 > bestD2) { bestD2 = d2; best = i; }
142 }
143 return best;
144 };
145 size_t i1 = farthest_from(i0);
146 size_t i2 = farthest_from(i1);
147
148 // 2) p1 と p2 を直径とする球から開始
149 Vec3 p1 = points[i1], p2 = points[i2];
150 s.center = v3_mid(p1, p2);
151 s.radius = T(0.5) * v3_len(v3_sub(p2, p1));
152
153 // 3) すべての点を走査し、外側の点があれば最小拡張
154 for (size_t i = 0; i < count; ++i) {
155 s.expandToFit(points[i]);
156 }
157 return s;
158 }

176 auto ball_from_1 = [&](const Vec3& a) {

183 auto ball_from_3 = [&](const Vec3& a, const Vec3& b, const Vec3& c) {
184 // 三点を通る円（3Dでは3点共面の外接円）→ その円の中心と半径（平面内）
185 // 3点がほぼ一直線なら2点からの球へフォールバック
186 Vec3 ab = v3_sub(b, a), ac = v3_sub(c, a);
187 auto abXac_x = ab.y * ac.z - ab.z * ac.y;
188 auto abXac_y = ab.z * ac.x - ab.x * ac.z;
189 auto abXac_z = ab.x * ac.y - ab.y * ac.x;
190 T denom = T(2) * (abXac_x * abXac_x + abXac_y * abXac_y + abXac_z * abXac_z);
191 const T eps = T(1e-12);
192 if (std::abs(denom) < eps) {
193 // 退化：2点の最大直径
194 BoundingSphere s2; s2.radius = T(-1);
195 auto try2 = [&](const Vec3& p, const Vec3& q) {
196 BoundingSphere t; ball_from_2(p, q);
197 if (t.radius > s2.radius) s2 = t;
198 };
199 try2(a, b); try2(a, c); try2(b, c);
200 s = s2;
201 return;
202 }
203 auto ab2 = v3_len2(ab);
204 auto ac2 = v3_len2(ac);
205 Vec3 num = {
206 (ab2 * ac.y - ac2 * ab.y) * abXac_z - (ab2 * ac.z - ac2 * ab.z) * abXac_y,
207 (ab2 * ac.z - ac2 * ab.z) * abXac_x - (ab2 * ac.x - ac2 * ab.x) * abXac_z,
208 (ab2 * ac.x - ac2 * ab.x) * abXac_y - (ab2 * ac.y - ac2 * ab.y) * abXac_x
209 };
210 s.center = v3_add(a, v3_muls(num, T(1) / denom));
211 s.radius = v3_len(v3_sub(s.center, a));
212 };
213 auto ball_from_4 = [&](const Vec3& p, const Vec3& q, const Vec3& r, const Vec3& t) {
214 // 4点の外接球。行列式ベースで求める（数値的にはやや繊細）。
215 // 退化（ほぼ共面/共直線）の場合は3点/2点にフォールバック。
216 auto det4 = [](T a11, T a12, T a13, T a14,
217 T a21, T a22, T a23, T a24,
218 T a31, T a32, T a33, T a34,
219 T a41, T a42, T a43, T a44)->T {
220 // ラプラス展開の手書き版（小規模なのでOK）
221 T m11 = a22 * (a33 * a44 - a34 * a43) - a23 * (a32 * a44 - a34 * a42) + a24 * (a32 * a43 - a33 * a42);
222 T m12 = a21 * (a33 * a44 - a34 * a43) - a23 * (a31 * a44 - a34 * a41) + a24 * (a31 * a43 - a33 * a41);
223 T m13 = a21 * (a32 * a44 - a34 * a42) - a22 * (a31 * a44 - a34 * a41) + a24 * (a31 * a42 - a32 * a41);
224 T m14 = a21 * (a32 * a43 - a33 * a42) - a22 * (a31 * a43 - a33 * a41) + a23 * (a31 * a42 - a32 * a41);
225 return a11 * m11 - a12 * m12 + a13 * m13 - a14 * m14;
226 };
227 auto S = [&](const Vec3& v) { return v.x * v.x + v.y * v.y + v.z * v.z; };
228 T a = det4(p.x, p.y, p.z, T(1),
229 q.x, q.y, q.z, T(1),
230 r.x, r.y, r.z, T(1),
231 t.x, t.y, t.z, T(1));
232 const T eps = T(1e-12);
233 if (std::abs(a) < eps) {
234 // 退化：3点/2点からの最大にフォールバック
235 BoundingSphere best; best.radius = T(-1);
236 auto try3 = [&](const Vec3& A, const Vec3& B, const Vec3& C) {
237 BoundingSphere tmp; ball_from_3(A, B, C);
238 if (tmp.radius > best.radius) best = tmp;
239 };
240 auto try2 = [&](const Vec3& A, const Vec3& B) {
241 BoundingSphere tmp; ball_from_2(A, B);
242 if (tmp.radius > best.radius) best = tmp;
243 };
244 try3(p, q, r); try3(p, q, t); try3(p, r, t); try3(q, r, t);
245 try2(p, q); try2(p, r); try2(p, t); try2(q, r); try2(q, t); try2(r, t);
246 s = best;
247 return;
248 }
249 T dx = det4(S(p), p.y, p.z, T(1),
250 S(q), q.y, q.z, T(1),
251 S(r), r.y, r.z, T(1),
252 S(t), t.y, t.z, T(1));
253 T dy = det4(p.x, S(p), p.z, T(1),
254 q.x, S(q), q.z, T(1),
255 r.x, S(r), r.z, T(1),
256 t.x, S(t), t.z, T(1));
257 T dz = det4(p.x, p.y, S(p), T(1),
258 q.x, q.y, S(q), T(1),
259 r.x, r.y, S(r), T(1),
260 t.x, t.y, S(t), T(1));
261 T c = det4(p.x, p.y, p.z, S(p),
262 q.x, q.y, q.z, S(q),
263 r.x, r.y, r.z, S(r),
264 t.x, t.y, t.z, S(t));
265 s.center = Vec3{ dx / (T(2) * a), dy / (T(2) * a), dz / (T(2) * a) };
266 s.radius = std::sqrt((dx * dx + dy * dy + dz * dz) / (T(4) * a * a) - c / a);
267 };

269 auto make_ball_from_R = [&](const std::array<Vec3, 4>& R, size_t rsz) {
270 switch (rsz) {
271 case 1: ball_from_1(R[0]); break;
272 case 2: ball_from_2(R[0], R[1]); break;
273 case 3: ball_from_3(R[0], R[1], R[2]); break;
274 case 4: ball_from_4(R[0], R[1], R[2], R[3]); break;
275 default: s.center = Vec3{ T(0),T(0),T(0) }; s.radius = T(0); break;
276 }
277 };
278
279 // 非再帰版 Welzl：R を拡張しながらループ
280 s.center = pts[0]; s.radius = T(0);
281 for (size_t i = 0; i < n; ++i) {
282 if (s.contains(pts[i])) continue;
283 // pts[i] を必ず境界に載せる最小球を作る
284 R[0] = pts[i]; rsz = 1; make_ball_from_R(R, rsz);
285 for (size_t j = 0; j < i; ++j) {
286 if (s.contains(pts[j])) continue;
287 R[1] = pts[j]; rsz = 2; make_ball_from_R(R, rsz);
288 for (size_t k = 0; k < j; ++k) {
289 if (s.contains(pts[k])) continue;
290 R[2] = pts[k]; rsz = 3; make_ball_from_R(R, rsz);
291 for (size_t m = 0; m < k; ++m) {
292 if (s.contains(pts[m])) continue;
293 R[3] = pts[m]; rsz = 4; make_ball_from_R(R, rsz);
294 }
295 }
296 }
297 }
298 return s;
299 }

301 // ------------- 汎用トランスフォーム -------------
302 // 一様スケール + 並進だけなら正確。非一様スケール/回転を含む場合は
303 // 「線形部の最大伸長率（列ベクトル長の最大）」で半径を保守的に拡大。
304 template<class Mat4, class Vec3Like>
305 static BoundingSphere Transform(const BoundingSphere& s, const Mat4& M,
306 const Vec3Like& col0, const Vec3Like& col1, const Vec3Like& col2,
307 const Vec3Like& translation) {
308 // center' = R*center + t, radius' = radius * max( ||col0||, ||col1||, ||col2|| )
309 Vec3 newC{
310 M(0,0) * s.center.x + M(0,1) * s.center.y + M(0,2) * s.center.z + M(0,3),
311 M(1,0) * s.center.x + M(1,1) * s.center.y + M(1,2) * s.center.z + M(1,3),
312 M(2,0) * s.center.x + M(2,1) * s.center.y + M(2,2) * s.center.z + M(2,3)
313 };
314 auto len0 = std::sqrt(col0.x * col0.x + col0.y * col0.y + col0.z * col0.z);
315 auto len1 = std::sqrt(col1.x * col1.x + col1.y * col1.y + col1.z * col1.z);
316 auto len2 = std::sqrt(col2.x * col2.x + col2.y * col2.y + col2.z * col2.z);
317 T scale = (T)(std::max)({ len0,len1,len2 });
318 return { newC, s.radius * scale };
319 }

448 const T ndc_pyx = safe_div(pyx, pyw);
449 const T ndc_pyy = safe_div(pyy, pyw);
450
451 const T ndc_pzz = safe_div(pzz, pzw); // z方向の奥/手前評価に利用
452
453 // 4) 画面半径（保守的近似）
454 const T r_ndc_x = std::fabs(ndc_pxx - ndc_cx);
455 const T r_ndc_y = std::fabs(ndc_pyy - ndc_cy);
456 const T r_ndc = (std::max)(r_ndc_x, r_ndc_y);
457
458 // 5) NDC 矩形
459 T xmin = ndc_cx - r_ndc;
460 T xmax = ndc_cx + r_ndc;
461 T ymin = ndc_cy - r_ndc;
462 T ymax = ndc_cy + r_ndc;
463
464 // 6) クリップとの交差（x,y は [-1,1]、z は [0,1]）
465 const T zmin_est = (std::min)(ndc_cz, ndc_pzz);
466 const T zmax_est = (std::max)(ndc_cz, ndc_pzz);
467
468 const bool x_overlap = !(xmax < -1.0f || xmin > 1.0f);
469 const bool y_overlap = !(ymax < -1.0f || ymin > 1.0f);
470 const bool z_overlap = !(zmax_est < 0.0f || zmin_est > 1.0f);
471
472 // 出力
473 if (outNDC) {
474 outNDC->xmin = xmin;
475 outNDC->xmax = xmax;
476 outNDC->ymin = ymin;
477 outNDC->ymax = ymax;
478 const T raw_minw = (std::min)((std::min)(cw, pxw), (std::min)(pyw, pzw));
479 const T epsW = 1e-6f; // 数値安定用
480 outNDC->wmin = (raw_minw < epsW) ? epsW : raw_minw;
481 }
482
483 if (depth) *depth = cw; // 手前側の深度の代表として中心の W を返す
484
485 return x_overlap && y_overlap && z_overlap;
486 }

488 // center: 球中心（ワールド）/ radius: 半径
489 // camRight, camUp, camForward: カメラのワールド空間基底（正規直交・単位長を想定）
490 //
491 // 可視なら true。outNDC は NDC の AABB（x,y は [-1,1]、z は [0,1] で交差判定）
492 // depth には「中心の clip.w」を返します（既存コード互換）。
493 template<class Mat4, class NDC>
494 bool IsVisible_WVP_CamBasis(
495 const Mat4& WVP,
496 const Vec3& camRight, const Vec3& camUp, const Vec3& camForward,
497 NDC* outNDC, T* depth = nullptr) const noexcept
498 {
499 using ::SFW::Math::MulPoint_RowMajor_ColVec;
500
501 auto safe_div = [](T a, T b) {
502 const T eps = 1e-6f;
503 return a / ((std::fabs(b) < eps) ? (b < 0 ? -eps : eps) : b);
504 };
505
506 // 1) 中心を clip に
507 T cx, cy, cz, cw;
508 MulPoint_RowMajor_ColVec(WVP, center.x, center.y, center.z, cx, cy, cz, cw);
509
510 // 2) 基底方向に ±radius だけずらした点を投影
511 auto proj_pt = [&](const Vec3& p, T& x, T& y, T& z, T& w) {
512 MulPoint_RowMajor_ColVec(WVP, p.x, p.y, p.z, x, y, z, w);
513 };
514
515 const Vec3 Prp = { center.x + camRight.x * radius, center.y + camRight.y * radius, center.z + camRight.z * radius };
516 const Vec3 Prm = { center.x - camRight.x * radius, center.y - camRight.y * radius, center.z - camRight.z * radius };
517 const Vec3 Pup = { center.x + camUp.x * radius, center.y + camUp.y * radius, center.z + camUp.z * radius };
518 const Vec3 Pum = { center.x - camUp.x * radius, center.y - camUp.y * radius, center.z - camUp.z * radius };
519 const Vec3 Pfp = { center.x + camForward.x * radius,center.y + camForward.y * radius,center.z + camForward.z * radius };
520 const Vec3 Pfm = { center.x - camForward.x * radius,center.y - camForward.y * radius,center.z - camForward.z * radius };
521
522 T rpx, rpy, rpz, rpw; proj_pt(Prp, rpx, rpy, rpz, rpw); // +Right
523 T rm_x, rm_y, rm_z, rm_w; proj_pt(Prm, rm_x, rm_y, rm_z, rm_w); // -Right
524 T upx, upy, upz, upw; proj_pt(Pup, upx, upy, upz, upw); // +Up
525 T umx, umy, umz, umw; proj_pt(Pum, umx, umy, umz, umw); // -Up
526 T fpx, fpy, fpz, fpw; proj_pt(Pfp, fpx, fpy, fpz, fpw); // +Forward（奥）
527 T fmx, fmy, fmz, fmw; proj_pt(Pfm, fmx, fmy, fmz, fmw); // -Forward（手前）
528
529 // 3) NDC へ（同次除算）
530 const T ndc_cx = safe_div(cx, cw);
531 const T ndc_cy = safe_div(cy, cw);
532 const T ndc_cz = safe_div(cz, cw);
533
534 const T ndc_rp_x = safe_div(rpx, rpw);
535 const T ndc_rm_x = safe_div(rm_x, rm_w);
536 const T ndc_up_y = safe_div(upy, upw);
537 const T ndc_um_y = safe_div(umy, umw);
538
539 const T ndc_fp_z = safe_div(fpz, fpw);
540 const T ndc_fm_z = safe_div(fmz, fmw);
541
542 // 4) スクリーン半径（保守的）：左右は Right±、上下は Up± の最大差分
543 const T r_ndc_x = (std::max)(std::fabs(ndc_rp_x - ndc_cx),
544 std::fabs(ndc_rm_x - ndc_cx));
545 const T r_ndc_y = (std::max)(std::fabs(ndc_up_y - ndc_cy),
546 std::fabs(ndc_um_y - ndc_cy));
547 const T r_ndc = (std::max)(r_ndc_x, r_ndc_y);
548
549 // 5) NDC 矩形
550 T xmin = ndc_cx - r_ndc;
551 T xmax = ndc_cx + r_ndc;
552 T ymin = ndc_cy - r_ndc;
553 T ymax = ndc_cy + r_ndc;
554
555 // 6) z 範囲（保守的）：中心と Forward± の 3 サンプル
556 const T zmin_est = (std::min)((std::min)(ndc_cz, ndc_fm_z), ndc_fp_z);
557 const T zmax_est = (std::max)((std::max)(ndc_cz, ndc_fm_z), ndc_fp_z);
558
559 // 7) クリップとの交差（x,y は [-1,1]、z は [0,1]）
560 const bool x_overlap = !(xmax < -1.0f || xmin > 1.0f);
561 const bool y_overlap = !(ymax < -1.0f || ymin > 1.0f);
562 const bool z_overlap = !(zmax_est < 0.0f || zmin_est > 1.0f);
563
564 // 8) 出力
565 if (outNDC) {
566 outNDC->xmin = xmin; outNDC->xmax = xmax;
567 outNDC->ymin = ymin; outNDC->ymax = ymax;
568
569 const T raw_minw = (std::min)(
570 (std::min)((std::min)(cw, rpw), (std::min)(rm_w, upw)),
571 (std::min)(umw, (std::min)(fpw, fmw)));
572 const T epsW = 1e-6f;
573 outNDC->wmin = (raw_minw < epsW) ? epsW : raw_minw;
574 }
575 if (depth) *depth = cw; // 互換用
576
577 return x_overlap && y_overlap && z_overlap;
578 }

584 template<class Mat4, class NDC>
585 bool IsVisible_WVP_CamBasis_Fast(
586 const Mat4& WVP,
587 const Vec3& camRight, const Vec3& camUp, const Vec3& camForward,
588 NDC* outNDC, T* depth = nullptr) const noexcept
589 {
590 using ::SFW::Math::MulPoint_RowMajor_ColVec;
591
592 // 1) 中心を clip に
593 T cx, cy, cz, cw;
594 MulPoint_RowMajor_ColVec(WVP, center.x, center.y, center.z, cx, cy, cz, cw);
595
596 // 2) WVP の線形部分を抽出
597 const T m00 = WVP.m00, m01 = WVP.m01, m02 = WVP.m02;
598 const T m10 = WVP.m10, m11 = WVP.m11, m12 = WVP.m12;
599 const T m20 = WVP.m20, m21 = WVP.m21, m22 = WVP.m22;
600 const T m30 = WVP.m30, m31 = WVP.m31, m32 = WVP.m32; // w行成分も必要
601
602 // 3) 半径方向を clip空間へ変換（ベクトル変換：平行移動は含まない）
603 auto to_clip = [&](const Vec3& v)->Vec3 {
604 return {
605 m00 * v.x + m01 * v.y + m02 * v.z,
606 m10 * v.x + m11 * v.y + m12 * v.z,
607 m20 * v.x + m21 * v.y + m22 * v.z
608 };
609 };
610 auto to_clip_w = [&](const Vec3& v)->T {
611 return m30 * v.x + m31 * v.y + m32 * v.z;
612 };
613
614 const Vec3 clipR = to_clip(camRight);
615 const Vec3 clipU = to_clip(camUp);
616 const Vec3 clipF = to_clip(camForward);
617 const T wR = to_clip_w(camRight);
618 const T wU = to_clip_w(camUp);
619 const T wF = to_clip_w(camForward);
620
621 // 4) clip 半径（+R, +U の大きい方）
622 // 割り算を一回で済ませるため invCW を先に計算
623 const T invCW = 1.0f / ((std::fabs(cw) < 1e-6f) ? ((cw < 0) ? -1e-6f : 1e-6f) : cw);
624
625 const T ndc_cx = cx * invCW;
626 const T ndc_cy = cy * invCW;
627 const T ndc_cz = cz * invCW;
628
629 // 5) NDC半径計算（近似：微分ベース）
630 // d(x/w) ≈ (dx*w - x*dw)/w²
631 const T dxR = clipR.x * cw - cx * wR;
632 const T dyU = clipU.y * cw - cy * wU;
633 const T dzF = clipF.z * cw - cz * wF;
634 const T invCW2 = invCW * invCW;
635 const T r_ndc_x = std::fabs(radius * dxR * invCW2);
636 const T r_ndc_y = std::fabs(radius * dyU * invCW2);
637 const T r_ndc = (std::max)(r_ndc_x, r_ndc_y);
638
639 // 6) z 範囲（前後 ±radius * Forward）
640 const T r_ndc_z = std::fabs(radius * dzF * invCW2);
641 const T zmin_est = ndc_cz - r_ndc_z;
642 const T zmax_est = ndc_cz + r_ndc_z;
643
644 // 7) 矩形・交差判定
645 const T xmin = ndc_cx - r_ndc;
646 const T xmax = ndc_cx + r_ndc;
647 const T ymin = ndc_cy - r_ndc;
648 const T ymax = ndc_cy + r_ndc;
649
650 const bool x_overlap = !(xmax < -1.0f || xmin > 1.0f);
651 const bool y_overlap = !(ymax < -1.0f || ymin > 1.0f);
652 const bool z_overlap = !(zmax_est < 0.0f || zmin_est > 1.0f);
653
654 if (outNDC) {
655 outNDC->xmin = xmin; outNDC->xmax = xmax;
656 outNDC->ymin = ymin; outNDC->ymax = ymax;
657 outNDC->wmin = cw;
658 }
659 if (depth) *depth = cw;
660
661 return x_overlap && y_overlap && z_overlap;
662 }

664 enum class VisState : uint32_t {
665 Culled = 0, // 画面外（この関数のNDC/rect判定で落ちた）
666 Visible = 1, // 可視（rectが画面と交差）
667 NeedFrustum = 2 // 近すぎ/背面などでrect推定が不安定 → フラスタム球判定へ
668 };

683 template<class Mat4, class NDC>
684 static VisState IsVisible_LocalCenter_WorldRadius(
685 const Mat4& WVP, // = VP * World
686 const Mat4& VP, // = Projection * View
687 const Vec3& centerLocal, // モデルローカルの中心
688 T radiusWorld, // ワールド空間の半径（スケール反映済み）
689 const Vec3& camRightWS, // ワールド空間のカメラ軸
690 const Vec3& camUpWS,
691 const Vec3& camForwardWS,
692 NDC* outNDC,
693 T* depth = nullptr
694 ) noexcept
695 {
696 using ::SFW::Math::MulPoint_RowMajor_ColVec;
697
698 // 1) center -> clip
699 T cx, cy, cz, cw;
700 MulPoint_RowMajor_ColVec(WVP,
701 centerLocal.x, centerLocal.y, centerLocal.z,
702 cx, cy, cz, cw);
703
704 // ---- フォールバック条件（近すぎ/背面）----
705 // ここはプロジェクトに合わせて閾値調整。
706 // 目的：cw が小さくて r_ndc が爆発しやすい/背面で不安定なケースを除外
707 const T epsW = static_cast<T>(1e-4f); // 1e-6 より少し大きめ推奨
708 if (cw <= epsW) {
709 if (outNDC) {
710 outNDC->xmin = -1; outNDC->xmax = 1;
711 outNDC->ymin = -1; outNDC->ymax = 1;
712 outNDC->wmin = cw;
713 }
714 if (depth) { *depth = cw; }
715 return VisState::NeedFrustum;
716 }
717
718 const T invCW = static_cast<T>(1) / cw;
719 const T invCW2 = invCW * invCW;
720
721 const T ndc_cx = cx * invCW;
722 const T ndc_cy = cy * invCW;
723 const T ndc_cz = cz * invCW;
724
725 // VP の行（clipX, clipY, clipZ, clipW の xyz だけ）
726 const T x0 = VP.m00, x1 = VP.m01, x2 = VP.m02;
727 const T y0 = VP.m10, y1 = VP.m11, y2 = VP.m12;
728 const T z0 = VP.m20, z1 = VP.m21, z2 = VP.m22;
729 const T w0 = VP.m30, w1 = VP.m31, w2 = VP.m32;
730
731 // dot(row, v) : v は world 方向（camRightWS 等）
732 auto dot3 = [&](T r0, T r1, T r2, const Vec3& v)->T {
733 return r0 * v.x + r1 * v.y + r2 * v.z;
734 };
735
736 // df = d(X/W) along v ≈ (dX * W - X * dW) / W^2
737 auto d_ndc_along = [&](T X, T W, T dX, T dW)->T {
738 return (dX * W - X * dW) * invCW2;
739 };
740
741 auto grad_norm_ndc = [&](T X, T W, T row0, T row1, T row2)->T
742 {
743 // dX, dW for each camera basis direction
744 const T dX_R = dot3(row0, row1, row2, camRightWS);
745 const T dX_U = dot3(row0, row1, row2, camUpWS);
746 const T dX_F = dot3(row0, row1, row2, camForwardWS);
747
748 const T dW_R = dot3(w0, w1, w2, camRightWS);
749 const T dW_U = dot3(w0, w1, w2, camUpWS);
750 const T dW_F = dot3(w0, w1, w2, camForwardWS);
751
752 const T a = d_ndc_along(X, W, dX_R, dW_R);
753 const T b = d_ndc_along(X, W, dX_U, dW_U);
754 const T c = d_ndc_along(X, W, dX_F, dW_F);
755
756 return std::sqrt(a * a + b * b + c * c); // ≒ ||∇(X/W)||
757 };
758
759 const T absR = std::fabs(radiusWorld);
760
761 const T r_ndc_x = absR * grad_norm_ndc(cx, cw, x0, x1, x2); // for ndc_x
762 const T r_ndc_y = absR * grad_norm_ndc(cy, cw, y0, y1, y2); // for ndc_y
763 const T r_ndc_z = absR * grad_norm_ndc(cz, cw, z0, z1, z2); // for ndc_z
764
765 // 4) NDC矩形
766 const T xmin = ndc_cx - r_ndc_x;
767 const T xmax = ndc_cx + r_ndc_x;
768 const T ymin = ndc_cy - r_ndc_y;
769 const T ymax = ndc_cy + r_ndc_y;
770
771 // z範囲（深度前提に合わせる）
772 const T zmin_est = ndc_cz - r_ndc_z;
773 const T zmax_est = ndc_cz + r_ndc_z;
774
775 const bool x_overlap = !(xmax < -1.0f || xmin > 1.0f);
776 const bool y_overlap = !(ymax < -1.0f || ymin > 1.0f);
777 const bool z_overlap = !(zmax_est < 0.0f || zmin_est > 1.0f); // [0,1] 前提
778
779 if (outNDC) {
780 outNDC->xmin = xmin; outNDC->xmax = xmax;
781 outNDC->ymin = ymin; outNDC->ymax = ymax;
782 outNDC->wmin = cw;
783 }
784 if (depth) {
785 *depth = cw; // 用途次第で ndc_cz にしてもOK
786 }
787
788 if (x_overlap && y_overlap && z_overlap) return VisState::Visible;
789 return VisState::Culled;
790 }

943 const __m256 r_ndc = _mm256_max_ps(r_ndc_x, r_ndc_y);
944

946

948 const __m256 xmax = _mm256_add_ps(ndcX, r_ndc);
949 const __m256 ymin = _mm256_sub_ps(ndcY, r_ndc);
950 const __m256 ymax = _mm256_add_ps(ndcY, r_ndc);
951
952 const __m256 zmin = _mm256_sub_ps(ndcZ, r_ndc_z);
953 const __m256 zmax = _mm256_add_ps(ndcZ, r_ndc_z);
954
955 if (outDepthMin) _mm256_storeu_ps(outDepthMin + i, zmin);
956 if (outDepthMax) _mm256_storeu_ps(outDepthMax + i, zmax);
957
958 // ---- overlap 判定 ----
959 const __m256 x_ok = _mm256_and_ps(_mm256_cmp_ps(xmax, vNegOne, _CMP_GE_OQ),
960 _mm256_cmp_ps(xmin, vPosOne, _CMP_LE_OQ));

962 _mm256_cmp_ps(ymin, vPosOne, _CMP_LE_OQ));

964 _mm256_cmp_ps(zmin, vMaxZ, _CMP_LE_OQ));
965
966 const __m256 ok = _mm256_and_ps(_mm256_and_ps(x_ok, y_ok), z_ok);
967
968 int mask = _mm256_movemask_ps(ok);
969 if (laneCount < 8) mask &= ((1 << (int)laneCount) - 1);
970 return (std::uint32_t)mask;
971#else

973 (void)outNdcX; (void)outNdcY; (void)outNdcZ; (void)outDepthMin; (void)outDepthMax;
974 return 0;
975#endif
