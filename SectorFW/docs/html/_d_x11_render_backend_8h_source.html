166 void ExecuteDrawIndexedInstancedImpl(const VecT& cmds, std::optional<PSOHandle> psoOverride, bool usePSORasterizer, bool rebindPSO)
167 {
168 struct DrawBatch {
169 uint32_t mesh;
170 uint32_t material;
171 uint32_t pso;
172 uint32_t base;
173 uint32_t instanceCount; // instances[idx]
174 };
175
176 BeginIndexStream();
177
178 size_t i = 0;
179 size_t cmdCount = cmds.size();
180 std::vector<DrawBatch> batches;
181 if (psoOverride.has_value()) {
182 // PSOオーバーライドがある場合、すべて同じPSOで描画する
183 uint32_t overriddenPSO = psoOverride.value().index;
184 while (i < cmdCount) {
185 auto currentMat = cmds[i].material;
186
187 uint32_t currentMesh = cmds[i].mesh;
188 uint32_t instanceCount = 0;
189
190 const uint32_t base = m_idxHead; // このドローの index 先頭
191
192 // 1) 同PSO/Mat/Mesh を束ねつつ、index を SRV に“直接”書く
193 auto* dst = reinterpret_cast<uint32_t*>(m_idxMapped) + m_idxHead;
194
195 // 同じPSO + Material + Meshをまとめる
196 while (i < cmdCount &&
197 cmds[i].material == currentMat &&
198 cmds[i].mesh == currentMesh &&
199 instanceCount < MAX_DRAW_CALL_INSTANCES_NUM) {
200 dst[instanceCount++] = cmds[i].instanceIndex.index; // ← 直接書く = instances[idx];
201 ++i;
202 }
203 m_idxHead += instanceCount;
204
205 batches.emplace_back(currentMesh, currentMat, overriddenPSO, base, instanceCount);
206 }
207 }
208 else
209 {
210 while (i < cmdCount) {
211 auto currentPSO = cmds[i].pso;
212 auto currentMat = cmds[i].material;
213
214 uint32_t currentMesh = cmds[i].mesh;
215 uint32_t instanceCount = 0;
216
217 const uint32_t base = m_idxHead; // このドローの index 先頭
218
219 // 1) 同PSO/Mat/Mesh を束ねつつ、index を SRV に“直接”書く
220 auto* dst = reinterpret_cast<uint32_t*>(m_idxMapped) + m_idxHead;
221
222 // 同じPSO + Material + Meshをまとめる
223 while (i < cmdCount &&
224 cmds[i].pso == currentPSO &&
225 cmds[i].material == currentMat &&
226 cmds[i].mesh == currentMesh &&
227 instanceCount < MAX_DRAW_CALL_INSTANCES_NUM) {
228 dst[instanceCount++] = cmds[i].instanceIndex.index; // ← 直接書く = instances[idx];
229 ++i;
230 }
231 m_idxHead += instanceCount;
232
233 batches.emplace_back(currentMesh, currentMat, currentPSO, base, instanceCount);
234 }
235 }
236
237 EndIndexStream();
238
239 context->VSSetShaderResources(0, 1, m_instanceSRV.GetAddressOf()); // InstanceData (t0)
240 context->VSSetConstantBuffers(1, 1, m_perDrawCB.GetAddressOf()); // (b1)
241 for (const auto& b : batches) {
242 // PerDraw CB に base と count を設定
243 struct { uint32_t base, count, pad0, pad1; }
244 perDraw{ b.base, b.instanceCount, 0, 0 };
245 D3D11_MAPPED_SUBRESOURCE m{};
246 context->Map(m_perDrawCB.Get(), 0, D3D11_MAP_WRITE_DISCARD, 0, &m);
247 memcpy(m.pData, &perDraw, sizeof(perDraw));
248 context->Unmap(m_perDrawCB.Get(), 0);
249
250 DrawInstanced(b.mesh, b.material, b.pso, b.instanceCount, usePSORasterizer, rebindPSO);
251 }
