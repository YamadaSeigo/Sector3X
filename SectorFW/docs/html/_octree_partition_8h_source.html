211 std::vector<SpatialChunk*> CullChunks(const Math::Vec3f& center, float radius) const noexcept
212 {
213 std::vector<SpatialChunk*> out;
214 if (!m_root) return out;
215
216 if (!std::isfinite(radius) || radius <= 0.0f) radius = 0.0f;
217 const float r = (radius < 0.0f) ? 0.0f : radius;
218 const float r2 = r * r;
219
220 std::function<void(const Node&)> rec = [&](const Node& n) {
221 // node AABB vs sphere
222 const Math::Vec3f c = levelOriginWS + (n.bounds.lb + n.bounds.ub) * 0.5f;
223 const Math::Vec3f e = (n.bounds.ub - n.bounds.lb) * 0.5f;
224 if (Dist2PointAABB3D(center, c, e) > r2) return;
225
226 if (n.isLeaf()) {
227 if (n.chunk.GetEntityManager().GetEntityCount() > 0)
228 out.push_back(const_cast<SpatialChunk*>(&n.chunk));
229 return;
230 }
231 for (int i = 0; i < 8; ++i) if (n.child[i]) rec(*n.child[i]);
232 };
233 rec(*m_root);
234
235 return out;
236 }

238 static inline float Dist2PointAABB3D(const Math::Vec3f& p,
239 const Math::Vec3f& c,
240 const Math::Vec3f& e)
241 {
242 auto clamp = [](float v, float lo, float hi) { return v < lo ? lo : (v > hi ? hi : v); };
243 const float qx = clamp(p.x, c.x - e.x, c.x + e.x);
244 const float qy = clamp(p.y, c.y - e.y, c.y + e.y);
245 const float qz = clamp(p.z, c.z - e.z, c.z + e.z);
246 const float dx = p.x - qx, dy = p.y - qy, dz = p.z - qz;
247 return dx * dx + dy * dy + dz * dz;
248 }

250 // 可視チャンクを近い順で上位K件
251 std::vector<SpatialChunk*> CullChunksNear(const Math::Frustumf& fr,
252 const Math::Vec3f& eye,
253 size_t maxCount = (std::numeric_limits<size_t>::max)()) const noexcept
254 {
255 struct Item { SpatialChunk* sc; float d2; };
256
257 std::vector<Item> items; items.reserve(128);
258 if (!m_root) return {};
259
260 // 可視葉 (chunk, bounds) を同時に収集する内部再帰
261 std::function<void(const Node&)> rec = [&](const Node& n) {
262 if (!nodeIntersectsFrustum3D(n, fr)) return;
263 if (n.isLeaf()) {
264 const Math::Vec3f c = levelOriginWS + (n.bounds.lb + n.bounds.ub) * 0.5f;
265 const Math::Vec3f e = (n.bounds.ub - n.bounds.lb) * 0.5f;
266 const float d2 = Dist2PointAABB3D(eye, c, e);
267 if (n.chunk.GetEntityManager().GetEntityCount() > 0)
268 items.push_back({ const_cast<SpatialChunk*>(&n.chunk), d2 });
269 return;
270 }
271 for (int i = 0; i < 8; ++i) if (n.child[i]) rec(*n.child[i]);
272 };
273 rec(*m_root);
274
275 if (items.empty()) return {};
276 const size_t K = (std::min)(maxCount, items.size());
277 std::nth_element(items.begin(), items.begin() + K, items.end(),
278 [](const Item& a, const Item& b) { return a.d2 < b.d2; });
279 items.resize(K);
280 std::sort(items.begin(), items.end(),
281 [](const Item& a, const Item& b) { return a.d2 < b.d2; });
282
283 std::vector<SpatialChunk*> out; out.reserve(K);
284 for (auto& it : items) { out.push_back(it.sc); }
285 return out;
286 }

298 uint32_t CullChunkLine(const Math::Frustumf& fr,
299 const Math::Vec3f& eye,
300 float, //dummy
301 Debug::LineVertex* outLine,
302 uint32_t capacity,
303 uint32_t displayCount) const noexcept
304 {
305 if (!m_root || capacity < 24 || displayCount == 0) return 0;
306
307 // 1) 可視ボックス収集（3Dフラスタム判定）
308 std::vector<AABB> boxes; boxes.reserve(256);
309 cullRecursive3D(*m_root, fr, boxes);
310 if (boxes.empty()) return 0;
311
312 // 2) 目線からの距離で前から優先（近い順）
313 struct Item { AABB box; float dist; };
314 std::vector<Item> items; items.reserve(boxes.size());
315 auto clampToBox = [&](const AABB& b, const Math::Vec3f& p) {
316 return Math::Vec3f{
317 std::clamp(p.x, b.lb.x + levelOriginWS.x, b.ub.x + levelOriginWS.x),
318 std::clamp(p.y, b.lb.y + levelOriginWS.y, b.ub.y + levelOriginWS.y),
319 std::clamp(p.z, b.lb.z + levelOriginWS.z, b.ub.z + levelOriginWS.z)
320 };
321 };
322 for (const auto& b : boxes) {
323 const Math::Vec3f q = clampToBox(b, eye);
324 const float d = (q - eye).length();
325 items.push_back({ b, d });
326 }
327 std::nth_element(items.begin(), items.begin() + std::min<size_t>(displayCount, items.size()), items.end(),
328 [](const Item& a, const Item& b) { return a.dist < b.dist; });
329 const size_t useN = std::min<size_t>(displayCount, items.size());
330
331 // 3) 色は距離グラデ（近=白→遠=黒）
332 float maxD = 0.f; for (size_t i = 0; i < useN; ++i) maxD = (std::max)(maxD, items[i].dist);
333 if (maxD <= 1e-6f) maxD = 1.f;
334
335 auto pushEdge = [&](uint32_t& w, const Math::Vec3f& a, const Math::Vec3f& b, uint32_t rgba) {
336 if (w + 2 > capacity) return false;
337 outLine[w++] = { a, rgba };
338 outLine[w++] = { b, rgba };
339 return true;
340 };
341
342 uint32_t written = 0;
343 for (size_t i = 0; i < useN; ++i) {
344 const auto& box = items[i].box;
345 const float t = items[i].dist / maxD; // 0..1
346 const uint32_t rgba = Math::LerpColor(0xFFFFFFFFu, 0x00000000u, t);
347
348 const Math::Vec3f c = box.center() + levelOriginWS;
349 const Math::Vec3f e = box.size() * 0.5f;
350 Math::Vec3f v[8]{
351 {c.x - e.x, c.y - e.y, c.z - e.z},
352 {c.x + e.x, c.y - e.y, c.z - e.z},
353 {c.x + e.x, c.y + e.y, c.z - e.z},
354 {c.x - e.x, c.y + e.y, c.z - e.z},
355 {c.x - e.x, c.y - e.y, c.z + e.z},
356 {c.x + e.x, c.y - e.y, c.z + e.z},
357 {c.x + e.x, c.y + e.y, c.z + e.z},
358 {c.x - e.x, c.y + e.y, c.z + e.z},
359 };
360 const int edge[12][2] = {
361 {0,1},{1,2},{2,3},{3,0},
362 {4,5},{5,6},{6,7},{7,4},
363 {0,4},{1,5},{2,6},{3,7}
364 };
365 for (int eidx = 0; eidx < 12; ++eidx) {
366 if (!pushEdge(written, v[edge[eidx][0]], v[edge[eidx][1]], rgba))
367 return written;
368 }
369 }
370 return written;
371 }

389 SpatialChunk* EnsureLeafForPoint(Math::Vec3f p)
390 {
391 if (!inBounds(p.x, p.y, p.z)) {
392 p.x = std::clamp(p.x, 0.f, float(m_worldX) - 1e-6f);
393 p.y = std::clamp(p.y, 0.f, float(m_worldY) - 1e-6f);
394 p.z = std::clamp(p.z, 0.f, float(m_worldZ) - 1e-6f);
395 }
396 Node* n = m_root.get();
397 while (canSplit(*n)) {
398 if (n->isLeaf()) ensureChildren(*n);
399 const int oi = octant(*n, p.x, p.y, p.z);
400 n = n->child[oi].get();
401 }
402 return &n->chunk;
403 }

563 static bool intersects(const AABB& a, const AABB& b) noexcept {
564 return !(a.ub.x <= b.lb.x || a.lb.x >= b.ub.x ||
565 a.ub.y <= b.lb.y || a.lb.y >= b.ub.y ||
566 a.ub.z <= b.lb.z || a.lb.z >= b.ub.z);
567 }
568 bool inBounds(float x, float y, float z) const noexcept {
569 return (0.f <= x && x < float(m_worldX)) &&
570 (0.f <= y && y < float(m_worldY)) &&
571 (0.f <= z && z < float(m_worldZ));
572 }
573 bool canSplit(const Node& n) const noexcept {
574 const Math::Vec3f s = n.bounds.size();
575 return (s.x > m_minLeaf) && (s.y > m_minLeaf) && (s.z > m_minLeaf);

578 int octant(const Node& n, float x, float y, float z) const noexcept {
579 const float mx = 0.5f * (n.bounds.lb.x + n.bounds.ub.x);
580 const float my = 0.5f * (n.bounds.lb.y + n.bounds.ub.y);
581 const float mz = 0.5f * (n.bounds.lb.z + n.bounds.ub.z);
582 const int ex = (x >= mx) ? 1 : 0; // East(1)/West(0)
583 const int uy = (y >= my) ? 1 : 0; // Up(1)/Down(0)
584 const int nz = (z >= mz) ? 1 : 0; // North(1)/South(0)
585 // ビット: x(1) | y(2) | z(4)
586 return (ex) | (uy << 1) | (nz << 2); // 0..7
587 }

589 void ensureChildren(Node& n)
590 {
591 if (!n.isLeaf()) return;
592 const float mx = 0.5f * (n.bounds.lb.x + n.bounds.ub.x);
593 const float my = 0.5f * (n.bounds.lb.y + n.bounds.ub.y);
594 const float mz = 0.5f * (n.bounds.lb.z + n.bounds.ub.z);
595
596 const AABB octs[8] = {
597 // x:[lb..mx], y:[lb..my], z:[lb..mz]
598 { { n.bounds.lb.x, n.bounds.lb.y, n.bounds.lb.z }, { mx, my, mz } }, // 0: W-D-S
599 { { mx, n.bounds.lb.y, n.bounds.lb.z }, { n.bounds.ub.x, my, mz } }, // 1: E-D-S
600 { { n.bounds.lb.x, my, n.bounds.lb.z }, { mx, n.bounds.ub.y, mz } }, // 2: W-U-S
601 { { mx, my, n.bounds.lb.z }, { n.bounds.ub.x, n.bounds.ub.y, mz } }, // 3: E-U-S
602 { { n.bounds.lb.x, n.bounds.lb.y, mz }, { mx, my, n.bounds.ub.z } }, // 4: W-D-N
603 { { mx, n.bounds.lb.y, mz }, { n.bounds.ub.x, my, n.bounds.ub.z } }, // 5: E-D-N
604 { { n.bounds.lb.x, my, mz }, { mx, n.bounds.ub.y, n.bounds.ub.z } }, // 6: W-U-N
605 { { mx, my, mz }, { n.bounds.ub.x, n.bounds.ub.y, n.bounds.ub.z } } // 7: E-U-N
606 };
607 for (int i = 0; i < 8; ++i) {
608 n.child[i] = std::make_unique<Node>();
609 n.child[i]->bounds = octs[i];
610 n.child[i]->depth = uint8_t(n.depth + 1);
611 n.child[i]->generation = 0;
612 }
613 m_leafCount += 7; // 葉1 → 子8
614 }

616 Node* descendToLeaf(Node& cur, float x, float y, float z, bool createIfMissing)
617 {
618 auto posFn = [](ECS::EntityID id, ECS::EntityManager& mgr) -> std::optional<Math::Vec3f> {
619 auto tf = mgr.ReadComponent<CTransform>(id);
620 if (!tf.has_value()) return std::nullopt;
621 return tf->location;
622 };
623
624 Node* n = &cur;
625 while (canSplit(*n)) {
626 if (n->isLeaf()) {
627 if (!createIfMissing) break;
628 if (n->chunk.GetEntityManager().GetEntityCount() > 0) {
629 subdivideAndReassign(*n, posFn); // 中身あり -> 分割して再配置
630 }
631 else {
632 ensureChildren(*n); // 空 -> 子生成のみ
633 }
634 }
635 const int oi = octant(*n, x, y, z);
636 n = n->child[oi].get();
637 }
638 return n;

687 return fr.IntersectsAABB(center, extent);

753 // ---- 分割と再配置 ----
754 void subdivideAndReassign(Node& leaf,
755 const std::function<std::optional<Math::Vec3f>(ECS::EntityID, ECS::EntityManager&)>& posFn)
756 {
757 if (!leaf.isLeaf() || !canSplit(leaf)) return;
758
759 ensureChildren(leaf);
760 ECS::EntityManager& src = leaf.chunk.GetEntityManager();
761
762 auto router = [&](ECS::EntityID id, const ECS::ComponentMask /*mask*/) -> ECS::EntityManager* {
763 const std::optional<Math::Vec3f> pos = posFn(id, src);
764 if (!pos) return nullptr;
765 const int oi = octant(leaf, pos->x, pos->y, pos->z);
766 return &leaf.child[oi]->chunk.GetEntityManager();
767 };
768 (void)src.SplitByAll(router); // 非スパース→逐次, スパース→IDバケット一括
769 ++leaf.generation;
770 }

776 size_t CoalesceUnderutilized()
777 {
778 if (!m_root) return 0;
779 std::vector<Node*> stack; stack.push_back(m_root.get());
780 std::vector<Node*> post;
781 while (!stack.empty()) {
782 Node* n = stack.back(); stack.pop_back();
783 post.push_back(n);
784 for (auto& c : n->child) if (c) stack.push_back(c.get());
785 }
786 size_t mergedParents = 0;
787
788 for (auto it = post.rbegin(); it != post.rend(); ++it) {
789 Node* n = *it;
790 bool allLeaf = true; for (int i = 0; i < 8; ++i) { if (!n->child[i] || !n->child[i]->isLeaf()) { allLeaf = false; break; } }
791 if (!allLeaf) continue;
792
793 size_t sum = 0; for (int i = 0; i < 8; ++i) sum += n->child[i]->chunk.GetEntityManager().GetEntityCount();
794 if (sum <= m_minPerLeafCount) {
795 auto& dst = n->chunk.GetEntityManager();
796 for (int i = 0; i < 8; ++i) {
797 auto& src = n->child[i]->chunk.GetEntityManager();
798 (void)dst.MergeFromAll(src);
799 }
800 n->child = {};
801 ++n->generation;
802 m_leafCount -= 7; // 8 -> 1
803 ++mergedParents;
804 }
805 }
806 return mergedParents;
807 }

809 // ---- キー生成（Morton3D 64bit 符号化） ----
810 static inline uint64_t ExpandBits3(uint64_t v)
811 {
812 v = (v | (v << 32)) & 0x1F00000000FFFFull;
813 v = (v | (v << 16)) & 0x1F0000FF0000FFull;
814 v = (v | (v << 8)) & 0x100F00F00F00F00Full;
815 v = (v | (v << 4)) & 0x10C30C30C30C30C3ull;
816 v = (v | (v << 2)) & 0x1249249249249249ull;
817 return v;
818 }
819 static inline uint64_t Morton3D64(uint64_t x, uint64_t y, uint64_t z)
820 {
821 return (ExpandBits3(x) << 0) | (ExpandBits3(y) << 1) | (ExpandBits3(z) << 2);
822 }

825 uint32_t ix, uint32_t iy, uint32_t iz,
