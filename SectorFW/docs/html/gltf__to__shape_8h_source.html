59 inline bool BuildMeshDescFromGLTFMesh(const cgltf_mesh* mesh, MeshDesc& out) {
60 if (!mesh) return false;
61
62 std::vector<Vec3f> allPos;
63 std::vector<uint32_t> allIdx;
64 uint32_t baseVertex = 0;
65
66 for (cgltf_size p = 0; p < mesh->primitives_count; ++p) {
67 const cgltf_primitive& prim = mesh->primitives[p];
68 // triangles のみ対象（その他はスキップ or 変換）
69 if (prim.type != cgltf_primitive_type_triangles) continue;
70
71 // POSITION
72 const cgltf_accessor* posAcc = nullptr;
73 for (cgltf_size a = 0; a < prim.attributes_count; ++a) {
74 const cgltf_attribute& attr = prim.attributes[a];
75 if (attr.type == cgltf_attribute_type_position) { posAcc = attr.data; break; }
76 }
77 if (!posAcc) continue;
78
79 // 頂点
80 std::vector<Vec3f> localPos;
81 if (!ReadPositions(posAcc, localPos)) continue;
82
83 // indices（無ければ 0..N-1 連番）
84 if (prim.indices) {
85 ReadIndices(prim.indices, allIdx, baseVertex);
86 }
87 else {
88 // 連番
89 for (uint32_t i = 0; i + 2 < (uint32_t)localPos.size(); i += 3) {
90 allIdx.push_back(baseVertex + i + 0);
91 allIdx.push_back(baseVertex + i + 1);
92 allIdx.push_back(baseVertex + i + 2);
93 }
94 }
95
96 // 末尾に連結
97 allPos.insert(allPos.end(), localPos.begin(), localPos.end());
98 baseVertex += (uint32_t)localPos.size();
99 }
100
101 if (allPos.empty() || allIdx.empty()) return false;
102 out.vertices = std::move(allPos);
103 out.indices = std::move(allIdx);
104 return true;
105 }
