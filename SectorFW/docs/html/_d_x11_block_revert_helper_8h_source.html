160 inline bool CreateRawUAV(ID3D11Device* dev, UINT byteSize, ComPtr<ID3D11Buffer>& buf, ComPtr<ID3D11UnorderedAccessView>& uav)
161 {
162 D3D11_BUFFER_DESC bd{};
163 bd.ByteWidth = byteSize;
164 bd.Usage = D3D11_USAGE_DEFAULT;
165 bd.BindFlags = D3D11_BIND_UNORDERED_ACCESS;
166 bd.CPUAccessFlags = 0;
167 bd.MiscFlags = D3D11_RESOURCE_MISC_BUFFER_ALLOW_RAW_VIEWS;
168 HRESULT hr = dev->CreateBuffer(&bd, nullptr, &buf);
169 if (FAILED(hr)) return false;
170
171 D3D11_UNORDERED_ACCESS_VIEW_DESC ud{};
172 ud.ViewDimension = D3D11_UAV_DIMENSION_BUFFER;
173 ud.Format = DXGI_FORMAT_R32_TYPELESS; // RAW
174 ud.Buffer.FirstElement = 0;
175 ud.Buffer.NumElements = byteSize / 4; // RAW: elements are 4-byte units
176 ud.Buffer.Flags = D3D11_BUFFER_UAV_FLAG_RAW;
177 hr = dev->CreateUnorderedAccessView(buf.Get(), &ud, &uav);
178 return SUCCEEDED(hr);
179 }

220 ComPtr<ID3D11Buffer>& buf,
221 ComPtr<ID3D11ShaderResourceView>* srv,
222 ComPtr<ID3D11UnorderedAccessView>& uav,

224 {

226 bd.ByteWidth = count * cascadeSize * sizeof(UINT);
227 bd.Usage = D3D11_USAGE_DEFAULT;
228 bd.BindFlags = D3D11_BIND_SHADER_RESOURCE | (asUAV ? D3D11_BIND_UNORDERED_ACCESS : 0);
229 bd.CPUAccessFlags = 0;

234 D3D11_SHADER_RESOURCE_VIEW_DESC sd{};
235 for (UINT c = 0; c < cascadeSize; ++c) {
236 sd.ViewDimension = D3D11_SRV_DIMENSION_BUFFER;
237 sd.Format = DXGI_FORMAT_UNKNOWN;
238 sd.Buffer.FirstElement = c * count;

241 }
242

244 D3D11_UNORDERED_ACCESS_VIEW_DESC ud{}; ud.ViewDimension = D3D11_UAV_DIMENSION_BUFFER; ud.Format = DXGI_FORMAT_UNKNOWN;
245 ud.Buffer.FirstElement = 0; ud.Buffer.NumElements = count * cascadeSize; ud.Buffer.Flags = 0;
246 hr = dev->CreateUnorderedAccessView(buf.Get(), &ud, &uav); if (FAILED(hr)) return false;

936 inline void GenerateClusterLODs_meshopt_fast(
937 const std::vector<uint32_t>& inIndexPool,
938 const std::vector<TerrainClustered::ClusterRange>& inRanges,
939 const float* positions, size_t vertexCount, size_t positionStrideBytes,
940 const std::vector<float>& lodTargets,
941 // outputs
942 std::vector<uint32_t>& outIndexPool,
943 std::vector<ClusterLodRange>& outLodRanges,
944 std::vector<uint32_t>& outLodBase,
945 std::vector<uint32_t>& outLodCount)
946 {
947
948 clock_t start = clock();
949
950 outIndexPool.clear();
951 outLodRanges.clear();
952 outLodBase.assign(inRanges.size(), 0);
953 outLodCount.assign(inRanges.size(), 0);
954
955 const size_t levels = lodTargets.size();
956 if (levels == 0 || inRanges.empty()) return;
957
958 // 1) 並列に「各クラスタのローカル出力」を作る
959 struct LocalOut {
960 std::vector<uint32_t> indices; // 後で outIndexPool に連結
961 std::vector<ClusterLodRange> ranges; // offset は local 基準（後で +global base）
962 };
963 std::vector<LocalOut> locals(inRanges.size());
964
965 std::vector<size_t> ids(inRanges.size());
966 std::iota(ids.begin(), ids.end(), size_t{ 0 });
967
968 std::for_each(std::execution::par, ids.begin(), ids.end(), [&](size_t cid)
969 {
970 const auto& r = inRanges[cid];
971 if (r.indexCount < 3) return;
972
973 // ---- thread_local scratch ----
974 struct Scratch {
975 std::vector<uint32_t> srcIndices;
976 std::vector<uint32_t> remap;
977 std::vector<uint32_t> invRemap;
978 std::vector<uint32_t> localIndices;
979 std::vector<float> localVerts;
980 std::vector<uint32_t> tmp; // for output / remap back
981 };
982 Scratch s;
983
984 // 元インデックスを取り出し
985 s.srcIndices.assign(inIndexPool.begin() + r.indexOffset,
986 inIndexPool.begin() + r.indexOffset + r.indexCount);
987
988 // 参照頂点だけに縮約
989 s.remap.resize(vertexCount);
990 size_t unique = meshopt_generateVertexRemap(
991 s.remap.data(), s.srcIndices.data(), s.srcIndices.size(),
992 positions, vertexCount, positionStrideBytes);
993
994 // ローカル頂点/インデックスにリマップ
995 s.localVerts.resize(unique * positionStrideBytes);
996 meshopt_remapVertexBuffer(s.localVerts.data(), positions, vertexCount,
997 positionStrideBytes, s.remap.data());
998
999 s.localIndices.resize(s.srcIndices.size());
1000 meshopt_remapIndexBuffer(s.localIndices.data(), s.srcIndices.data(),
1001 s.srcIndices.size(), s.remap.data());
1002
1003 // 逆リマップ表 (local -> global)
1004 s.invRemap.resize(unique);
1005 for (uint32_t g = 0; g < (uint32_t)vertexCount; ++g) {
1006 uint32_t l = s.remap[g];
1007 if (l != ~0u && l < unique) s.invRemap[l] = g;
1008 }
1009
1010 LocalOut local;
1011 local.indices.reserve(r.indexCount); // 少なくとも LOD0
1012
1013 auto appendLodLocal = [&](const uint32_t* idx, size_t count) {
1014 ClusterLodRange lr{ (uint32_t)local.indices.size(), (uint32_t)count };
1015 local.indices.insert(local.indices.end(), idx, idx + count);
1016 local.ranges.push_back(lr);
1017 };
1018
1019 // LOD0: ローカル→グローバルへ戻して格納
1020 s.tmp.resize(s.localIndices.size());
1021 for (size_t i = 0; i < s.localIndices.size(); ++i)
1022 s.tmp[i] = s.invRemap[s.localIndices[i]];
1023 appendLodLocal(s.tmp.data(), s.tmp.size());
1024
1025 const uint32_t triCount0 = r.indexCount / 3u;
1026 float error = r.bounds.extent().length() * 0.05f;
1027
1028 size_t prevWritten = s.localIndices.size();
1029
1030 for (size_t li = 1; li < levels; ++li) {
1031 const float scale = lodTargets[li];
1032 const uint32_t targetTris = (uint32_t)(std::max)(1.0f, std::floor(triCount0 * scale));
1033 const size_t targetIdx = size_t(targetTris) * 3;
1034
1035 s.tmp.resize(s.localIndices.size());
1036 size_t written = meshopt_simplify(
1037 s.tmp.data(), s.localIndices.data(), s.localIndices.size(),
1038 s.localVerts.data(), unique, positionStrideBytes,
1039 targetIdx, error, meshopt_SimplifyLockBorder);
1040
1041 if (written < 3 || written == prevWritten) break;
1042
1043 meshopt_optimizeVertexCache(s.tmp.data(), s.tmp.data(), written, unique);
1044 prevWritten = written;
1045

1047 for (size_t i = 0; i < written; ++i)

1054 locals[cid] = std::move(local);

1060 outIndexPool.reserve(totalIdx);
1061 outLodRanges.reserve(totalRanges);

1065 outLodBase[cid] = (uint32_t)outLodRanges.size();

1071 lr.offset += base; // ローカル→グローバル

1075 }

1080

1233 }
1234

1283 }

1293 sd.Texture2DArray.FirstArraySlice = 0;
1294 sd.Texture2DArray.ArraySize = ad.ArraySize;
1295
1296 ComPtr<ID3D11ShaderResourceView> srv;
1297 hr = dev->CreateShaderResourceView(arrayTex.Get(), &sd, &srv);
1298 if (FAILED(hr)) return false;

1302 }

1568 auto FloorLog2 = [](UINT v)->UINT { UINT n = 0; while (v > 1) { v >>= 1; ++n; } return n; };

1570 // 非圧縮用: 1 + floor(log2(max(w,h)))
1571 UINT mw = (w > h) ? w : h; return 1u + FloorLog2(mw);

1575 if ((w < 4) || (h < 4)) return 0; // そもそも不可
1576 if ((w % 4) != 0 || (h % 4) != 0) return 0; // タイルが4の倍数でない

1578 // 1 + min(floor(log2(w/4)), floor(log2(h/4)))
1579 return 1u + ((FloorLog2(mw) < FloorLog2(mh)) ? FloorLog2(mw) : FloorLog2(mh));
1580 };

1614 if (!dst) continue;

1636 0, 0, 0,

1639 );

1641 }
