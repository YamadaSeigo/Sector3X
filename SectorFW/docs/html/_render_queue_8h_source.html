203 // ====== 比較ソート用：間接ソート + 一括適用 ======
204 template<class VecT>
205 void IndirectSortStd(VecT& cmds) {
206 const size_t N = cmds.size();
207 if (N <= 1) [[unlikely]] return;
208
209 EnsureScratch(N);
210 if (indexBuf.size() < N) indexBuf.resize(N);
211 if (keysBuf.size() < N) keysBuf.resize(N);
212
213 for (uint32_t i = 0; i < N; ++i) {
214 indexBuf[i] = i;
215 keysBuf[i] = cmds[i].sortKey;
216 }
217
218 if (N >= 32768) {
219 constexpr int TOP_BITS = 12;
220 constexpr uint32_t B = 1u << TOP_BITS; // 4096
221
222 // ここは「固定サイズ」なので thread_local の配列で確保ゼロに
223 thread_local std::array<uint32_t, B> count{};
224 thread_local std::array<uint32_t, B> offset{};
225
226 // クリア
227 count.fill(0);
228 offset.fill(0);
229
230 auto bucketId = [](uint64_t k) noexcept {
231 return uint32_t(k >> (64 - TOP_BITS));
232 };
233
234 for (size_t i = 0; i < N; ++i) ++count[bucketId(keysBuf[indexBuf[i]])];
235 for (uint32_t b = 1; b < B; ++b) offset[b] = offset[b - 1] + count[b - 1];
236
237 // ← ここで pmr と同型の一時バッファを使用
238 auto& tmpIdx = tmpIdxBuf;
239 tmpIdx.resize(N);
240
241 for (size_t i = 0; i < N; ++i)
242 tmpIdx[offset[bucketId(keysBuf[indexBuf[i]])]++] = indexBuf[i];
243
244 indexBuf.swap(tmpIdx); // 同じ IndexVec 同士なので OK
245
246 // バケット内を keys で比較ソート
247 uint32_t start = 0;
248 for (uint32_t b = 0; b < B; ++b) {
249 uint32_t len = count[b];
250 if (len > 1) {
251 std::sort(indexBuf.begin() + start, indexBuf.begin() + start + len,
252 [k = keysBuf.data()](uint32_t a, uint32_t b) noexcept { return k[a] < k[b]; });
253 }
254 start += len;
255 }
256 }
257 else {
258 std::sort(indexBuf.begin(), indexBuf.begin() + N,
259 [k = keysBuf.data()](uint32_t a, uint32_t b) noexcept { return k[a] < k[b]; });
260 }
261
262 ApplyPermutationInPlace(cmds, indexBuf);
263 }

265 template<class VecTLocal, class IndexContainer>
266 void ApplyPermutationInPlace(VecTLocal& cmds, IndexContainer& idx) {
267 const size_t N = cmds.size();
268 visited_.assign(N, 0);
269 uint8_t* visited = visited_.data();
270
271 for (size_t i = 0; i < N; ++i) {
272 if (visited[i] || idx[i] == i) continue;
273 size_t cur = i;
274 auto tmp = std::move(cmds[cur]);
275 while (!visited[cur]) {
276 visited[cur] = 1;
277 size_t nxt = idx[cur];
278 if (nxt == i) { cmds[cur] = std::move(tmp); break; }
279 cmds[cur] = std::move(cmds[nxt]);
280 cur = nxt;
281 }
282 }

285 // Single-threaded Radix Sort
286 template<class VecT>
287 void RadixSortSingle(VecT& cmds, VecT& temp) {
288 constexpr int BITS = 8;
289 constexpr int BUCKETS = 1 << BITS;
290 constexpr int PASSES = 64 / BITS;
291 for (int pass = 0; pass < PASSES; ++pass) {
292 int shift = pass * BITS;
293 std::array<size_t, BUCKETS> count = {};
294 for (auto& cmd : cmds)
295 ++count[(cmd.sortKey >> shift) & (BUCKETS - 1)];
296
297 std::array<size_t, BUCKETS> offset = {};
298 size_t sum = 0;
299 for (int i = 0; i < BUCKETS; ++i) {
300 offset[i] = sum;
301 sum += count[i];
302 }
303
304 for (auto& cmd : cmds) {
305 size_t idx = (cmd.sortKey >> shift) & (BUCKETS - 1);
306 temp[offset[idx]++] = std::move(cmd);
307 }
308
309 cmds.swap(temp);
310 }
311 }

313 // Multi-threaded Radix Sort
314 template<class VecT>
315 void RadixSortMulti(VecT& cmds, VecT& temp, int threadCount) {
316 static constexpr int BITS = RADIX_BITS;
317 static constexpr int BUCKETS = RADIX_BUCKETS;
318 static constexpr int PASSES = 64 / BITS;
319
320 const size_t N = cmds.size();
321 VecT* in = &cmds;
322 VecT* out = &temp;
323
324 // --- 再確保ゼロのプール利用 ---
325 // histPool/offPool は呼び出し側 EnsureScratch により容量確保済み想定
326 auto& histPoolRef = const_cast<SortContext*>(this)->histPool;
327 auto& offPoolRef = const_cast<SortContext*>(this)->offPool;
328
329 for (int pass = 0; pass < PASSES; ++pass) {
330 int shift = pass * BITS;
331 size_t chunkSize = (N + threadCount - 1) / threadCount;
332
333 // Histogram phase
334 std::vector<std::thread> workers;
335 for (int t = 0; t < threadCount; ++t) {
336 size_t start = t * chunkSize;
337 size_t end = (std::min)(start + chunkSize, N);
338 workers.emplace_back([=, &in, &histPoolRef]() {
339 size_t* hist = &histPoolRef[t * BUCKETS];
340 std::fill_n(hist, BUCKETS, size_t(0));
341 for (size_t i = start; i < end; ++i) {
342 uint64_t key = ((*in)[i].sortKey >> shift) & (BUCKETS - 1);
343 ++hist[key];
344 }
345 });
346 }
347 for (auto& w : workers) w.join();
348 workers.clear();
349
350 // Global offset
351 std::array<size_t, BUCKETS> globalOffset = {};
352 size_t running = 0;
353 for (int b = 0; b < BUCKETS; ++b) {
354 globalOffset[b] = running;
355 size_t sum_b = 0;
356 for (int t = 0; t < threadCount; ++t)
357 sum_b += histPoolRef[t * BUCKETS + b];
358 running += sum_b;
359 }
360
361 for (int t = 0; t < threadCount; ++t) {
362 size_t accum = 0;
363 for (int b = 0; b < BUCKETS; ++b) {
364 offPoolRef[t * BUCKETS + b] = globalOffset[b] + accum;
365 accum += histPoolRef[t * BUCKETS + b];
366 }
367 }
368
369 // Scatter phase
370 for (int t = 0; t < threadCount; ++t) {
371 size_t start = t * chunkSize;
372 size_t end = (std::min)(start + chunkSize, N);
373 workers.emplace_back([=, &in, &out, &offPoolRef]() mutable {
374 size_t* localOff = &offPoolRef[t * BUCKETS];
375 for (size_t i = start; i < end; ++i) {
376 uint64_t key = ((*in)[i].sortKey >> shift) & (BUCKETS - 1);
377 (*out)[localOff[key]++] = std::move((*in)[i]);
378 }
379 });
380 }
381 for (auto& w : workers) w.join();
382
383 std::swap(in, out);
384 }
385
386 if (in != &cmds)
387 cmds = std::move(*in);
388 }
389 };
