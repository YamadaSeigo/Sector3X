105 bool Add(const CreateDescType& desc, HandleType& out) {
106 if (auto h = static_cast<Derived*>(this)->FindExisting(desc)) {
107 // 既存を +1
108 AddRef(*h);
109 // もし削除要求が入っていればキャンセル
110 CancelPending((*h).index);
111
112 out = *h;
113 return true;
114 }
115
116 // 新規確保(再利用できる場合はそのハンドルを返す)
117 HandleType h = AllocateHandle();
118
119 // リソースの実際の生成はロック外でやってもよい（重いので）
120 ResourceType res = static_cast<Derived*>(this)->CreateResource(desc, h);
121
122 {
123 std::unique_lock lk(mapMutex);
124 slots[h.index].data = std::move(res);
125 slots[h.index].alive = true;
126 refCount[h.index].store(1, std::memory_order_relaxed);
127 }
128
129 static_cast<Derived*>(this)->RegisterKey(desc, h);
130
131 out = h;
132 return false;
133 }

138 void AddRef(HandleType h) {
139 assert(IsValid(h));
140 refCount[h.index].fetch_add(1, std::memory_order_relaxed);
141 }
142
147 void Release(HandleType h, uint64_t deleteSync = 0) {
148 assert(IsValid(h));
149 uint32_t prev = refCount[h.index].fetch_sub(1, std::memory_order_acq_rel);
150 assert(prev > 0 && "Release underflow");
151 if (prev == 1) EnqueueDelete(h.index, deleteSync);
152 }

158 void EnqueueDelete(uint32_t index, uint64_t deleteSync) {
159 std::lock_guard lk(pendingMutex_);
160 if (auto it = pendingByIndex.find(index); it == pendingByIndex.end()) {
161 pendingDelete.push_back({ index, deleteSync });
162 pendingByIndex[index] = pendingDelete.size() - 1;
163 }
164 else {
165 pendingDelete[it->second].deleteSync = deleteSync; // 後ろへ延ばす
166 }
167 }
168
172 void CancelPending(uint32_t index) {
173 std::lock_guard lk(pendingMutex_);
174 auto it = pendingByIndex.find(index);
175 if (it != pendingByIndex.end()) {
176 const size_t pos = it->second;
177 pendingDelete.erase(pendingDelete.begin() + pos);
178 pendingByIndex.erase(it);
179 // pos 以降の位置がずれるので、必要なら詰め替え最適化を入れてもよい
180 }
181 }

225 void ProcessDeferredDeletes(uint64_t currentFrame) {
226 std::lock_guard lk(pendingMutex_);
227 size_t i = 0;
228 while (i < pendingDelete.size()) {
229 auto& req = pendingDelete[i];
230 if (currentFrame < req.deleteSync) { ++i; continue; }
231
232 const uint32_t idx = req.index;
233 if (refCount[idx].load(std::memory_order_acquire) == 0) {
234 // ここで初めて alive=false。以降は Derived の責務。
235 slots[idx].alive = false;
236
237 // 登録済みキーや名前→handleなどを掃除
238 static_cast<Derived*>(this)->RemoveFromCaches(idx);
239
240 // 実体破棄（子の Release 連鎖などがあれば currentFrame を渡して遅延にも対応）
241 static_cast<Derived*>(this)->DestroyResource(idx, currentFrame);
242
243 // スロット再利用
244 freeList.push_back(idx);
245 }
246 // 破棄済み or キャンセル、どちらでも要求は捨てる
247 pendingByIndex.erase(idx);
248 pendingDelete.erase(pendingDelete.begin() + i);
249 }

316 HandleType AllocateHandle() {
317 uint32_t idx;
318 std::unique_lock lock(mapMutex);
319 if (!freeList.empty()) {
320 idx = freeList.back();
321 freeList.pop_back();
322 slots[idx].generation++;
323 }
324 else {
325 idx = static_cast<uint32_t>(slots.size());
326 slots.push_back({});
327 refCount.push_back(0);
328 }
329 return { idx, slots[idx].generation };
