102 static void NormalizeDesc(D3D11_SAMPLER_DESC& d)
103 {
104 // 既定を軽く穴埋め（必要に応じて調整）
105 if (d.AddressU == 0) d.AddressU = D3D11_TEXTURE_ADDRESS_WRAP;
106 if (d.AddressV == 0) d.AddressV = D3D11_TEXTURE_ADDRESS_WRAP;
107 if (d.AddressW == 0) d.AddressW = D3D11_TEXTURE_ADDRESS_WRAP;
108
109 // 境界色未設定なら 0 で初期化
110 if (d.AddressU == D3D11_TEXTURE_ADDRESS_BORDER ||
111 d.AddressV == D3D11_TEXTURE_ADDRESS_BORDER ||
112 d.AddressW == D3D11_TEXTURE_ADDRESS_BORDER) {
113 // どれかが BORDER なら念のため全部埋める
114 for (int i = 0; i < 4; ++i) d.BorderColor[i] = 0.0f;
115 }
116
117 if (d.ComparisonFunc == D3D11_COMPARISON_NEVER &&
118 (d.Filter == D3D11_FILTER_COMPARISON_MIN_MAG_MIP_LINEAR ||
119 d.Filter == D3D11_FILTER_COMPARISON_MIN_MAG_MIP_POINT ||
120 d.Filter == D3D11_FILTER_COMPARISON_ANISOTROPIC)) {
121 // 比較フィルタなのに ComparisonFunc が未設定っぽい場合、LESS_EQUALに
122 d.ComparisonFunc = D3D11_COMPARISON_LESS_EQUAL;
123 }
124
125 if ((d.Filter == D3D11_FILTER_ANISOTROPIC ||
126 d.Filter == D3D11_FILTER_COMPARISON_ANISOTROPIC) &&
127 d.MaxAnisotropy == 0) {
128 d.MaxAnisotropy = 8;
129 }
130 }

146 SamplerHandle AddWithDesc(const D3D11_SAMPLER_DESC& in)
147 {
148 D3D11_SAMPLER_DESC desc = in;
149 NormalizeDesc(desc);
150
151 // 1) キャッシュヒットなら再利用（+1）
152 {
153 std::scoped_lock lock(cacheMutex);
154 auto it = samplerCache.find(desc);
155 if (it != samplerCache.end()) {
156 AddRef(it->second);
157 return it->second;
158 }
159 }
160
161 // 2) 新規作成（名前をハッシュから生成）
162 const size_t h = HashBufferContent(&desc, sizeof(desc));
163 std::string genName = "samp_" + Hex64(static_cast<uint64_t>(h));
164
165 SamplerHandle handle;
166 Add({ genName, desc }, handle); // Add は ref=1 を返す契約
167
168 // 3) キャッシュ登録（逆引きも）
169 {
170 std::scoped_lock lock(cacheMutex);
171 samplerCache.emplace(desc, handle);
172 handleToDesc.emplace(handle.index, desc);
173 }
174
175 return handle;
176 }
