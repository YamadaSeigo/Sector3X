151 template<size_t R, size_t K, size_t C, typename T>
152 struct MatMulKernel {
153 static Matrix<R, C, T> eval(const Matrix<R, K, T>& A, const Matrix<K, C, T>& B) noexcept {
154 Matrix<R, C, T> out{};
155 for (size_t i = 0; i < R; ++i) {
156 for (size_t j = 0; j < C; ++j) {
157 T s = T(0);
158
159 for (size_t k = 0; k < K; ++k) s += A.m[i][k] * B.m[k][j];
160 out.m[i][j] = s;
161 }
162 }
163 return out;
164 }
165 };

291 // AABB 変換（列ベクトル規約対応）
292 //==============================
293 template<typename T, typename VecT>
294 inline AABB<T, VecT> operator*(const Matrix<4, 4, T>& M, const AABB<T, VecT>& box) noexcept {
295 // ADL 安全な abs（float/double 両対応）
296 using std::abs;
297
298 static_assert(std::is_same_v<VecT, Vec3<T>>, "This overload expects Vec3<T> AABB.");
299
300 const VecT c = (box.lb + box.ub) * T(0.5);
301 const VecT e = (box.ub - box.lb) * T(0.5);
302
303 // c' = R*c + t （右列が t）
304 VecT c2;
305 c2.x = M.m[0][0] * c.x + M.m[0][1] * c.y + M.m[0][2] * c.z + M.m[0][3];
306 c2.y = M.m[1][0] * c.x + M.m[1][1] * c.y + M.m[1][2] * c.z + M.m[1][3];
307 c2.z = M.m[2][0] * c.x + M.m[2][1] * c.y + M.m[2][2] * c.z + M.m[2][3];
308
309 const T ex = abs(e.x), ey = abs(e.y), ez = abs(e.z);
310
311 // e' = |R| * e （列ベクトル規約は「行の絶対値」を使う）
312 VecT e2;
313 e2.x = ex * abs(M.m[0][0]) + ey * abs(M.m[0][1]) + ez * abs(M.m[0][2]);
314 e2.y = ex * abs(M.m[1][0]) + ey * abs(M.m[1][1]) + ez * abs(M.m[1][2]);
315 e2.z = ex * abs(M.m[2][0]) + ey * abs(M.m[2][1]) + ez * abs(M.m[2][2]);
316
317 AABB<T, VecT> out;
318 out.lb = c2 - e2;
319 out.ub = c2 + e2;
320 return out;
321 }

414 // AVX2/FMA: 同一 VP × 複数 W(3x4) → C(4x4)
415 // 2体ずつ処理。端数1体は SSE 版に回します。
416 static inline void Mul4x4x3x4_Batch_To4x4_AVX2_RowCombine(
417 const Matrix4x4f& VP,
418 const Matrix3x4f* __restrict W,
419 Matrix4x4f* __restrict C,
420 std::size_t N) noexcept

428 const __m256 b3_row = _mm256_set_ps(1.f, 0.f, 0.f, 0.f,
429 1.f, 0.f, 0.f, 0.f);
430
431 // VP の各行の係数を __m256（全レーン同値）で保持
432 __m256 a0[4], a1[4], a2[4], a3[4];
433 for (int r = 0; r < 4; ++r) {
434 a0[r] = _mm256_set1_ps(VP.m[r][0]);
435 a1[r] = _mm256_set1_ps(VP.m[r][1]);
436 a2[r] = _mm256_set1_ps(VP.m[r][2]);
437 a3[r] = _mm256_set1_ps(VP.m[r][3]);
438 }
439
440 std::size_t i = 0;
441 for (; i + 1 < N; i += 2) {
442 // 2ワールド分の B の「行」を 256bit にパック
443 const __m128 b0_lo = _mm_loadu_ps(&W[i + 0].m[0][0]); // [r01 r02 r03 t01]
444 const __m128 b0_hi = _mm_loadu_ps(&W[i + 1].m[0][0]);
445 const __m128 b1_lo = _mm_loadu_ps(&W[i + 0].m[1][0]); // [r11 r12 r13 t02]

447 const __m128 b2_lo = _mm_loadu_ps(&W[i + 0].m[2][0]); // [r21 r22 r23 t03]
448 const __m128 b2_hi = _mm_loadu_ps(&W[i + 1].m[2][0]);
449
450 const __m256 b0 = _mm256_set_m128(b0_hi, b0_lo);
451 const __m256 b1 = _mm256_set_m128(b1_hi, b1_lo);
452 const __m256 b2 = _mm256_set_m128(b2_hi, b2_lo);
453
454 // 4 行ぶんまとめて： row_r(C) = a0*b0 + a1*b1 + a2*b2 + a3*[0,0,0,1]
455 for (int r = 0; r < 4; ++r) {
456 __m256 v = _mm256_mul_ps(a0[r], b0);
457#if defined(__FMA__) || (defined(_MSC_VER) && defined(__AVX2__))
458 v = _mm256_fmadd_ps(a1[r], b1, v);
459 v = _mm256_fmadd_ps(a2[r], b2, v);
460 v = _mm256_fmadd_ps(a3[r], b3_row, v);
461#else
462 v = _mm256_add_ps(v, _mm256_mul_ps(a1[r], b1));
463 v = _mm256_add_ps(v, _mm256_mul_ps(a2[r], b2));
464 v = _mm256_add_ps(v, _mm256_mul_ps(a3[r], b3_row));
465#endif

694 __m128& c0, __m128& c1,

847#if defined(__FMA__) || (defined(_MSC_VER) && defined(__AVX2__))
848 __m128 v = _mm_mul_ps(rx, c0);
849 v = _mm_fmadd_ps(ry, c1, v);
850 v = _mm_fmadd_ps(rz, c2, v);
851 v = _mm_fmadd_ps(rw, c3, v);
852 return v;
853#else
854 __m128 v = _mm_add_ps(_mm_mul_ps(rx, c0), _mm_mul_ps(ry, c1));
855 v = _mm_add_ps(v, _mm_mul_ps(rz, c2));
856 v = _mm_add_ps(v, _mm_mul_ps(rw, c3));
857 return v;
858#endif
859 };
860
861 __m128 o0 = rowDotCols(r0, c0, c1, c2, c3);
862 __m128 o1 = rowDotCols(r1, c0, c1, c2, c3);
863 __m128 o2 = rowDotCols(r2, c0, c1, c2, c3);
864
865 // oN = [c0 c1 c2 t] の形で来るので、そのまま row-major 3x4 に格納
866 _mm_storeu_ps(&out.m[0][0], o0);
867 _mm_storeu_ps(&out.m[1][0], o1);
868 _mm_storeu_ps(&out.m[2][0], o2);
869 }

873 // det 近傍 0 の場合は恒等を返す（Release 安全）

875 template<typename T>
876 Matrix<4, 4, T> Inverse(const Matrix<4, 4, T>& M) noexcept {
877 using std::abs;
878
879 Matrix<4, 4, T> inv{};
880 const auto& a = M.m;
881
882 inv[0][0] = a[1][1] * a[2][2] * a[3][3] - a[1][1] * a[2][3] * a[3][2] - a[2][1] * a[1][2] * a[3][3]
883 + a[2][1] * a[1][3] * a[3][2] + a[3][1] * a[1][2] * a[2][3] - a[3][1] * a[1][3] * a[2][2];
884
885 inv[0][1] = -a[0][1] * a[2][2] * a[3][3] + a[0][1] * a[2][3] * a[3][2] + a[2][1] * a[0][2] * a[3][3]
886 - a[2][1] * a[0][3] * a[3][2] - a[3][1] * a[0][2] * a[2][3] + a[3][1] * a[0][3] * a[2][2];
887
888 inv[0][2] = a[0][1] * a[1][2] * a[3][3] - a[0][1] * a[1][3] * a[3][2] - a[1][1] * a[0][2] * a[3][3]
889 + a[1][1] * a[0][3] * a[3][2] + a[3][1] * a[0][2] * a[1][3] - a[3][1] * a[0][3] * a[1][2];
890
891 inv[0][3] = -a[0][1] * a[1][2] * a[2][3] + a[0][1] * a[1][3] * a[2][2] + a[1][1] * a[0][2] * a[2][3]
892 - a[1][1] * a[0][3] * a[2][2] - a[2][1] * a[0][2] * a[1][3] + a[2][1] * a[0][3] * a[1][2];
893
894 inv[1][0] = -a[1][0] * a[2][2] * a[3][3] + a[1][0] * a[2][3] * a[3][2] + a[2][0] * a[1][2] * a[3][3]
895 - a[2][0] * a[1][3] * a[3][2] - a[3][0] * a[1][2] * a[2][3] + a[3][0] * a[1][3] * a[2][2];
896
897 inv[1][1] = a[0][0] * a[2][2] * a[3][3] - a[0][0] * a[2][3] * a[3][2] - a[2][0] * a[0][2] * a[3][3]
898 + a[2][0] * a[0][3] * a[3][2] + a[3][0] * a[0][2] * a[2][3] - a[3][0] * a[0][3] * a[2][2];
899
900 inv[1][2] = -a[0][0] * a[1][2] * a[3][3] + a[0][0] * a[1][3] * a[3][2] + a[1][0] * a[0][2] * a[3][3]
901 - a[1][0] * a[0][3] * a[3][2] - a[3][0] * a[0][2] * a[1][3] + a[3][0] * a[0][3] * a[1][2];
902
903 inv[1][3] = a[0][0] * a[1][2] * a[2][3] - a[0][0] * a[1][3] * a[2][2] - a[1][0] * a[0][2] * a[2][3]
904 + a[1][0] * a[0][3] * a[2][2] + a[2][0] * a[0][2] * a[1][3] - a[2][0] * a[0][3] * a[1][2];
905
906 inv[2][0] = a[1][0] * a[2][1] * a[3][3] - a[1][0] * a[2][3] * a[3][1] - a[2][0] * a[1][1] * a[3][3]
907 + a[2][0] * a[1][3] * a[3][1] + a[3][0] * a[1][1] * a[2][3] - a[3][0] * a[1][3] * a[2][1];
908
909 inv[2][1] = -a[0][0] * a[2][1] * a[3][3] + a[0][0] * a[2][3] * a[3][1] + a[2][0] * a[0][1] * a[3][3]
910 - a[2][0] * a[0][3] * a[3][1] - a[3][0] * a[0][1] * a[2][3] + a[3][0] * a[0][3] * a[2][1];
911
912 inv[2][2] = a[0][0] * a[1][1] * a[3][3] - a[0][0] * a[1][3] * a[3][1] - a[1][0] * a[0][1] * a[3][3]
913 + a[1][0] * a[0][3] * a[3][1] + a[3][0] * a[0][1] * a[1][3] - a[3][0] * a[0][3] * a[1][1];
914
915 inv[2][3] = -a[0][0] * a[1][1] * a[2][3] + a[0][0] * a[1][3] * a[2][1] + a[1][0] * a[0][1] * a[2][3]
916 - a[1][0] * a[0][3] * a[2][1] - a[2][0] * a[0][1] * a[1][3] + a[2][0] * a[0][3] * a[1][1];
917
918 inv[3][0] = -a[1][0] * a[2][1] * a[3][2] + a[1][0] * a[2][2] * a[3][1] + a[2][0] * a[1][1] * a[3][2]
919 - a[2][0] * a[1][2] * a[3][1] - a[3][0] * a[1][1] * a[2][2] + a[3][0] * a[1][2] * a[2][1];
920
921 inv[3][1] = a[0][0] * a[2][1] * a[3][2] - a[0][0] * a[2][2] * a[3][1] - a[2][0] * a[0][1] * a[3][2]
922 + a[2][0] * a[0][2] * a[3][1] + a[3][0] * a[0][1] * a[2][2] - a[3][0] * a[0][2] * a[2][1];
923
924 inv[3][2] = -a[0][0] * a[1][1] * a[3][2] + a[0][0] * a[1][2] * a[3][1] + a[1][0] * a[0][1] * a[3][2]
925 - a[1][0] * a[0][2] * a[3][1] - a[3][0] * a[0][1] * a[1][2] + a[3][0] * a[0][2] * a[1][1];
926
927 inv[3][3] = a[0][0] * a[1][1] * a[2][2] - a[0][0] * a[1][2] * a[2][1] - a[1][0] * a[0][1] * a[2][2]
928 + a[1][0] * a[0][2] * a[2][1] + a[2][0] * a[0][1] * a[1][2] - a[2][0] * a[0][2] * a[1][1];
929
930 T det = a[0][0] * inv[0][0] + a[0][1] * inv[1][0] + a[0][2] * inv[2][0] + a[0][3] * inv[3][0];
931 // 数値安定化: 行列要素のスケールに応じた相対しきい値
932 T s = T(0);
933 for (int i = 0; i < 4; ++i)
934 for (int j = 0; j < 4; ++j)
935 s = (abs(a[i][j]) > s) ? abs(a[i][j]) : s;
936 const T eps = std::numeric_limits<T>::epsilon() * T(100) * (s > T(1) ? s : T(1));
937 if (abs(det) <= eps) {
938 // 非正則：恒等返し（用途に応じて変更可）
939 return Matrix<4, 4, T>::Identity();
940 }
941 const T invDet = T(1) / det;
942 for (int i = 0; i < 4; ++i) for (int j = 0; j < 4; ++j) inv[i][j] *= invDet;
943 return inv;
944 }

949 inline bool IsOrthonormalRotation3x3(const Matrix4x4f& M, float eps = 1e-4f) noexcept {
950 const Vec3<float> r0{ M.m[0][0],M.m[0][1],M.m[0][2] };
951 const Vec3<float> r1{ M.m[1][0],M.m[1][1],M.m[1][2] };
952 const Vec3<float> r2{ M.m[2][0],M.m[2][1],M.m[2][2] };
953 auto near1 = [&](float v) { return std::fabs(v - 1.f) <= eps; };
954 auto near0 = [&](float v) { return std::fabs(v) <= eps; };
955 const bool ortho = near1(r0.dot(r0)) && near1(r1.dot(r1)) && near1(r2.dot(r2)) &&
956 near0(r0.dot(r1)) && near0(r1.dot(r2)) && near0(r2.dot(r0));
957 const float det = r0.dot(r1.cross(r2));
958 return ortho && (std::fabs(det - 1.f) <= 2e-3f); // det≈+1 もチェック
959 }

962 // R が正規直交（R^{-1}=R^T）想定：M=[R t; 0 1] → M^{-1}=[R^T -R^T t; 0 1]
963 inline Matrix4x4f InverseAffineOrthonormal(const Matrix4x4f& M) noexcept {
964 // 行を読み込んで転置→列を得る
965 __m128 r0 = _mm_loadu_ps(M.m[0]);
966 __m128 r1 = _mm_loadu_ps(M.m[1]);
967 __m128 r2 = _mm_loadu_ps(M.m[2]);
968 __m128 r3 = _mm_loadu_ps(M.m[3]);
969 _MM_TRANSPOSE4_PS(r0, r1, r2, r3); // r0=col0=[m00 m10 m20 m30], r3=col3=[tx ty tz 1]
970
971 // R^T は行としては元の行列の上左3x3 転置
972 // col0..2 の xyz をそれぞれ行に置く
973 Matrix4x4f Out = Matrix4x4f::Identity();
974
975 alignas(16) float c0[4], c1[4], c2[4], c3[4];
976 _mm_storeu_ps(c0, r0);
977 _mm_storeu_ps(c1, r1);
978 _mm_storeu_ps(c2, r2);
979 _mm_storeu_ps(c3, r3); // t = (c3[0],c3[1],c3[2])
980
981 // 上左3x3 = R^T
982 Out.m[0][0] = c0[0]; Out.m[0][1] = c1[0]; Out.m[0][2] = c2[0];
983 Out.m[1][0] = c0[1]; Out.m[1][1] = c1[1]; Out.m[1][2] = c2[1];
984 Out.m[2][0] = c0[2]; Out.m[2][1] = c1[2]; Out.m[2][2] = c2[2];
985
986 // t' = -R^T * t
987 const float tx = c3[0], ty = c3[1], tz = c3[2];
988 Out.m[0][3] = -(Out.m[0][0] * tx + Out.m[0][1] * ty + Out.m[0][2] * tz);
989 Out.m[1][3] = -(Out.m[1][0] * tx + Out.m[1][1] * ty + Out.m[1][2] * tz);
990 Out.m[2][3] = -(Out.m[2][0] * tx + Out.m[2][1] * ty + Out.m[2][2] * tz);
991 return Out;
992 }

995 inline Matrix4x4f InverseAffine(const Matrix4x4f& M) noexcept {
996 // 上左3x3 を逆行列、右列の t を -R^{-1}t に
997 const float a00 = M.m[0][0], a01 = M.m[0][1], a02 = M.m[0][2];
998 const float a10 = M.m[1][0], a11 = M.m[1][1], a12 = M.m[1][2];
999 const float a20 = M.m[2][0], a21 = M.m[2][1], a22 = M.m[2][2];
1000
1001 const float det =
1002 a00 * (a11 * a22 - a12 * a21) -
1003 a01 * (a10 * a22 - a12 * a20) +
1004 a02 * (a10 * a21 - a11 * a20);
1005 const float eps = 1e-20f;
1006 if (std::fabs(det) <= eps) {
1007 return Matrix4x4f::Identity();
1008 }
1009 const float invDet = 1.0f / det;
1010
1011 Matrix4x4f Rinv = Matrix4x4f::Identity();
1012 Rinv.m[0][0] = (a11 * a22 - a12 * a21) * invDet;
1013 Rinv.m[0][1] = -(a01 * a22 - a02 * a21) * invDet;
1014 Rinv.m[0][2] = (a01 * a12 - a02 * a11) * invDet;
1015
1016 Rinv.m[1][0] = -(a10 * a22 - a12 * a20) * invDet;
1017 Rinv.m[1][1] = (a00 * a22 - a02 * a20) * invDet;
1018 Rinv.m[1][2] = -(a00 * a12 - a02 * a10) * invDet;
1019
1020 Rinv.m[2][0] = (a10 * a21 - a11 * a20) * invDet;
1021 Rinv.m[2][1] = -(a00 * a21 - a01 * a20) * invDet;
1022 Rinv.m[2][2] = (a00 * a11 - a01 * a10) * invDet;
1023
1024 const float tx = M.m[0][3], ty = M.m[1][3], tz = M.m[2][3];
1025
1026 Matrix4x4f Out = Matrix4x4f::Identity();
1027 for (int r = 0; r < 3; ++r) for (int c = 0; c < 3; ++c) Out.m[r][c] = Rinv.m[r][c];
1028
1029 Out.m[0][3] = -(Rinv.m[0][0] * tx + Rinv.m[0][1] * ty + Rinv.m[0][2] * tz);
1030 Out.m[1][3] = -(Rinv.m[1][0] * tx + Rinv.m[1][1] * ty + Rinv.m[1][2] * tz);
1031 Out.m[2][3] = -(Rinv.m[2][0] * tx + Rinv.m[2][1] * ty + Rinv.m[2][2] * tz);
1032 return Out;
1033 }

1036#if (defined(_MSC_VER) && defined(_M_X64)) || defined(__SSE2__)
1037 if (IsOrthonormalRotation3x3(M)) return InverseAffineOrthonormal(M);
1038#endif
1039 return InverseAffine(M);

1177 m[0][0] = T(1) - T(2) * (yy + zz);
1178 m[0][1] = T(2) * (xy - wz);
1179 m[0][2] = T(2) * (xz + wy);
1180 m[0][3] = T(0);
1181
1182 m[1][0] = T(2) * (xy + wz);
1183 m[1][1] = T(1) - T(2) * (xx + zz);
1184 m[1][2] = T(2) * (yz - wx);
1185 m[1][3] = T(0);
1186
1187 m[2][0] = T(2) * (xz - wy);
1188 m[2][1] = T(2) * (yz + wx);
1189 m[2][2] = T(1) - T(2) * (xx + yy);
1190 m[2][3] = T(0);
1191
1192 m[3][0] = T(0); m[3][1] = T(0); m[3][2] = T(0); m[3][3] = T(1);
1193 return m;

1197 template<typename T>

1199 const Vec3<T>& target,
1200 const Vec3<T>& up) noexcept
1201 {

1215 m[1][0] = y.x; m[1][1] = y.y; m[1][2] = y.z; m[1][3] = -y.dot(eye);
1216 m[2][0] = z.x; m[2][1] = z.y; m[2][2] = z.z; m[2][3] = -z.dot(eye);

1242 //==============================================================
1243 // 透視投影（FOV）— 列ベクトル規約 / row-major / 右列=平行移動
1244 // 注意: w' = z_view になるように m[3][2] を ±1 に置くのが肝
1245 // LH+DX[0,1]: m[3][2] = +1
1246 // RH+DX[0,1]: m[3][2] = -1
1247 // LH/RH+GL[-1,1] でも m[3][2] の ±1 は維持（z 行の係数だけ変化）
1248 //==============================================================
1249 template<Handedness H, ClipZRange Z, typename T>
1250 constexpr Matrix<4, 4, T>
1251 MakePerspectiveFovT(T fovY, T aspect, T nearZ, T farZ) noexcept
1252 {
1253 const T s = T(1) / std::tan(fovY / T(2));
1254 Matrix<4, 4, T> m{}; // 0 初期化
1255 m[0][0] = s / aspect;
1256 m[1][1] = s;
1257 // m[3][3] = 0
1258
1259 if constexpr (H == Handedness::LH) {
1260 // ---- Left-handed (+Z forward)
1261 if constexpr (Z == ClipZRange::ZeroToOne) {
1262 // DX [0,1]
1263 m[2][2] = farZ / (farZ - nearZ);
1264 m[2][3] = (-nearZ * farZ) / (farZ - nearZ);
1265 m[3][2] = T(1); // w' = z_view
1266 }
1267 else {
1268 // GL [-1,1]
1269 m[2][2] = (farZ + nearZ) / (farZ - nearZ);
1270 m[2][3] = (-T(2) * nearZ * farZ) / (farZ - nearZ);
1271 m[3][2] = T(1);
1272 }
1273 }
1274 else {
1275 // ---- Right-handed (−Z forward)
1276 if constexpr (Z == ClipZRange::ZeroToOne) {
1277 // DX [0,1]
1278 m[2][2] = farZ / (nearZ - farZ);
1279 m[2][3] = (nearZ * farZ) / (nearZ - farZ);
1280 m[3][2] = T(-1); // w' = −z_view
1281 }
1282 else {
1283 // GL [-1,1]
1284 m[2][2] = (farZ + nearZ) / (nearZ - farZ);
1285 m[2][3] = (T(2) * nearZ * farZ) / (nearZ - farZ);
1286 m[3][2] = T(-1);
1287 }
1288 }
1289 return m;
1290 }

1292 //==============================================================
1293 // 直交投影 — 列ベクトル規約 / row-major / 右列=平行移動
1294 // ※ オルソは w' = 1 のままなので m[3][2] は 0、m[3][3] = 1
1295 //==============================================================
1296 template<Handedness H, ClipZRange Z, typename T>
1297 constexpr Matrix<4, 4, T>
1298 MakeOrthographicT(T l, T r, T b, T t, T n, T f_) noexcept
1299 {
1300 Matrix<4, 4, T> m{}; // 0 初期化
1301 m[0][0] = T(2) / (r - l);
1302 m[1][1] = T(2) / (t - b);
1303 m[0][3] = -(r + l) / (r - l);
1304 m[1][3] = -(t + b) / (t - b);
1305 m[3][3] = T(1); // w' = 1
1306
1307 if constexpr (H == Handedness::LH) {
1308 if constexpr (Z == ClipZRange::ZeroToOne) {
1309 // DX [0,1]
1310 m[2][2] = T(1) / (f_ - n);
1311 m[2][3] = -n / (f_ - n);
1312 }
1313 else {
1314 // GL [-1,1]
1315 m[2][2] = T(2) / (f_ - n);
1316 m[2][3] = -(f_ + n) / (f_ - n);
1317 }
1318 }
1319 else {
1320 if constexpr (Z == ClipZRange::ZeroToOne) {
1321 // DX [0,1]
1322 m[2][2] = T(-1) / (f_ - n);
1323 m[2][3] = -n / (f_ - n);
1324 }
1325 else {
1326 // GL [-1,1]
1327 m[2][2] = T(-2) / (f_ - n);
1328 m[2][3] = -(f_ + n) / (f_ - n);
1329 }
1330 }
1331 return m;
1332 }

1482 __m128 s = _mm_add_ps(_mm_mul_ps(x, x),
1483 _mm_add_ps(_mm_mul_ps(y, y),
1484 _mm_add_ps(_mm_mul_ps(z, z), _mm_mul_ps(w, w))));
