164 Math::Vec2f ScreenToWorld(const Math::Vec2f& screen) const noexcept {
165 std::shared_lock lock(sharedMutex);
166 // スクリーン→NDC（左上(0,0)、右下(w,h)）
167 float ndcX = (screen.x / virtualWidth) * 2.0f - 1.0f;
168 float ndcY = -(screen.y / virtualHeight) * 2.0f + 1.0f;
169
170 // 逆行列（viewProj の逆）で同次座標をワールドへ
171 Math::Matrix4x4f inv = cameraBufferInv;
172 Math::Vec4f p = inv * Math::Vec4f{ ndcX, ndcY, 0.0f, 1.0f };
173 if (std::abs(p.w) > 1e-6f) { p.x /= p.w; p.y /= p.w; }
174 return { p.x, p.y };
175 }
176
178 Math::Vec2f WorldToScreen(const Math::Vec2f& world) const noexcept {
179 std::shared_lock lock(sharedMutex);
180 Math::Vec4f p = cameraBuffer[currentSlot].viewProj * Math::Vec4f{ world.x, world.y, 0.0f, 1.0f };
181 if (std::abs(p.w) > 1e-6f) { p.x /= p.w; p.y /= p.w; }
182 // NDC→スクリーン
183 float sx = (p.x * 0.5f + 0.5f) * virtualWidth;
184 float sy = (-p.y * 0.5f + 0.5f) * virtualHeight;
185 return { sx, sy };
186 }

189 Math::AABB<float, Math::Vec2f> MakeViewAABB() const noexcept {
190 // スクリーン四隅をワールドに戻す
191 Math::Vec2f tl = ScreenToWorld({ 0.0f, 0.0f });
192 Math::Vec2f br = ScreenToWorld({ virtualWidth, virtualHeight });
193 Math::Vec2f lb = { (std::min)(tl.x, br.x), (std::min)(tl.y, br.y) };
194 Math::Vec2f ub = { (std::max)(tl.x, br.x), (std::max)(tl.y, br.y) };
195 return { lb, ub };
196 }

198 // 行列計算（内部専用）
199 void RecomputeMatrices_NoLock(uint16_t slot) noexcept {
200 // 1) 正射影行列（左上原点のピクセル直感に寄せるため、virtual解像度基準）
201 // 世界→スクリーンを素直にするため、世界単位→ピクセル換算(pixelsPerUnit)とzoomを投影に反映
202 // ここでは「世界座標系」をそのまま使い、viewで中心や回転・ズームを扱う構成にする。
203 // 正射影は -1..+1 に線形写像するだけなので、OffCenterLH を使う。
204 //
205 // 可視幅/高さ（世界単位）: virtualWidth/Height を PPU と zoom で割り戻す
206 const float worldW = (virtualWidth / pixelsPerUnit) / zoom;
207 const float worldH = (virtualHeight / pixelsPerUnit) / zoom;
208
209 // 中心 center を基準に「左上/右下」の世界座標を決定
210 const float left = center.x - worldW * 0.5f;
211 const float right = center.x + worldW * 0.5f;
212 const float top = center.y + worldH * 0.5f;
213 const float bottom = center.y - worldH * 0.5f;
214
215 cameraBuffer[slot].proj = Math::MakeOrthographicT<Math::Handedness::LH, Math::ClipZRange::ZeroToOne>(
216 left, right, bottom, top, nearClip, farClip);
217
218 // 2) View 行列
219 // 2Dの一般的なカメラは T(-center) → Rz(-rotZ) で十分
220 Math::Matrix4x4f T = Math::MakeTranslationMatrix(Math::Vec3f{ -center.x, -center.y, 0.0f });
221 Math::Matrix4x4f R = Math::MakeRotationMatrix(Math::Quatf::FromEuler(0.0f, 0.0f, -rotZ));
222
223 cameraBuffer[slot].view = T * R;
224
225 cameraBuffer[slot].viewProj = cameraBuffer[slot].proj * cameraBuffer[slot].view;
226
227 // 逆行列（座標変換ユーティリティ用）
228 cameraBufferInv = Math::Inverse(cameraBuffer[slot].viewProj); // 実装がなければ適宜追加
229 }
