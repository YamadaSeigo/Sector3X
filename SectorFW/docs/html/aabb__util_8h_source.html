35 inline AABB3f MakeAABB_FromAoS(const Vec3f* positions, size_t count, size_t strideBytes = sizeof(Vec3f)) noexcept
36 {
37 AABB3f out;
38 if (!positions || count == 0) {
39 out.lb = { 0,0,0 };
40 out.ub = { 0,0,0 };
41 return out;
42 }
43
44 // 初期値
45 float minx = std::numeric_limits<float>::infinity();
46 float miny = std::numeric_limits<float>::infinity();
47 float minz = std::numeric_limits<float>::infinity();
48 float maxx = -std::numeric_limits<float>::infinity();
49 float maxy = -std::numeric_limits<float>::infinity();
50 float maxz = -std::numeric_limits<float>::infinity();
51
52 const unsigned char* base = reinterpret_cast<const unsigned char*>(positions);
53
54#if (defined(_MSC_VER) && defined(_M_X64)) || defined(__SSE2__)
55 // SSE2: 3軸を同時に更新（1頂点/ループ）
56 __m128 vmin = _mm_set1_ps(std::numeric_limits<float>::infinity());
57 __m128 vmax = _mm_set1_ps(-std::numeric_limits<float>::infinity());
58
59 for (size_t i = 0; i < count; ++i) {
60 const Vec3f* p = reinterpret_cast<const Vec3f*>(base + i * strideBytes);
61 // 安全のため各成分はスカラーロードし、w は +inf（min 計算で影響を与えない）にする
62 const float x = SanitizeFinite(p->x);
63 const float y = SanitizeFinite(p->y);
64 const float z = SanitizeFinite(p->z);
65
66 __m128 v = _mm_set_ps(std::numeric_limits<float>::infinity(), z, y, x);
67 vmin = _mm_min_ps(vmin, v);
68 vmax = _mm_max_ps(vmax, v);
69 }
70 // 結果を取り出し
71 alignas(16) float mins[4], maxs[4];
72 _mm_store_ps(mins, vmin);
73 _mm_store_ps(maxs, vmax);
74 minx = (std::min)(minx, mins[0]); maxx = (std::max)(maxx, maxs[0]); // x
75 miny = (std::min)(miny, mins[1]); maxy = (std::max)(maxy, maxs[1]); // y
76 minz = (std::min)(minz, mins[2]); maxz = (std::max)(maxz, maxs[2]); // z
77
78#else
79 // スカラー版
80 for (size_t i = 0; i < count; ++i) {
81 const Vec3f* p = reinterpret_cast<const Vec3f*>(base + i * strideBytes);
82 const float x = SanitizeFinite(p->x);
83 const float y = SanitizeFinite(p->y);
84 const float z = SanitizeFinite(p->z);
85 minx = (std::min)(minx, x); maxx = (std::max)(maxx, x);
86 miny = (std::min)(miny, y); maxy = (std::max)(maxy, y);
87 minz = (std::min)(minz, z); maxz = (std::max)(maxz, z);
88 }
89#endif
90
91 out.lb = { minx, miny, minz };
92 out.ub = { maxx, maxy, maxz };
93 return out;
94 }

323 template<class T>
324 inline AABB<T, Vec3<T>>
325 TransformAABB_Affine(const Matrix<4, 4, T>& M, const AABB<T, Vec3<T>>& box) noexcept
326 {
327 // 中心と半径（extent）
328 const Vec3<T> c = (box.lb + box.ub) * T(0.5);
329 const Vec3<T> e = (box.ub - box.lb) * T(0.5);
330
331 // 列ベクトル規約なので x' = dot(row0.xyz, c) + M[0][3] などで OK
332 const T cx = M[0][0] * c.x + M[0][1] * c.y + M[0][2] * c.z + M[0][3];
333 const T cy = M[1][0] * c.x + M[1][1] * c.y + M[1][2] * c.z + M[1][3];
334 const T cz = M[2][0] * c.x + M[2][1] * c.y + M[2][2] * c.z + M[2][3];
335
336 // extent は線形部の絶対値を使って押し広げる
337 const T ex = std::abs(M[0][0]) * e.x + std::abs(M[0][1]) * e.y + std::abs(M[0][2]) * e.z;
338 const T ey = std::abs(M[1][0]) * e.x + std::abs(M[1][1]) * e.y + std::abs(M[1][2]) * e.z;
339 const T ez = std::abs(M[2][0]) * e.x + std::abs(M[2][1]) * e.y + std::abs(M[2][2]) * e.z;
340
341 AABB<T, Vec3<T>> out;
342 out.lb = { cx - ex, cy - ey, cz - ez };
343 out.ub = { cx + ex, cy + ey, cz + ez };
344 return out;
345 }

347 // 48B（3x4）ワールド行列にもそのまま対応
348 template<class T>
349 inline AABB<T, Vec3<T>>
350 TransformAABB_Affine(const Matrix<3, 4, T>& M, const AABB<T, Vec3<T>>& box) noexcept
351 {
352 const Vec3<T> c = (box.lb + box.ub) * T(0.5);
353 const Vec3<T> e = (box.ub - box.lb) * T(0.5);
354
355 const T cx = M[0][0] * c.x + M[0][1] * c.y + M[0][2] * c.z + M[0][3];
356 const T cy = M[1][0] * c.x + M[1][1] * c.y + M[1][2] * c.z + M[1][3];
357 const T cz = M[2][0] * c.x + M[2][1] * c.y + M[2][2] * c.z + M[2][3];
358
359 const T ex = std::abs(M[0][0]) * e.x + std::abs(M[0][1]) * e.y + std::abs(M[0][2]) * e.z;
360 const T ey = std::abs(M[1][0]) * e.x + std::abs(M[1][1]) * e.y + std::abs(M[1][2]) * e.z;
361 const T ez = std::abs(M[2][0]) * e.x + std::abs(M[2][1]) * e.y + std::abs(M[2][2]) * e.z;
362
363 AABB<T, Vec3<T>> out;
364 out.lb = { cx - ex, cy - ey, cz - ez };
365 out.ub = { cx + ex, cy + ey, cz + ez };
366 return out;
