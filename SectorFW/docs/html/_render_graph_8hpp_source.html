199 void Execute() {
200 //フレームのインクリメントとリソースの破棄処理
201 backend.ProcessDeferredDeletes(++currentFrame);
202
203#ifdef _ENABLE_IMGUI
204 {
205 auto g = Debug::BeginTreeWrite(); // lock & back buffer
206 auto& frame = g.data();
207
208 // 例えばプリオーダ＋depth 指定で平坦化したツリーを詰める
209 frame.items.push_back({ /*id=*/frame.items.size(), /*depth=*/Debug::WorldTreeDepth::TREEDEPTH_RENDERGRAPH, /*leaf=*/false, "RenderGraph" });
210 } // guard のデストラクトで unlock。swap は UI スレッドで。
211#endif // _ENABLE_IMGUI
212
213 uint16_t prevSlot = consumeSlot.exchange((consumeSlot.load(std::memory_order_relaxed) + 1) % RENDER_BUFFER_COUNT, std::memory_order_acq_rel);
214
215 backend.BeginFrameUpload(sharedInstanceArena->Data(prevSlot), sharedInstanceArena->Size(prevSlot));
216 sharedInstanceArena->ResetSlot(prevSlot);
217
218 renderService.CallPreDrawCustomFunc(prevSlot); // カスタム関数の更新
219
220 // グループごとに Submit して cmds を埋める
221 for (size_t gi = 0; gi < groups.size(); ++gi) {
222 auto& g = groups[gi];
223 auto& gs = groupStates[gi];
224
225 // 中身だけ消す。capacity は保持される。
226 gs.cmds.clear();
227 gs.indices.clear();
228 gs.ranges.clear();
229
230 g.queue->Submit(prevSlot, gs.cmds); // ここで1回だけ取り出す
231
232#ifdef _ENABLE_IMGUI
233 {
234 auto guard = Debug::BeginTreeWrite(); // lock & back buffer
235 auto& frame = guard.data();
236
237 // 例えばプリオーダ＋depth 指定で平坦化したツリーを詰める
238 frame.items.push_back({ /*id=*/frame.items.size(), /*depth=*/Debug::WorldTreeDepth::TREEDEPTH_GROUP, /*leaf=*/false, "Group : " + g.name });
239 frame.items.push_back({ /*id=*/frame.items.size(), /*depth=*/Debug::WorldTreeDepth::TREEDEPTH_DRAWCOMMAND, /*leaf=*/true, "DrawCommand : " + std::to_string(gs.cmds.size()) });
240 } // guard のデストラクトで unlock。swap は UI スレッドで。
241#endif // _ENABLE_IMGUI
242
243 const size_t passCount = g.passes.size();
244 gs.ranges.resize(passCount);
245
246 if (gs.cmds.empty() || passCount == 0) continue;
247
248 // --- 1st pass: 各パスがいくつ index を持つかカウント ---
249 std::vector<uint32_t> counts(passCount, 0);
250
251 for (uint32_t ci = 0; ci < (uint32_t)gs.cmds.size(); ++ci) {
252 uint16_t m = gs.cmds[ci].viewMask;
253 while (m > 0) {
254 uint16_t p = std::countr_zero(m);
255
256 m &= (m - 1); // 最下位の1ビットを落とす
257
258 //passIndex がそのままbitならこれでOK
259 counts[p]++; // C++20 (MSVCなら _tzcnt_u32 など)
260 }
261 }
262
263 // --- 2nd: prefix-sum でオフセット決定 ---
264 uint32_t total = 0;
265 for (size_t p = 0; p < passCount; ++p) {
266 gs.ranges[p].offset = total;
267 gs.ranges[p].count = counts[p];
268 total += counts[p];
269 }
270
271 gs.indices.resize(total); // ここだけ 1回の確保
272
273 // カーソル（書き込み位置）
274 std::vector<uint32_t> cursor(passCount);
275 for (size_t p = 0; p < passCount; ++p)
276 cursor[p] = gs.ranges[p].offset;
277
278 // --- 3rd: 実際に index を突っ込む ---
279 for (uint32_t ci = 0; ci < (uint32_t)gs.cmds.size(); ++ci) {
280 uint16_t m = gs.cmds[ci].viewMask;
281 while (m > 0) {
282 uint16_t p = std::countr_zero(m);
283
284 m &= (m - 1); // 最下位の1ビットを落とす
285
286 gs.indices[cursor[p]++] = ci;
287 }
288 }
289 }
290
291 for (const auto& node : executionOrder) {
292 auto& g = groups[node.groupIndex];
293 auto& gs = groupStates[node.groupIndex];
294
295 auto* pass = g.passes[node.passIndex];
296 const auto& r = gs.ranges[node.passIndex];
297
298 if (r.count == 0) {
299 if (!pass->customExecute.empty()) {
300 BindPassGlobalResources(pass);
301
302 for (auto& func : pass->customExecute)
303 func(currentFrame);
304 }
305
306 continue;
307 }
308
309 bool useRasterizer = BindPassGlobalResources(pass);
310
311 // このパス用の indexView
312 auto* idxBegin = gs.indices.data() + r.offset;
313 auto* idxEnd = idxBegin + r.count;
314
315 // 共通cmds + このパスの indexビューで描画
316 backend.ExecuteDrawIndexedInstanced(gs.cmds, std::span<const uint32_t>(idxBegin, idxEnd), pass->psoOverride, !useRasterizer, pass->rebindPSO);
317
318 for (auto& func : pass->customExecute)
319 func(currentFrame);
320 }
321 }

354 struct PassRuntime {
355 struct TrackingResource : std::pmr::memory_resource {
356 std::pmr::memory_resource* upstream{};
357 size_t used = 0;
358 explicit TrackingResource(std::pmr::memory_resource* up = std::pmr::get_default_resource()) : upstream(up) {}
359 private:
360 void* do_allocate(size_t bytes, size_t align) override {
361 used += ((bytes + align - 1) / align) * align;
362 return upstream->allocate(bytes, align);
363 }
364 void do_deallocate(void* p, size_t bytes, size_t align) override {
365 upstream->deallocate(p, bytes, align);
366 }
367 bool do_is_equal(const std::pmr::memory_resource& other) const noexcept override {
368 return this == &other;
369 }
370 };

372 static constexpr size_t kStackBuf = 256 * 1024;
373 struct PerFlight {
374 // 初期バッファ（stack か heap のどちらかを使用）
375 alignas(64) std::byte stack[kStackBuf] = {};
376 std::unique_ptr<std::byte[]> heap{};
377 size_t heapSize{ 0 };
378 TrackingResource tracker{ std::pmr::get_default_resource() };
379 // arena は「初期バッファ」を参照する。基本は再構築せず release() でリセット
380 std::pmr::monotonic_buffer_resource* arena{ nullptr };
381 // cmds は arena 常駐の pmr::vector
382 std::pmr::vector<DrawCommand> cmds{ std::pmr::get_default_resource() };
383
384 void init(size_t hint) {
385 void* initialBuf = stack;
386 size_t initialSize = kStackBuf;
387 if (hint > kStackBuf) {
388 heapSize = hint;
389 heap = std::make_unique<std::byte[]>(heapSize);
390 initialBuf = heap.get();
391 initialSize = heapSize;
392 }
393 // arena を配置new（以後は release でリセット）
394 arenaStorage = std::make_unique<std::pmr::monotonic_buffer_resource>(initialBuf, initialSize, &tracker);
395 arena = arenaStorage.get();
396 cmds = std::pmr::vector<DrawCommand>{ arena };
397 tracker.used = 0;
398 }
399
400 void release() {
401 tracker.used = 0;
402 arena->release(); // 実メモリは保持。次の allocate はO(1)
403 cmds.clear();
404 }
405
406 void maybe_grow(size_t newHint) {
407 if (newHint <= kStackBuf || newHint <= heapSize) return;
408 heapSize = newHint;
409 heap = std::make_unique<std::byte[]>(heapSize);
410 // arena を作り直す（成長時だけ・まれ）
411 arenaStorage = std::make_unique<std::pmr::monotonic_buffer_resource>(heap.get(), heapSize, &tracker);
412 arena = arenaStorage.get();
413 cmds = std::pmr::vector<DrawCommand>{ arena };
414 tracker.used = 0;
415 }
416 private:
417 std::unique_ptr<std::pmr::monotonic_buffer_resource> arenaStorage{};
418 };
419
420 std::string name;
421 size_t hint{ 0 };
422 std::vector<PerFlight> perFlight = std::vector<PerFlight>(kFlights);
423
424 bool needs_grow() const {
425 for (uint32_t i = 0; i < kFlights; ++i) {
426 // 直近 used が hint に近い or 超えたフライトがあるなら拡張候補
427 if (perFlight[i].tracker.used > hint) return true;
428 }
429 return false;
430 }
