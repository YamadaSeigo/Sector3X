27 public:
28 //RenderThraedのためがスレッドを1つ使っているので-1する
29 explicit SimpleThreadPool(uint32_t n = std::thread::hardware_concurrency() - 1)
30 : stop_(false)
31 , busy_(0)
32 {
33 if (n == 0) n = 1;
34
35 LOG_INFO("SimpleThreadPoolService: starting with {%d} threads", n);
36 threadCount = n;
37 workers_.reserve(n);
38 for (uint32_t i = 0; i < n; ++i) {
39 workers_.emplace_back([this] {
40 tls_inPool_ = true;
41 for (;;) {
42 std::function<void()> job;
43 {
44 std::unique_lock lk(m_);
45 cv_.wait(lk, [&] { return stop_ || !q_.empty(); });
46 if (stop_ && q_.empty())
47 return;
48 job = std::move(q_.front());
49 q_.pop();
50 }
51
52 busy_.fetch_add(1, std::memory_order_relaxed);
53 ++tls_depth_;
54 job();
55 --tls_depth_;
56 busy_.fetch_sub(1, std::memory_order_relaxed);
57 }
58 });
59 }
60 }

62 ~SimpleThreadPool() override {
63 {
64 std::lock_guard lk(m_);
65 stop_ = true;
66 }
67 cv_.notify_all();
68 for (auto& w : workers_)
69 w.join();
70 }

74 // プール内からのネストした Submit で、
75 // かつ全ワーカーがビジーなら、その場で実行
76 if (tls_inPool_

82 return;
83 }

85 {
86 std::lock_guard lk(m_);
87 q_.push(std::move(job));
88 }

90 }
