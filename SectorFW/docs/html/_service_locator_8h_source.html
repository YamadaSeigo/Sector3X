113 void UnregisterDynamicService() {
114 static_assert(!T::isStatic, "Cannot unregister static service.");
115
116 std::unique_lock<std::shared_mutex> lock(*mapMutex);
117 auto iter = services.find(typeid(T));
118
119 if (iter != services.end())
120 {
121 if constexpr (isUpdateService<T>) {
122 auto index = iter->second.updateIndex;
123 if (updateServices.empty() || index >= updateServices.size()) {
124 SFW_ASSERT(false && "Invalid update service index.");
125 return;
126 }
127
128 auto updateService = updateServices[index];
129 auto it = services.find(updateServices.back()->typeIndex);
130 if (it != services.end()) {
131 it->second.updateIndex = index;
132 std::swap(updateServices[index], updateServices.back());
133 updateServices.pop_back();
134 }
135 else {
136 SFW_ASSERT(false && "Update service not found in the update services list.");
137 }
138 }
139 if constexpr (std::is_base_of_v<ICommitService, T>) {
140 auto commitService = static_cast<ICommitService*>(iter->second.servicePtr);
141 auto it = std::find(commitServices.begin(), commitServices.end(), commitService);
142 if (it != commitServices.end()) {
143 commitServices.erase(it);
144 }
145 else {
146 SFW_ASSERT(false && "Commit service not found in the commit services list.");
147 }
148 }
149 services.erase(iter);
150 }
151
152 RebuildPlan_NeedLock();

175 void UpdateService(double dt, IThreadExecutor* executor) {
176 // ロック不要：不変 plan_ を読むだけ
177 auto p = plan_; // shared_ptr のコピーは lock-free
178 if (!p) {
179 // 初回だけ保険（通常は初期化で構築済み）
180 std::shared_lock<std::shared_mutex> lk(*mapMutex);
181 // Executor 取得や束ね直しは行わず、直列フォールバックでも良いが
182 // プラン未構築なら何もしないか、旧直列ループに退避してもOK
183 for (auto* s : updateServices) if (s) s->PreUpdate(dt); // フォールバック
184 return;
185 }
186 for (const auto& phase : plan_->phases) {
187 // group>=1のグループを並列で更新
188 ThreadCountDownLatch latch((int)phase.parallelGroups.size());
189 for (auto& g : phase.parallelGroups)
190 {
191 executor->Submit([&g, &latch, dt]() {
192 // グループ内は order 順に直列
193 for (IUpdateService* s : g.serial) {
194 s->PreUpdate(dt);
195 }
196 latch.CountDown();
197 });
198 }
199
200 // group==0 の“直列レーン”をメインスレッドで順に実行
201 for (auto* s : phase.serialLane) s->PreUpdate(dt);
202
203 //並列更新待ち
204 latch.Wait();
205 }
206 }

323 void RebuildPlan_NeedLock() {
324 // 登録済み UpdateEntry をフェーズ→グループ→order で束ね直す
325 // ここは、前回案の通り UpdateEntry ベクタを持っている前提
326 std::vector<UpdateEntry> entries;
327 entries.reserve(updateServices.size());
328 for (auto* s : updateServices) {
329 if (!s) continue;
330 UpdateEntry e{};
331 e.ptr = s;
332 e.typeIndex = s->typeIndex; // 既存コードが設定済み :contentReference[oaicite:4]{index=4}
333 e.phase = s->phase;
334 e.group = s->group;
335 e.order = s->order;
336 entries.push_back(e);
337 }
338 uint16_t minPhase = UINT16_MAX, maxPhase = 0;
339 for (auto& e : entries) { minPhase = (std::min)(minPhase, e.phase); maxPhase = (std::max)(maxPhase, e.phase); }
340 if (entries.empty()) { return; }
341
342 plan_->phases.resize(size_t(maxPhase - minPhase + 1));
343 // group id は密でない可能性あり → まず grouping
344 std::unordered_map<uint16_t, std::vector<UpdateEntry>> tmpGroups;
345
346 for (uint16_t ph = minPhase; ph <= maxPhase; ++ph) {
347 // フェーズ内：group==0 は“直列レーン”、group>=1 は“並列グループ”
348 std::unordered_map<uint16_t, std::vector<UpdateEntry>> groups;
349 std::vector<UpdateEntry> lane0;
350 for (auto& e : entries) if (e.phase == ph) {
351 if (e.group == 0) lane0.push_back(e);
352 else groups[e.group].push_back(e);
353 }
354
355 auto& phase = plan_->phases[size_t(ph - minPhase)];
356
357 // lane0 は order 昇順で直列実行
358 std::sort(lane0.begin(), lane0.end(), [](auto& a, auto& b) { return a.order < b.order; });
359 phase.serialLane.clear();
360 phase.serialLane.reserve(lane0.size());
361 for (auto& e : lane0) phase.serialLane.push_back(e.ptr);
362
363 // group>=1 は各グループ内を order ソート → parallelGroups に入れる
364 phase.parallelGroups.clear();
365 phase.parallelGroups.reserve(groups.size());
366 for (auto& kv : groups) {
367 auto& vec = kv.second;
368 std::sort(vec.begin(), vec.end(), [](auto& a, auto& b) { return a.order < b.order; });
369 ExecPlan::GroupPlan gp;
370 gp.serial.reserve(vec.size());
371 for (auto& e : vec) gp.serial.push_back(e.ptr);
372 phase.parallelGroups.push_back(std::move(gp));
373 }
374 }
375 }
