35 inline int DistanceBin(float d2, float near2, float far2) noexcept {
36 float t = (d2 - near2) / (std::max)(far2 - near2, 1e-12f);
37 int b = (int)(t * B);
38 return (b < 0 ? 0 : (b >= B ? B - 1 : b));
39 }

41 // dx^2 + dy^2 + dz^2（カメラ位置からの距離^2）
42 inline float Dist2(const SoAPositions& T, uint32_t i,
43 float cx, float cy, float cz) noexcept {
44 float dx = T.x[i] - cx;
45 float dy = T.y[i] - cy;
46 float dz = T.z[i] - cz;
47 return dx * dx + dy * dy + dz * dz;

50 // 近→遠の近似順で order を構築（安定）
51 template<int B = 32>
52 void BuildNearToFar_Order_Buckets(const SoAPositions& T,
53 float cx, float cy, float cz,
54 float near2, float far2,
55 std::vector<uint32_t>& order)
56 {
57 const uint32_t N = T.N;
58 order.resize(N);
59
60 // 1) 各ビンの個数カウント
61 uint32_t count[B] = {}; // 0 init
62 std::vector<uint8_t> bins(N); // i番目がどのビンか
63
64 for (uint32_t i = 0; i < N; ++i) {
65 float d2 = Dist2(T, i, cx, cy, cz);
66 int b = DistanceBin<B>(d2, near2, far2);
67 bins[i] = (uint8_t)b;
68 ++count[b];
69 }
70
71 // 2) prefix sum → 書き込み開始位置
72 uint32_t offset[B];
73 uint32_t sum = 0;
74 for (int b = 0; b < B; ++b) {
75 offset[b] = sum;
76 sum += count[b];
77 }
78
79 // 3) 安定に連結
80 for (uint32_t i = 0; i < N; ++i) {
81 int b = bins[i];
82 order[offset[b]++] = i;
83 }
84 }

86 // d2 を [near2, far2] → [0, 65535] に量子化（固定小数キー）
87 inline uint16_t QuantizeD2(float d2, float near2, float far2) noexcept {
88 float t = (d2 - near2) / (std::max)(far2 - near2, 1e-12f);
89 int v = (int)std::lrintf(t * 65535.0f);
90 if (v < 0) v = 0; else if (v > 65535) v = 65535;
91 return (uint16_t)v;
92 }

94 // 16bitキーを使った安定な近→遠ソート（Radix16）。ビン分割よりも精細な順序が得られる。安定。
95 static void BuildOrder_FixedRadix16(const SoAPositions& T,
96 float cx, float cy, float cz,
97 float near2, float far2,
98 std::vector<uint32_t>& order)
99 {
100 const uint32_t N = T.N;
101 order.resize(N);
102 std::vector<uint16_t> key(N);
103
104#if SFW_HAS_AVX2
105 const __m256 vxC = _mm256_set1_ps(cx);
106 const __m256 vyC = _mm256_set1_ps(cy);
107 const __m256 vzC = _mm256_set1_ps(cz);
108 const __m256 vNear2 = _mm256_set1_ps(near2);
109 const __m256 vScale = _mm256_set1_ps(65535.0f / (std::max)(far2 - near2, 1e-12f));
110
111 uint32_t i = 0;
112 for (; i + 8 <= N; i += 8) {
113 __m256 vx = _mm256_loadu_ps(T.x + i);
114 __m256 vy = _mm256_loadu_ps(T.y + i);
115 __m256 vz = _mm256_loadu_ps(T.z + i);
116
117 __m256 dx = _mm256_sub_ps(vx, vxC);
118 __m256 dy = _mm256_sub_ps(vy, vyC);
119 __m256 dz = _mm256_sub_ps(vz, vzC);
120
121 __m256 d2 = _mm256_fmadd_ps(dx, dx,
122 _mm256_fmadd_ps(dy, dy, _mm256_mul_ps(dz, dz)));
123
124 // t = (d2 - near2) * scale
125 __m256 t = _mm256_mul_ps(_mm256_sub_ps(d2, vNear2), vScale);
126
127 // clamp [0,65535]
128 __m256 zero = _mm256_set1_ps(0.0f);
129 __m256 maxv = _mm256_set1_ps(65535.0f);
130 t = _mm256_max_ps(zero, _mm256_min_ps(t, maxv));
131
132 // 量子化（丸め）
133 __m256i qi = _mm256_cvtps_epi32(t); // 32bit int
134 alignas(32) uint32_t tmp[8];
135 _mm256_store_si256((__m256i*)tmp, qi);
136 for (int k = 0; k < 8; ++k) key[i + k] = (uint16_t)tmp[k];
137 }
138 for (; i < N; ++i) {
139 key[i] = QuantizeD2(Dist2(T, i, cx, cy, cz), near2, far2);
140 }
141#else
142 for (uint32_t i = 0; i < N; ++i) {
143 key[i] = QuantizeD2(Dist2(T, i, cx, cy, cz), near2, far2);
144 }
145#endif
146
147 // ---- 16bit key の2パス計数/Radix（安定） ----
148 // LSB 8bit → MSB 8bit
149 std::vector<uint32_t> tmpOrder(N);
150 // 初期順（安定性維持のため）
151 for (uint32_t i = 0; i < N; ++i) tmpOrder[i] = i;
152
153 auto pass = [&](int shift, const std::vector<uint32_t>& src, std::vector<uint32_t>& dst) {
154 uint32_t cnt[256] = {};
155 for (uint32_t i = 0; i < N; ++i) ++cnt[(key[src[i]] >> shift) & 0xFF];
156 uint32_t ofs[256]; uint32_t sum = 0;
157 for (int b = 0; b < 256; ++b) { ofs[b] = sum; sum += cnt[b]; }
158 for (uint32_t i = 0; i < N; ++i) {
159 uint16_t k = key[src[i]];
160 uint32_t b = (k >> shift) & 0xFF;
161 dst[ofs[b]++] = src[i];
162 }
163 };
164
165 pass(0, tmpOrder, order); // LSB
166 pass(8, order, tmpOrder); // MSB
167 order.swap(tmpOrder); // 最終結果を order に
168 }

172 // 厳密な近→遠ソート（安定）。距離^2をキーにして std::nth_element + sort。
173 static void BuildFrontK_Strict(const SoAPositions& T,
174 float cx, float cy, float cz,
175 uint32_t K,
176 std::vector<uint32_t>& order)
177 {
178 const uint32_t N = T.N;
179 order.resize(N);
180 std::vector<KeyRow> keys(N);
181 for (uint32_t i = 0; i < N; ++i) {
182 keys[i] = { Dist2(T, i, cx, cy, cz), i };
183 }
184 if (K > N) K = N;
185 std::nth_element(keys.begin(), keys.begin() + K, keys.end(),
186 [](auto& a, auto& b) { return a.dist2 < b.dist2; });
187 std::sort(keys.begin(), keys.begin() + K,
188 [](auto& a, auto& b) { return a.dist2 < b.dist2; });
189
190 for (uint32_t i = 0; i < N; ++i) order[i] = keys[i].row;
191 }
