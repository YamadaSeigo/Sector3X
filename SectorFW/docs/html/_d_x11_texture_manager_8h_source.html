25 inline std::string NormalizePath(std::string path) {
26 // 1) すべて '\\' に統一
27 std::replace(path.begin(), path.end(), '/', '\\');
28 // 2) 末尾の '\\' は削除
29 while (!path.empty() && path.back() == '\\') path.pop_back();
30 // 3) 小文字化
31 std::transform(path.begin(), path.end(), path.begin(),
32 [](unsigned char c) { return static_cast<char>(std::tolower(c)); });
33 return path;
34 }

36 inline bool EndsWithI(std::string_view s, std::string_view suf) {
37 if (s.size() < suf.size()) return false;
38 auto it1 = s.end(); auto it2 = suf.end();
39 while (it2 != suf.begin()) {
40 --it1; --it2;
41 if (std::tolower(static_cast<unsigned char>(*it1)) !=
42 std::tolower(static_cast<unsigned char>(*it2))) return false;
43 }
44 return true;
45 }

55 struct TextureKeyHash {
56 size_t operator()(const TextureKey& k) const noexcept {
57 std::hash<std::string> hs; std::hash<bool> hb;
58 size_t h = hs(k.normPath);
59 // 合成（適当なミックス）
60 return (h ^ (hb(k.forceSRGB) + 0x9e3779b97f4a7c15ULL + (h << 6) + (h >> 2)));
