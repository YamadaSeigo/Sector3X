74def generate_key_to_vk_array():
75 lines = []
76 lines.append("constexpr std::array<int, static_cast<size_t>(Key::Count)> KeyToVKMap = [] {")
77 lines.append(" std::array<int, static_cast<size_t>(Key::Count)> map{};")
78 for key in key_enum_order:
79 vk = key_to_vk.get(key, "0") # 未定義は 0（Unknown）
80 lines.append(f" map[static_cast<size_t>(Key::{key})] = {vk};")
81 lines.append(" return map;")
82 lines.append("}();")
83 return "\n".join(lines)
