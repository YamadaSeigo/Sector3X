79 template<class Partition>
80 inline bool MoveIfCrossed(ECS::EntityID id,
81 const Math::Vec3f& newPos,
82 Partition& partition,
83 SpatialChunkRegistry& reg,
84 LevelID level,
85 ChunkHandle& handle,
86 EOutOfBoundsPolicy policy = EOutOfBoundsPolicy::ClampToEdge)
87 {
88 auto dstOpt = partition.GetChunk(newPos, reg, level, policy);
89 if (!dstOpt) return false;
90 SpatialChunk* dst = *dstOpt;
91
92 if constexpr (PartitionTraits<Partition>::stable_ptr) {
93 if (dst == handle.cached) return false;
94 if (handle.cached) {
95 auto& srcEM = handle.cached->GetEntityManager();
96 auto& dstEM = dst->GetEntityManager();
97 if (&srcEM != &dstEM) RelocateEntityBetweenManagers(id, srcEM, dstEM);
98 }
99 handle.cached = dst;
100 handle.key = dst->GetNodeKey();
101 return true;
102 }
103 else {
104 const SpatialChunkKey dstKey = dst->GetNodeKey();
105 if (dstKey.code == handle.key.code && dstKey.level == handle.key.level) {
106 handle.cached = dst;
107 return false;
108 }
109 ECS::EntityManager* srcEM = reg.ResolveOwner(handle.key);
110 ECS::EntityManager* dstEM = reg.ResolveOwner(dstKey);
111 if (srcEM && dstEM && srcEM != dstEM) {
112 RelocateEntityBetweenManagers(id, *srcEM, *dstEM);
113 }
114 handle.key = dstKey;
115 handle.cached = dst;
116 return true;
117 }

166 template<class Partition>
167 void UpdateSpatialAttachment(ECS::EntityID id,
168 const Math::Vec3f& pos,
169 const Math::Vec3f& vel,
170 Partition& partition,
171 SpatialChunkRegistry& reg,
172 LevelID level,
173 CSpatialMotionTag& tag,
174 ECS::EntityManager& floatEM,
175 const SettleRule& rule)
176 {
177 auto dstOpt = partition.GetChunk(pos, reg, level, EOutOfBoundsPolicy::ClampToEdge);
178 const SpatialChunkKey dstKey = (dstOpt ? (*dstOpt)->GetNodeKey() : SpatialChunkKey{});
179 bool moving = (vel.length() > rule.vThreshold);
180
181 if (tag.state == SpatialState::Attached) {
182 if (moving) {
183 auto* srcEM = reg.ResolveOwnerEM(tag.handle.key);
184 if (srcEM && srcEM != &floatEM) RelocateEntityBetweenManagers(id, *srcEM, floatEM);
185 tag.state = SpatialState::Detached;
186 tag.pendingKey = dstKey;
187 tag.stableFrames = 0;
188 }
189 else {
190 MoveIfCrossed(id, pos, partition, reg, level, tag.handle);
191 }
192 }
193 else {
194 tag.pendingKey = dstKey;
195 if (!moving) {
196 if (++tag.stableFrames >= rule.frames && dstKey.code) {
197 auto* dstEM = reg.ResolveOwnerEM(dstKey);
198 if (dstEM && dstEM != &floatEM) {
199 RelocateEntityBetweenManagers(id, floatEM, *dstEM);
200 tag.handle.key = dstKey;
201 tag.handle.cached = (dstOpt ? *dstOpt : nullptr);
202 tag.state = SpatialState::Attached;
203 }
204 }
205 }
206 else {
207 tag.stableFrames = 0;
208 }
209 }

317 // そのフレームの上限 budget 件まで処理し、残りは次フレームへ持ち越す
318 size_t Flush(const SpatialChunkRegistry& reg, size_t budget) {
319 // 共有キューから先頭 budget 件を取り出す（臨界区間は最小限）
320 if (budget == 0) return 0;
321
322 temp_.clear();
323 {
324 std::lock_guard<std::mutex> lock(mtx_);
325 if (queue_.empty()) return 0;
326 const size_t n = (std::min)(queue_.size(), budget);
327 temp_.assign(queue_.begin(), queue_.begin() + n);
328 queue_.erase(queue_.begin(), queue_.begin() + n);
329 }

331 // ここからはロック無しで処理
332 struct EMKeyPair { ECS::EntityManager* src; ECS::EntityManager* dst; };
333 struct PairHash { size_t operator()(const EMKeyPair& p) const noexcept { return (reinterpret_cast<uintptr_t>(p.src) >> 3) ^ (reinterpret_cast<uintptr_t>(p.dst) << 1); } };
334 struct PairEq { bool operator()(const EMKeyPair& a, const EMKeyPair& b) const noexcept { return a.src == b.src && a.dst == b.dst; } };
335
336 std::unordered_map<EMKeyPair, std::vector<ECS::EntityID>, PairHash, PairEq> buckets;
337 buckets.reserve(temp_.size());
338
339 for (const PendingMove& pm : temp_) {
340 ECS::EntityManager* srcEM = reg.ResolveOwnerEM(pm.srcKey);
341 ECS::EntityManager* dstEM = reg.ResolveOwnerEM(pm.dstKey);
342 if (!srcEM || !dstEM || srcEM == dstEM) continue;
343 buckets[{srcEM, dstEM}].push_back(pm.id);
344 }
345
346 size_t moved = 0;
347 for (auto& [pair, ids] : buckets) {
348 if (ids.empty()) continue;
349 ECS::EntityManager& src = *pair.src;
350 ECS::EntityManager& dst = *pair.dst;
351
352 for (ECS::EntityID id : ids) {
353 bool ok = ECS::EntityManager::InsertWithID_ForManagerMove(id, src, dst);
354 if (ok) ++moved;
355 }
356 src.MoveSparseIDsTo(dst, ids);
357 }
358 temp_.clear();
359 return moved;
360 }
361
362 // キューを全破棄
363 void Clear() { queue_.clear(); temp_.clear(); }
364 size_t Size() const { return queue_.size(); }
365 };

372 template<class Partition>
373 inline bool MoveIfCrossed_Deferred(ECS::EntityID id,
374 const Math::Vec3f& newPos,
375 Partition& partition,
376 SpatialChunkRegistry& reg,
377 LevelID level,
378 ChunkHandle& handle,
379 BudgetMover::LocalBatch& moverBatch,
380 EOutOfBoundsPolicy policy = EOutOfBoundsPolicy::ClampToEdge)
381 {
382 std::optional<SpatialChunk*> dstOpt = partition.GetChunk(newPos, reg, level, policy);
383 if (!dstOpt) return false;
384 SpatialChunk* dst = *dstOpt;
385
386 if constexpr (PartitionTraits<Partition>::stable_ptr) {
387 if (dst == handle.cached) return false;
388 // 旧所属があれば、(srcKey,dstKey) を積む
389 if (handle.cached) {
390 moverBatch.Add(id, handle.key, dst->GetNodeKey());
391 }
392 handle.cached = dst;
393 handle.key = dst->GetNodeKey();
394 return true;
395 }
396 else {
397 const SpatialChunkKey dstKey = dst->GetNodeKey();
398 // 変化無し
399 if (dstKey.code == handle.key.code && dstKey.level == handle.key.level) {
400 handle.cached = dst;
401 return false;
402 }
403 // 実移送せず、(srcKey,dstKey) を積む
404 if (handle.key.code) moverBatch.Add(id, handle.key, dstKey);
405 handle.key = dstKey;
406 handle.cached = dst;
407 return true;
408 }
