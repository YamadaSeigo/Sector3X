23 // .chullbin を読み込んで全部のハルを outHulls に積む
24 // 戻り値: 成功なら true / 失敗なら false
25 inline bool LoadVHACDFile(
26 const std::filesystem::path& binPath,
27 std::vector<VHACDHull>& outHulls,
28 const Math::Vec3f scale = { 1,1,1 },
29 bool flip = false)
30 {
31 outHulls.clear();
32
33 std::ifstream ifs(binPath, std::ios::binary);
34 if (!ifs)
35 {
36 LOG_ERROR("PhysicsConvexHullLoader: Failed to open VHACD file: {%s}", binPath.string().c_str());
37
38 // ファイルが開けない
39 return false;
40 }
41
42 // ----- ヘッダー読込 -----
43 char magic[4];
44 ifs.read(magic, 4);
45 if (!ifs || magic[0] != 'C' || magic[1] != 'V' || magic[2] != 'X' || magic[3] != 'H')
46 {
47 LOG_ERROR("PhysicsConvexHullLoader: Invalid VHACD file format: {%s}", binPath.string().c_str());
48
49 // フォーマット不一致
50 return false;
51 }
52
53 auto read_u32 = [&ifs]() -> uint32_t
54 {
55 uint32_t v{};
56 ifs.read(reinterpret_cast<char*>(&v), sizeof(v));
57 return v;
58 };
59
60 uint32_t version = read_u32();
61 uint32_t hullCount = read_u32();
62 if (!ifs) return false;
63 if (version != 1)
64 {
65 LOG_ERROR("PhysicsConvexHullLoader: Unsupported VHACD file version (%u) in file: {%s}", version, binPath.string().c_str());
66
67 // とりあえず version 1 のみ対応
68 return false;
69 }
70
71 outHulls.reserve(hullCount);
72
73 // ----- 各ハル -----
74 for (uint32_t h = 0; h < hullCount; ++h)
75 {
76 uint32_t vCount = read_u32();
77 uint32_t iCount = read_u32();
78 if (!ifs) return false;
79
80 VHACDHull hull;
81 hull.points.resize(vCount);
82 hull.indices.resize(iCount);
83
84 auto flip_vec3z = [](float& x, float& y, float& z) { x = -x; };
85
86 // 頂点 (vCount * 3 float)
87 for (uint32_t i = 0; i < vCount; ++i)
88 {
89 float xyz[3];
90 ifs.read(reinterpret_cast<char*>(xyz), sizeof(float) * 3);
91 if (!ifs) return false;
92
93 if (flip) flip_vec3z(xyz[0], xyz[1], xyz[2]);
94 hull.points[i] = Vec3f{ xyz[0], xyz[1], xyz[2] } *scale;
95 }
96
97 // インデックス (iCount * uint32)
98 if (iCount > 0)
99 {
100 ifs.read(reinterpret_cast<char*>(hull.indices.data()),
101 sizeof(uint32_t) * iCount);
102 if (!ifs) return false;
103 }
104
105 outHulls.emplace_back(std::move(hull));
106 }
107
108 return true;
109 }
