21// 33x33 グリッドの VB/IB を作成（頂点: float2 gridUV）
22// indexCountPerPatch には 32*32*6 (= 6144) が入る
23//-----------------------------------------------
24inline bool CreateTerrainGrid_33x33(
25 ID3D11Device* device,
26 ComPtr<ID3D11Buffer>& outVB,
27 ComPtr<ID3D11Buffer>& outIB,
28 uint32_t& outIndexCountPerPatch)
29{
30 assert(device);
31 constexpr uint32_t V = 33;
32 constexpr uint32_t Q = V - 1; // 32
33 constexpr uint32_t VertexCount = V * V; // 1089
34 constexpr uint32_t IndexCount = Q * Q * 6; // 6144
35
36 struct Vtx { float uv[2]; };
37
38 std::vector<Vtx> verts(VertexCount);
39 for (uint32_t j = 0; j < V; ++j) {
40 for (uint32_t i = 0; i < V; ++i) {
41 const uint32_t idx = j * V + i;
42 verts[idx].uv[0] = float(i) / float(Q); // [0,1]
43 verts[idx].uv[1] = float(j) / float(Q); // [0,1]
44 }
45 }
46
47 std::vector<uint32_t> indices(IndexCount);
48 uint32_t w = V;
49 uint32_t k = 0;
50 for (uint32_t j = 0; j < Q; ++j) {
51 for (uint32_t i = 0; i < Q; ++i) {
52 uint32_t v0 = j * w + i;
53 uint32_t v1 = v0 + 1;
54 uint32_t v2 = v0 + w;
55 uint32_t v3 = v2 + 1;
56 // 2 Triangles (v0,v2,v1) , (v1,v2,v3) — 風向はお好みで
57 indices[k++] = v0; indices[k++] = v2; indices[k++] = v1;
58 indices[k++] = v1; indices[k++] = v2; indices[k++] = v3;
59 }
60 }
61
62 // VB
63 {
64 D3D11_BUFFER_DESC bd{};
65 bd.BindFlags = D3D11_BIND_VERTEX_BUFFER;
66 bd.ByteWidth = UINT(verts.size() * sizeof(Vtx));
67 bd.Usage = D3D11_USAGE_IMMUTABLE;
68
69 D3D11_SUBRESOURCE_DATA init{};
70 init.pSysMem = verts.data();
71
72 ComPtr<ID3D11Buffer> vb;
73 HRESULT hr = device->CreateBuffer(&bd, &init, vb.GetAddressOf());
74 if (FAILED(hr)) return false;
75 outVB = vb;
76 }
77 // IB
78 {
79 D3D11_BUFFER_DESC bd{};
80 bd.BindFlags = D3D11_BIND_INDEX_BUFFER;
81 bd.ByteWidth = UINT(indices.size() * sizeof(uint32_t));
82 bd.Usage = D3D11_USAGE_IMMUTABLE;
83
84 D3D11_SUBRESOURCE_DATA init{};
85 init.pSysMem = indices.data();
86
87 ComPtr<ID3D11Buffer> ib;
88 HRESULT hr = device->CreateBuffer(&bd, &init, ib.GetAddressOf());
89 if (FAILED(hr)) return false;
90 outIB = ib;
91 }
92
93 outIndexCountPerPatch = IndexCount;
94 return true;
95}

100inline bool CreateFloat1TextureSRV(
101 ID3D11Device* device,
102 uint32_t width, uint32_t height,
103 const float* pixels, uint32_t rowFloatStride, // 1行あたりの float 数（=width が普通）
104 ComPtr<ID3D11ShaderResourceView>& outSRV)
105{
106 assert(device && pixels);
107 D3D11_TEXTURE2D_DESC td{};
108 td.Width = width;
109 td.Height = height;
110 td.MipLevels = 1;
111 td.ArraySize = 1;
112 td.Format = DXGI_FORMAT_R32_FLOAT;
113 td.SampleDesc.Count = 1;
114 td.Usage = D3D11_USAGE_IMMUTABLE;
115 td.BindFlags = D3D11_BIND_SHADER_RESOURCE;
116
117 D3D11_SUBRESOURCE_DATA init{};
118 init.pSysMem = pixels;
119 init.SysMemPitch = rowFloatStride * sizeof(float);
120
121 ComPtr<ID3D11Texture2D> tex;
122 if (FAILED(device->CreateTexture2D(&td, &init, &tex))) return false;
123
124 D3D11_SHADER_RESOURCE_VIEW_DESC sd{};
125 sd.Format = td.Format;
126 sd.ViewDimension = D3D11_SRV_DIMENSION_TEXTURE2D;
127 sd.Texture2D.MipLevels = 1;
128
129 ComPtr<ID3D11ShaderResourceView> srv;
130 if (FAILED(device->CreateShaderResourceView(tex.Get(), &sd, &srv))) return false;
131 outSRV = srv;
132 return true;
133}

135//-----------------------------------------------
136// 画像ファイルから高さSRVを作成（R32_FLOAT・1ch）
137// - PNG/JPG などの LDR → 輝度(Y) を 0..1 にして float 化
138// - DDS で既に 1ch/float の場合はそのまま
139// ※ DirectXTex の導入がある場合はそれを使ってもOK。
140// ここでは WIC 読み込み前提の簡易版（自前変換）例を示します。
141//-----------------------------------------------
142#if __has_include(<wincodec.h>)
143#include <wincodec.h> // WIC
144#pragma comment(lib, "windowscodecs.lib")
145
146inline bool CreateOrLoadHeightSRV_FromFile(
147 ID3D11Device* device,
148 const std::wstring& path,
149 float heightScaleMeters, // テクスチャ値(0..1)→実高さ[m]にスケール
150 ComPtr<ID3D11ShaderResourceView>& outSRV)
151{
152 assert(device);
153 ComPtr<IWICImagingFactory> factory;
154 if (FAILED(CoCreateInstance(CLSID_WICImagingFactory, nullptr, CLSCTX_INPROC_SERVER,
155 IID_PPV_ARGS(&factory)))) return false;
156
157 ComPtr<IWICBitmapDecoder> decoder;
158 if (FAILED(factory->CreateDecoderFromFilename(path.c_str(), nullptr, GENERIC_READ,
159 WICDecodeMetadataCacheOnDemand, &decoder))) return false;
160
161 ComPtr<IWICBitmapFrameDecode> frame;
162 if (FAILED(decoder->GetFrame(0, &frame))) return false;
163
164 // 32bpp BGRA に統一して読み出す
165 ComPtr<IWICFormatConverter> conv;
166 factory->CreateFormatConverter(&conv);
167 if (FAILED(conv->Initialize(frame.Get(), GUID_WICPixelFormat32bppBGRA,
168 WICBitmapDitherTypeNone, nullptr, 0.0, WICBitmapPaletteTypeCustom))) return false;
169
170 UINT w = 0, h = 0;
171 conv->GetSize(&w, &h);
172 std::vector<uint8_t> bgra(size_t(w) * h * 4);
173 if (FAILED(conv->CopyPixels(nullptr, w * 4, UINT(bgra.size()), bgra.data()))) return false;
174
175 // BGRA → 輝度（0..1）→ 高さ[m]（R32_FLOAT 1ch）
176 std::vector<float> height(size_t(w) * h);
177 for (size_t i = 0, n = size_t(w) * h; i < n; ++i) {
178 const uint8_t B = bgra[i * 4 + 0];
179 const uint8_t G = bgra[i * 4 + 1];
180 const uint8_t R = bgra[i * 4 + 2];
181 // Rec.709 輝度
182 float y = (0.2126f * (R / 255.0f) + 0.7152f * (G / 255.0f) + 0.0722f * (B / 255.0f));
183 height[i] = y * heightScaleMeters;
184 }
185
186 return CreateFloat1TextureSRV(device, w, h, height.data(), w, outSRV);
187}
188#endif

190//-----------------------------------------------
191// 手続き生成で高さSRVを作成（R32_FLOAT・1ch）
192// gen(x,y) はワールド/テクスチャ座標に応じた高さ[m]を返す関数
193//-----------------------------------------------
194template<class HeightGenerator>
195inline bool CreateOrLoadHeightSRV_Procedural(
196 ID3D11Device* device,
197 uint32_t width, uint32_t height,
198 HeightGenerator&& gen, // float gen(float x, float y)
199 ComPtr<ID3D11ShaderResourceView>& outSRV)
200{
201 std::vector<float> hmap(size_t(width) * height);
202 for (uint32_t y = 0; y < height; ++y) {
203 for (uint32_t x = 0; x < width; ++x) {
204 hmap[size_t(y) * width + x] = gen(x, y);
205 }
206 }
207 return CreateFloat1TextureSRV(device, width, height, hmap.data(), width, outSRV);
208}

213struct HeightCreateDesc {
214 // いずれかを使用
215 std::wstring filePath; // 画像から
216 uint32_t procWidth = 0; // 手続き生成
217 uint32_t procHeight = 0;
218
219 // 画像→高さのスケール [m]
220 float heightScaleMeters = 100.0f;
221
222 // 手続き生成用（オプション）
223 // 例：Perlin パラメータなどを捕まえたラムダを渡す
224 // float operator()(uint32_t x, uint32_t y) を満たす関数オブジェクト
225 std::function<float(uint32_t, uint32_t)> generator;
226};
227
228inline bool CreateOrLoadHeightSRV(
229 ID3D11Device* device,
230 const HeightCreateDesc& desc,
231 ComPtr<ID3D11ShaderResourceView>& outSRV)
232{
233#if __has_include(<wincodec.h>)
234 if (!desc.filePath.empty()) {
235 return CreateOrLoadHeightSRV_FromFile(device, desc.filePath, desc.heightScaleMeters, outSRV);
236 }
237#endif
238 if (desc.procWidth > 0 && desc.procHeight > 0 && desc.generator) {
239 return CreateOrLoadHeightSRV_Procedural(device, desc.procWidth, desc.procHeight, desc.generator, outSRV);
240 }
241 return false;
242}
