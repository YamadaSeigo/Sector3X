43 EntityID Create() {

66 void Destroy(EntityID id) {
67 if (id.index >= maxEntities) [[unlikely]] return;
68
69 // Invalidate old ID
70 generations[id.index].fetch_add(1, std::memory_order_acq_rel);
71
72 // Reuse the index

81
