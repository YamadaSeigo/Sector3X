105 std::optional<SpatialChunk*> GetChunk(Math::Vec3f p,
106 EOutOfBoundsPolicy /*policy*/ = EOutOfBoundsPolicy::ClampToEdge) noexcept
107 {
108 if (root < 0) return std::nullopt;
109 int32_t n = root;
110 while (n >= 0) {
111 const Node& node = nodes[n];
112 if (!node.box.contains(p)) return std::nullopt; // 根で含まなければ終了
113 if (node.IsLeaf()) {
114 return &leaves[node.leaf].chunk;
115 }
116 // どちらかに含まれるまで潜る（両方含む場合は左優先）
117 const Node& L = nodes[node.left];
118 const Node& R = nodes[node.right];
119 bool inL = L.box.contains(p);
120 bool inR = R.box.contains(p);
121 n = inL ? node.left : (inR ? node.right : -1);
122 }
123 return std::nullopt;
124 }

176 uint32_t CullChunkLine(const Math::Frustumf& fr,
177 Math::Vec3f cp,
178 Debug::LineVertex* outLine,
179 uint32_t capacity,
180 uint32_t displayCount) const noexcept
181 {
182 if (root < 0) return 0;
183 const float maxLen = float(displayCount);
184 uint32_t written = 0;
185
186 auto push_edge = [&](const Math::Vec3f& a, const Math::Vec3f& b, uint32_t rgba) {
187 if (capacity - written < 2) return false;
188 outLine[written + 0] = { a, rgba };
189 outLine[written + 1] = { b, rgba };
190 written += 2;
191 return true;
192 };
193
194 auto draw_box = [&](const AABB& b) {
195 const Math::Vec3f c = b.center();
196 const float len = (c - cp).length();
197 if (len > maxLen) return; // 距離制限（任意単位）
198 const uint32_t col = Math::LerpColor(0xFFFFFFFFu, 0x000000FFu, (std::min)(1.0f, len / maxLen));
199
200 const Math::Vec3f mn = b.lb;
201 const Math::Vec3f mx = b.ub;
202 Math::Vec3f v000{ mn.x, mn.y, mn.z };
203 Math::Vec3f v001{ mn.x, mn.y, mx.z };
204 Math::Vec3f v010{ mn.x, mx.y, mn.z };
205 Math::Vec3f v011{ mn.x, mx.y, mx.z };
206 Math::Vec3f v100{ mx.x, mn.y, mn.z };
207 Math::Vec3f v101{ mx.x, mn.y, mx.z };
208 Math::Vec3f v110{ mx.x, mx.y, mn.z };
209 Math::Vec3f v111{ mx.x, mx.y, mx.z };
210
211 push_edge(v000, v001, col); push_edge(v000, v010, col); push_edge(v000, v100, col);
212 push_edge(v111, v101, col); push_edge(v111, v110, col); push_edge(v111, v011, col);
213 push_edge(v010, v011, col); push_edge(v010, v110, col); push_edge(v100, v101, col);
214 push_edge(v100, v110, col); push_edge(v001, v011, col); push_edge(v001, v101, col);
215 };
216
217 // 走査
218 std::vector<int32_t> stack;
219 stack.reserve(64);
220 stack.push_back(root);
221 while (!stack.empty() && written + 24 <= capacity) {
222 int32_t n = stack.back(); stack.pop_back();
223 const Node& node = nodes[n];
224 if (!fr.IntersectsAABB(node.box.center(), node.box.extent())) continue;
225 if (node.IsLeaf()) { draw_box(leaves[node.leaf].box); }
226 else { stack.push_back(node.left); stack.push_back(node.right); }
227 }
228 return written;
229 }

245 // ====== 再帰構築（中央値分割：最大軸） ======
246 template<class It>
247 int32_t BuildRecursive(It first, It last)
248 {
249 const int32_t count = static_cast<int32_t>(last - first);
250 int32_t nodeIdx = (int32_t)nodes.size();
251 nodes.push_back(Node{});
252 Node& node = nodes.back();
253
254 // 範囲のAABB
255 AABB bounds = leaves[*first].box;
256 for (It it = first + 1; it != last; ++it) bounds = AABB::Union(bounds, leaves[*it].box);
257 node.box = bounds;
258
259 if (count == 1) {
260 node.leaf = *first;
261 return nodeIdx;
262 }
263
264 // 分割軸 = 最長軸
265 Math::Vec3f ext = bounds.extent();
266 int axis = (ext.x >= ext.y && ext.x >= ext.z) ? 0 : (ext.y >= ext.z ? 1 : 2);
267 auto mid = first + count / 2;
268 std::nth_element(first, mid, last, [&](int32_t a, int32_t b) {
269 const float ca = leaves[a].box.center()[axis];
270 const float cb = leaves[b].box.center()[axis];
271 return ca < cb;
272 });
273
274 int32_t L = BuildRecursive(first, mid);
275 int32_t R = BuildRecursive(mid, last);
276 nodes[L].parent = nodes[R].parent = nodeIdx;
277 node.left = L; node.right = R;
278 return nodeIdx;
279 }

281 void RefitDFS(int32_t n) noexcept
282 {
283 Node& node = nodes[n];
284 if (node.IsLeaf()) {
285 node.box = leaves[node.leaf].box;
286 return;
287 }
288 RefitDFS(node.left);
289 RefitDFS(node.right);
290 node.box = AABB::Union(nodes[node.left].box, nodes[node.right].box);
291 }

295 {
296 const Node& node = nodes[n];

298 if (node.IsLeaf()) { out.push_back(const_cast<SpatialChunk*>(&leaves[node.leaf].chunk)); return; }
299 CullDFS(node.left, fr, out);

301 }
