253 static Vec3<T> makeRight(const Vec3<T>& up, const Vec3<T>& forward) noexcept {
254 if constexpr (std::is_same_v<Convention, RH_ZBackward>) {
255 return up.cross(forward).normalized(); // 右手系: right = up × forward
