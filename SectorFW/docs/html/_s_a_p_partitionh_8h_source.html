156 uint32_t CullChunkLine(const Math::Frustumf& fr,
157 Math::Vec3f cp,
158 Debug::LineVertex* outLine,
159 uint32_t capacity,
160 uint32_t displayCount) const noexcept
161 {
162 const float maxLen = float(displayCount);
163 uint32_t written = 0;
164
165 auto push_edge = [&](const Math::Vec3f& a, const Math::Vec3f& b, uint32_t rgba) {
166 if (capacity - written < 2) return false;
167 outLine[written + 0] = { a, rgba };
168 outLine[written + 1] = { b, rgba };
169 written += 2;
170 return true;
171 };
172
173 auto draw_box = [&](const AABB& box) {
174 const Math::Vec3f c = box.center();
175 const float len = (c - cp).length();
176 if (len > maxLen) return; // 距離制限
177 const uint32_t col = Math::LerpColor(0xFFFFFFFFu, 0x000000FFu, (std::min)(1.0f, len / maxLen));
178
179 const Math::Vec3f mn = box.lb, mx = box.ub;
180 Math::Vec3f v000{ mn.x, mn.y, mn.z };
181 Math::Vec3f v001{ mn.x, mn.y, mx.z };
182 Math::Vec3f v010{ mn.x, mx.y, mn.z };
183 Math::Vec3f v011{ mn.x, mx.y, mx.z };
184 Math::Vec3f v100{ mx.x, mn.y, mn.z };
185 Math::Vec3f v101{ mx.x, mn.y, mx.z };
186 Math::Vec3f v110{ mx.x, mx.y, mn.z };
187 Math::Vec3f v111{ mx.x, mx.y, mx.z };
188
189 push_edge(v000, v001, col); push_edge(v000, v010, col); push_edge(v000, v100, col);
190 push_edge(v111, v101, col); push_edge(v111, v110, col); push_edge(v111, v011, col);
191 push_edge(v010, v011, col); push_edge(v010, v110, col); push_edge(v100, v101, col);
192 push_edge(v100, v110, col); push_edge(v001, v011, col); push_edge(v001, v101, col);
193 };
194
195 for (const auto& b : bodies) {
196 if (!fr.IntersectsAABB(b.box.center(), b.box.extent())) continue;
197 if (written + 24 > capacity) break;
198 draw_box(b.box);
199 }
200 return written;
201 }

208 template<class PairFn>
209 void EnumerateOverlapPairs(PairFn&& fn) noexcept
210 {
211 ensureRebuilt();
212 const auto& order = orderPrimary();
213 // アクティブリスト方式
214 std::vector<uint32_t> active;
215 active.reserve(128);
216 for (uint32_t i = 0; i < order.size(); ++i) {
217 const uint32_t aIdx = order[i];
218 Body& A = bodies[aIdx];
219 const float aMin = A.minProj[(int)primary];
220 const float aMax = A.maxProj[(int)primary];
221
222 // active から、終端がAの始端よりも小さいものを除去
223 auto it = active.begin();
224 while (it != active.end()) {
225 if (bodies[*it].maxProj[(int)primary] < aMin) it = active.erase(it);
226 else ++it;
227 }
228
229 // active と交差候補
230 for (uint32_t j : active) {
231 Body& B = bodies[j];
232 if (A.box.Overlaps(B.box)) {
233 fn(aIdx, j); // A と B は重なり候補
234 }
235 }
236
237 active.push_back(aIdx);
238 }
239 }
