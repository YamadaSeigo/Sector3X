316 void PreUpdate(double dt) override {
317 m_accum += static_cast<float>(dt);
318
319 while (m_accum + 1e-6f >= plan.fixed_dt) { // 浮動誤差対策の微小マージン
320 DrainBudgetToDevice(plan.command_budget_per_step); // ここで一括適用
321 m_device.Step(plan.fixed_dt, plan.substeps);
322
323 // スナップショットを組み立てる
324 m_device.BuildSnapshot(m_snapshot);
325 m_prevSnapshot = std::move(m_currSnapshot); // 前フレームのスナップショットを保存
326 m_currSnapshot = std::move(m_snapshot); // 今回のスナップショットを更新
327
328 m_accum -= plan.fixed_dt;
329 }

427 template<class T>
428 bool TryEnqueue(const T& c) {
429 const PhysicsCommand& cmd = c;
430 // 失敗（満杯）の場合はリトライ or 一時的にブロッキングに切替など、運用ポリシー次第
431 return m_queue.push(cmd);
432 }

434 template<class T>
435 bool Enqueue(const T& c) {
436 const PhysicsCommand& cmd = c;
437
438 // まず ring に入るなら入れる
439 if (m_queue.push(cmd)) return true;
440
441 // 溢れたら pending に積んで “受理” は返す
442 m_pendingCmds.push_back(cmd);
443 return true;
444 }
445
446 void DrainAllToDevice() {
447 while (auto cmd = m_queue.pop()) {
448 m_device.ApplyCommand(*cmd);
449 }
450 }
451
452 void DrainBudgetToDevice(int budget) {
453 int used = 0;
454
455 // 1) ring を先に流す（順序が自然）
456 while (used < budget) {
457 auto cmd = m_queue.pop();
458 if (!cmd) break;
459 m_device.ApplyCommand(*cmd);
460 ++used;
461 }
462
463 // 2) pending を残り枠で流す
464 while (used < budget && m_pendingHead < m_pendingCmds.size()) {
465 m_device.ApplyCommand(m_pendingCmds[m_pendingHead]);
466 ++m_pendingHead;
467 ++used;
468 }
469
470 // pending の先頭を消費し続けているので、適当に詰め直し
471 if (m_pendingHead > 1024 && m_pendingHead * 2 > m_pendingCmds.size()) {
472 m_pendingCmds.erase(m_pendingCmds.begin(), m_pendingCmds.begin() + m_pendingHead);
473 m_pendingHead = 0;
474 }
475 }
