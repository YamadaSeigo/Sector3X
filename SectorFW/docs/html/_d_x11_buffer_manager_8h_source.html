138 BufferData CreateResource(const BufferCreateDesc& desc, BufferHandle h) {
139 BufferData out{};
140
141 D3D11_BUFFER_DESC bd = {};
142 bd.Usage = desc.usage;
143 bd.BindFlags = desc.bindFlags;
144 if (desc.usage == D3D11_USAGE_DYNAMIC) {
145 bd.CPUAccessFlags = D3D11_CPU_ACCESS_WRITE;
146 }
147 else if (desc.usage == D3D11_USAGE_STAGING) {
148 bd.CPUAccessFlags = desc.cpuAccessFlags; // CPU アクセスフラグを指定
149 }
150 bd.ByteWidth = static_cast<UINT>(desc.size);
151 bd.MiscFlags = desc.miscFlags;
152 if (bd.MiscFlags & D3D11_RESOURCE_MISC_BUFFER_STRUCTURED) {
153 bd.StructureByteStride = desc.structureByteStride;
154 }
155
156 assert(desc.bindFlags != D3D11_BIND_CONSTANT_BUFFER || desc.size % 16 == 0 && "定数バッファのサイズが16倍数ではありません");
157
158 HRESULT hr;
159 if (desc.initialData == nullptr) {
160 if (bd.Usage == D3D11_USAGE_IMMUTABLE) {
161 assert(false && "Immutable buffer must have initial data.");
162 }
163
164 hr = device->CreateBuffer(&bd, nullptr, &out.buffer);
165 }
166 else {
167 D3D11_SUBRESOURCE_DATA initData = {};
168 initData.pSysMem = desc.initialData;
169 hr = device->CreateBuffer(&bd, &initData, &out.buffer);
170 }
171
172 if (FAILED(hr)) {
173 assert(false && "Failed to create constant buffer");
174 }
175
176 // SRV が必要なら作成
177 if (desc.bindFlags & D3D11_BIND_SHADER_RESOURCE) {
178 D3D11_SHADER_RESOURCE_VIEW_DESC sd{};
179 if (desc.miscFlags & D3D11_RESOURCE_MISC_BUFFER_ALLOW_RAW_VIEWS) {
180 // ByteAddressBuffer (raw)
181 sd.Format = DXGI_FORMAT_R32_TYPELESS;
182 sd.ViewDimension = D3D11_SRV_DIMENSION_BUFFEREX;
183 sd.BufferEx.FirstElement = 0;
184 sd.BufferEx.NumElements = desc.size / 4;
185 sd.BufferEx.Flags = D3D11_BUFFEREX_SRV_FLAG_RAW;
186 }
187 else {
188 // StructuredBuffer
189 sd.Format = DXGI_FORMAT_UNKNOWN;
190 sd.ViewDimension = D3D11_SRV_DIMENSION_BUFFER;
191 sd.Buffer.FirstElement = 0;
192 sd.Buffer.NumElements = desc.size / desc.structureByteStride;
193 }
194 HRESULT hr = device->CreateShaderResourceView(out.buffer.Get(), &sd, &out.srv);
195 assert(SUCCEEDED(hr));
196 }
197
198 // UAV が必要なら作成（Compute 用）
199 if (desc.bindFlags & D3D11_BIND_UNORDERED_ACCESS) {
200 D3D11_UNORDERED_ACCESS_VIEW_DESC ud{};
201 if (desc.miscFlags & D3D11_RESOURCE_MISC_BUFFER_ALLOW_RAW_VIEWS) {
202 ud.Format = DXGI_FORMAT_R32_TYPELESS;
203 ud.ViewDimension = D3D11_UAV_DIMENSION_BUFFER;
204 ud.Buffer.FirstElement = 0;
205 ud.Buffer.NumElements = desc.size / 4;
206 ud.Buffer.Flags = D3D11_BUFFER_UAV_FLAG_RAW;
207 }
208 else {
209 ud.Format = DXGI_FORMAT_UNKNOWN;
210 ud.ViewDimension = D3D11_UAV_DIMENSION_BUFFER;
211 ud.Buffer.FirstElement = 0;
212 ud.Buffer.NumElements = desc.size / desc.structureByteStride;
213 ud.Buffer.Flags = 0;
214 }
215 HRESULT hr = device->CreateUnorderedAccessView(out.buffer.Get(), &ud, &out.uav);
216 assert(SUCCEEDED(hr));
217 }
218
219 return out;
220 }

238 BufferHandle AcquireWithContent(const void* data, uint32_t size) {
239 assert(data && size > 0);
240
241 BufferCacheKey key{ HashBufferContent(data, size), size };
242 if (auto it = cbvCache.find(key); it != cbvCache.end()) {
243 AddRef(it->second);
244 return it->second;
245 }
246 // 初回作成：匿名名で Add（Add が +1 を返す）
247 BufferHandle h;
248 BufferCreateDesc desc = {
249 .name = "auto_cb_" + std::to_string(key.hash),
250 .size = size,
251 .initialData = data
252 };
253 Add(desc, h);
254
255 cbvCache[key] = h;
256 handleToCacheKey[h.index] = key;
257 return h;
258 }

263 void UpdateBuffer(const BufferUpdateDesc& desc, uint16_t slot) noexcept {
264 uint32_t count = pendingCount[slot].fetch_add(1, std::memory_order_acq_rel);
265 if (count >= MAX_PENDING_UPDATE_NUM) {
266 //LOG_ERROR("最大更新処理数に達しました");
267 return;
268 }
269
270 pendingUpdates[slot][count] = desc;
271 }

275 void PendingUpdates(size_t frameIndex) {
276 uint16_t slot = frameIndex % RENDER_BUFFER_COUNT;
277 uint32_t count = pendingCount[slot].load(std::memory_order_relaxed);
278
279 if (count > 0) {
280 auto& pendings = pendingUpdates[slot];
281
282 for (uint32_t i = 0; i < count; ++i) {
283 auto& update = pendings[i];
284 assert(update.isValid() && "バッファの更新情報が正しくありません");
285
286 D3D11_MAPPED_SUBRESOURCE mapped;
287 HRESULT hr = context->Map(update.buffer.Get(), 0, D3D11_MAP_WRITE_DISCARD, 0, &mapped);
288 if (SUCCEEDED(hr)) [[likely]] {
289 if (update.customUpdateFunc) {
290 update.customUpdateFunc(mapped.pData, update.data, update.size);
291 }
292 else {
293 memcpy(mapped.pData, update.data, update.size);
294 }
295 context->Unmap(update.buffer.Get(), 0);
296 }
297 else {
298 assert(false && "Failed to map constant buffer for update");
299 }
300 if (update.data && update.isDelete) {
301 update.isArray ? delete[] update.data : delete update.data;
302 update.isDelete = false;
303 }
304 }
305
306 pendingCount[slot].store(0, std::memory_order_release);
307 }
