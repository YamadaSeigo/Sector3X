254 // AABBFrontFaceQuad を 2トライで提出できるバッチに変換
255 inline MocQuadBatch ConvertAABBFrontFaceQuadToMoc(
256 const Math::Vec4f* clips,
257 const SFW::Math::Matrix<4, 4, float>& viewProj,
258 float nearClipW /* = moc->GetNearClipPlane() と一致させるのが理想 */
259 )
260 {
261 using Vec3 = SFW::Math::Vec3f;
262 using Vec4 = SFW::Math::Vec4f;
263
264 MocQuadBatch out{};
265
266 std::memcpy(out.clip, clips, sizeof(Vec4) * 4);
267
268 // 2) 近クリップ整合（全頂点が near より奥側なら無効）
269 // MOC は depth = 1/w（逆深度）なので、near と合わせるなら w > near を可視空間とみなす。
270 int countFront = 0;
271 for (int i = 0; i < 4; ++i) if (out.clip[i].w > nearClipW) ++countFront;
272 if (countFront == 0) {
273 out.valid = false;
274 return out;
275 }
276
277 // 3) スクリーン空間の巻き方向を安定させる（CCW想定）
278 // NDC の x,y を使って符号付き面積で CCW を判定し、CW なら対角を入れ替える。
279 auto ndcXY_stable = [&](int i) {
280 // CCW判定用途なので「割り算が爆発しない」ことを優先
281 const float w = (std::max)(out.clip[i].w, nearClipW); // near でクランプ
282 const float invw = 1.0f / w;
283 return SFW::Math::Vec2f{ out.clip[i].x * invw, out.clip[i].y * invw };
284 };
285
286 auto triArea2 = [](SFW::Math::Vec2f p, SFW::Math::Vec2f q, SFW::Math::Vec2f r) {
287 // 2倍面積（z成分）: (q - p) × (r - p)
288 return (q.x - p.x) * (r.y - p.y) - (q.y - p.y) * (r.x - p.x);
289 };
290
291 const auto a = ndcXY_stable(0), b = ndcXY_stable(1), c = ndcXY_stable(2), d = ndcXY_stable(3);
292 float area = triArea2(a, b, c) + triArea2(a, c, d);
293 bool ccw = (area > 0.0f);
294 // CCW でなければ 1 と 2 を入れ替える（0,1,2 / 2,1,3 -> 0,2,1 / 1,2,3）
295 if (!ccw) {
296 std::swap(out.clip[0], out.clip[2]);
297 }
298
299 // 4) インデックス（CCW）
300 out.indices[0] = 0; out.indices[1] = 2; out.indices[2] = 1;
301 out.indices[3] = 2; out.indices[4] = 0; out.indices[5] = 3;
302
303 return out;
304 }
